    59→function divisionShort(code: string | null | undefined): string {
    60→  if (!code) return "—";

    64→function stripHtml(html: string): string {

    68→function sgLabel(grade: string | null | undefined, step: string | null | undefined): string {
    69→  if (!grade) return "—";
    70→  return step ? `SG${grade}/${step}` : `SG${grade}`;

    73→function pageWindow(current: number, total: number): (number | "…")[] {
    74→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1);
    75→  const out: (number | "…")[] = [1];
    76→  const lo = Math.max(2, current - 1);
    77→  const hi = Math.min(total - 1, current + 1);
    78→  if (lo > 2) out.push("…");
    79→  for (let i = lo; i <= hi; i++) out.push(i);
    80→  if (hi < total - 1) out.push("…");
    81→  out.push(total);
    82→  return out;
    83→}

    85→/** % of the application window elapsed between publish and deadline (0–100). */
    86→function deadlineProgress(job: JobWire): number {
    87→  const start = new Date(job.publishDate ?? job.publishedAt).getTime();
    88→  const end = job.deadlineDate ? new Date(job.deadlineDate).getTime() : Number.NaN;
    89→  if (Number.isNaN(start) || Number.isNaN(end) || end <= start) return job.deadlineDate ? 100 : 0;
    90→  const pct = ((Date.now() - start) / (end - start)) * 100;
    91→  return Math.min(100, Math.max(0, pct));
    92→}

   537→  const detailParam = route.params.job ?? null;
   538→  const { user } = useSession();
   539→
   540→  // Shell contract (responsive fix 3-a): PublicShell's <main> is a bare flex
   541→  // column (no padding), so the anonymous board paints its own 1600px container
   542→  // + gutters + vertical rhythm. AppShell's <main> already carries the exact
   543→  // same container (px-4 py-6 sm:px-6 sm:py-8 lg:px-8), so signed-in views must
   544→  // not repeat it (double gutters ~128px at lg) — a plain w-full wrapper keeps
   545→  // identical spacing since AppShell main provides the same py.
   546→  const shellPad = user
   547→    ? "w-full"
   548→    : "mx-auto w-full max-w-[1600px] px-4 py-6 sm:px-6 sm:py-8 lg:px-8";
   549→
   550→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   551→  const [error, setError] = useState<string | null>(null);
   552→  const [query, setQuery] = useState("");
   553→  const [divisionFilter, setDivisionFilter] = useState<string[]>([]);
   554→  const [sort, setSort] = useState<SortKey>("newest");
   555→  const [page, setPage] = useState(1);
   556→  const [expandedId, setExpandedId] = useState<number | null>(null);
   557→  const [filtersOpen, setFiltersOpen] = useState(false);
   558→  const [cancelling, setCancelling] = useState(false);
   559→  const lastJson = useRef("");
   560→
   561→  const { requestApply, dialogs } = useApplyFlow();
   562→
   563→  const load = useCallback(async (silent: boolean) => {
   564→    try {
   565→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   566→      const json = JSON.stringify(data);
   567→      if (!silent || json !== lastJson.current) {
   568→        lastJson.current = json;
   569→        setJobs(data);
   570→      }
   571→      setError(null);
   572→    } catch (e) {
   573→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   574→    }
   575→  }, []);
   576→
   577→  // Mount + 20s silent poll + refetch on focus (spec §13 cadence).
   578→  useEffect(() => {
   579→    void load(false);
   580→    const timer = setInterval(() => {
   581→      if (!document.hidden) void load(true);
   582→    }, 20_000);
   583→    const onFocus = () => void load(true);
   584→    window.addEventListener("focus", onFocus);
   585→    return () => {
   586→      clearInterval(timer);
   587→      window.removeEventListener("focus", onFocus);
   588→    };
   589→  }, [load]);
   590→
   591→  // Facet counts over the full corpus.
   592→  const facets = useMemo(() => {
   593→    const map = new Map<string, number>();
   594→    for (const j of jobs ?? []) {
   595→      const code = j.position?.division;
   596→      if (!code) continue;
   597→      map.set(code, (map.get(code) ?? 0) + 1);
   598→    }
   599→    return [...map.entries()].sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0]));
   600→  }, [jobs]);
   601→
   602→  // Filtering + sorting.
   603→  const filtered = useMemo(() => {
   604→    let list = jobs ?? [];
   605→    const needle = query.trim().toLowerCase();
   606→    if (needle) {
   607→      list = list.filter((j) =>
   608→        `${j.title} ${j.position?.positionTitle ?? ""} ${j.position?.itemNumber ?? ""}`
   609→          .toLowerCase()
   610→          .includes(needle)
   611→      );
   612→    }
   613→    if (divisionFilter.length > 0) {
   614→      list = list.filter((j) => divisionFilter.includes(j.position?.division ?? ""));
   615→    }
   616→    const deadlineTime = (j: JobWire) =>
   617→      j.deadlineDate ? new Date(j.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   618→    const sorted = [...list];
   619→    if (sort === "newest") {
   620→      sorted.sort(
   621→        (a, b) => new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime()
   622→      );
   623→    } else if (sort === "deadline") {
   624→      sorted.sort((a, b) => deadlineTime(a) - deadlineTime(b));
   625→    } else {
   626→      sorted.sort((a, b) => (b.position?.salaryAmount ?? -1) - (a.position?.salaryAmount ?? -1));
   627→    }
   628→    return sorted;
   629→  }, [jobs, query, divisionFilter, sort]);
   630→
   631→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   632→  const current = Math.min(page, totalPages);
   633→  const pageItems = filtered.slice((current - 1) * PAGE_SIZE, current * PAGE_SIZE);
   634→
   635→  const changeQuery = (v: string) => {
   636→    setQuery(v);
   637→    setPage(1);
   638→  };
   639→  const toggleDivision = (code: string) => {
   640→    setDivisionFilter((prev) =>
   641→      prev.includes(code) ? prev.filter((c) => c !== code) : [...prev, code]
   642→    );
   643→    setPage(1);
   644→  };
   645→  const clearFilters = () => {
   646→    setQuery("");
   647→    setDivisionFilter([]);
   648→    setPage(1);
   649→  };
   650→  const toggleExpanded = (id: number) => setExpandedId((prev) => (prev === id ? null : id));
   651→  const openDetail = (job: JobWire) => navigate("jobs", { job: String(job.id) });
   652→
   653→  const applyTo = useCallback(
   654→    (job: JobWire) => {
   655→      requestApply(job);
   656→    },
   657→    [requestApply]
   658→  );
   659→
   660→  // Cancel application (only while status "Applied" — spec §7.7).
   661→  const appliedId = useMemo(() => {
   662→    const detail = detailParam ? (jobs ?? []).find((j) => j.id === Number(detailParam)) : null;
   663→    return detail?.applications?.[0]?.id ?? null;
   664→  }, [detailParam, jobs]);
   665→
   666→  const cancelApplication = useCallback(async () => {
   667→    if (appliedId == null) return;
   668→    setCancelling(true);
   669→    try {
   670→      await apiFetch(`/api/applications/${appliedId}`, { method: "DELETE" });
   671→      toast.success("Application cancelled", {
   672→        description: "Your application was withdrawn. You can apply again while the position is open.",
   673→      });
   674→      await load(false);
   675→    } catch (e) {
   676→      toast.error(e instanceof Error ? e.message : "Could not cancel the application");
   677→    } finally {
   678→      setCancelling(false);
   679→    }
   680→  }, [appliedId, load]);
   681→
   682→  // ── Detail mode ───────────────────────────────────────────────────────────
   683→  if (detailParam && /^\d+$/.test(detailParam)) {
   684→    const detailId = Number(detailParam);
   685→    const detail = jobs?.find((j) => j.id === detailId) ?? null;
   686→
   687→    return (
   688→      <div className={shellPad}>
   689→        <button
   690→          onClick={() => navigate("jobs")}
   691→          className="mb-5 flex min-h-[44px] items-center gap-2 text-sm text-stone hover:text-ink"
   692→        >
   693→          <ArrowLeft className="h-4 w-4" aria-hidden="true" />
   694→          Back to Positions
   695→        </button>
   696→
   697→        {jobs === null && error === null ? (
   698→          <div className="space-y-6">
   699→            <div className="dlg-card space-y-4 p-8">
   700→              <Skeleton className="h-6 w-32" />
   701→              <Skeleton className="h-10 w-2/3" />
   702→              <Skeleton className="h-4 w-1/2" />
   703→            </div>
   704→            <div className="dlg-card space-y-3 p-8">
   705→              <Skeleton className="h-5 w-48" />
   706→              <Skeleton className="h-4 w-full" />
   707→              <Skeleton className="h-4 w-full" />
   708→              <Skeleton className="h-4 w-3/4" />
   709→            </div>
   710→          </div>
   711→        ) : error !== null ? (
   712→          <ErrorCard message={error} onRetry={() => void load(false)} />
   713→        ) : detail === null ? (
   714→          <div className="dlg-card p-10 text-center">
   715→            <p className="font-display text-xl text-carbon">Position not found</p>
   716→            <p className="mt-2 text-sm text-stone">
   717→              This position may have closed or is no longer published.
   718→            </p>
   719→            <button onClick={() => navigate("jobs")} className="dlg-cta mt-5 min-h-[44px] px-7 text-sm">
   720→              Back to Positions
   721→            </button>
   722→          </div>
   723→        ) : (
   724→          // Two-column from lg (matches the filter rail breakpoint): below
   725→          // xl the sticky summary used to render AFTER the whole long
   726→          // description, pushing the primary Apply CTA a full scroll away
   727→          // on exactly-lg desktops.
   728→          <div className="grid grid-cols-1 items-start gap-6 lg:grid-cols-[1fr_340px]">
   729→            <DetailBody job={detail} />
   730→            {/* Session-aware sticky offset: the public SiteHeader is sticky
   731→                h-16, the signed-in shell has no sticky chrome at lg+. */}
   732→            <aside className={`lg:sticky ${user ? "lg:top-6" : "lg:top-20"}`}>
   733→              <SummaryCard
   734→                job={detail}
   735→                onApply={applyTo}
   736→                onCancel={() => void cancelApplication()}
   737→                cancelling={cancelling}
   738→              />
   739→            </aside>
   740→          </div>
   741→        )}
   742→
   743→        {dialogs}
   744→      </div>
   745→    );
   746→  }
   747→
   748→  // ── List mode ─────────────────────────────────────────────────────────────
   749→  const activeFilterCount = (query.trim() ? 1 : 0) + divisionFilter.length;
   750→
   751→  // One rail definition, two mount points (mobile collapsible + lg sidebar).
   752→  // The id suffix keeps the duplicate DOM valid — two inputs must never share
   753→  // one id (label htmlFor / a11y lookups would hit the hidden copy).
   754→  const filterRail = (sfx: string) => (
   755→    <div className="dlg-card p-4">
   756→      <label htmlFor={`job-search-${sfx}`} className="sr-only">
   757→        Search positions
   758→      </label>
   759→      <div className="relative">
   760→        <Search
   761→          className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-pebble"
   762→          aria-hidden="true"
   763→        />
   764→        <Input
   765→          id={`job-search-${sfx}`}
   766→          value={query}
   767→          onChange={(e) => changeQuery(e.target.value)}
   768→          placeholder="Search positions…"
   769→          className="dlg-input min-h-[44px] pl-9"
   770→        />
   771→      </div>
   772→
   773→      <p className="mt-5 px-1 text-xs font-medium uppercase tracking-wider text-pebble">Division</p>
   774→      <div className="mt-2 space-y-0.5">
   775→        {facets.map(([code, count]) => (
   776→          <label
   777→            key={code}
   778→            className="flex min-h-[44px] cursor-pointer items-center gap-3 rounded-none px-2 hover:bg-fog"
   779→          >
   780→            <Checkbox
   781→              checked={divisionFilter.includes(code)}
   782→              onCheckedChange={() => toggleDivision(code)}
   783→            />
   784→            <span className="min-w-0 flex-1 truncate text-sm text-stone" title={divisionName(code)}>
   785→              {divisionShort(code)}
   786→            </span>
   787→            <span className="text-xs text-pebble">{count}</span>
   788→          </label>
   789→        ))}
   790→        {facets.length === 0 && <p className="px-2 py-2 text-sm text-pebble">No divisions on file</p>}
   791→      </div>
   792→
   793→      {(query.trim() || divisionFilter.length > 0) && (
   794→        <button
   795→          onClick={clearFilters}
   796→          className="min-h-[44px] px-2 text-sm text-stone underline underline-offset-4 hover:text-ink"
   797→        >
   798→          Clear filters
   799→        </button>
   800→      )}
   801→    </div>
   802→  );
   803→
   804→  let content: React.ReactNode;
   805→  if (jobs === null && error === null) {
   806→    content = (
   807→      <div className="space-y-4">
   808→        {[0, 1, 2, 3].map((i) => (
   809→          <div key={i} className="dlg-card p-6">
   810→            <Skeleton className="h-5 w-24" />
   811→            <Skeleton className="mt-3 h-6 w-2/3" />
   812→            <Skeleton className="mt-2 h-4 w-full" />
   813→          </div>
   814→        ))}
   815→      </div>
   816→    );
   817→  } else if (error !== null && jobs === null) {
   818→    content = <ErrorCard message={error} onRetry={() => void load(false)} />;
   819→  } else if ((jobs ?? []).length === 0) {
   820→    content = (
   821→      <div className="dlg-card p-10 text-center">
   822→        <p className="font-display text-xl text-carbon">No open positions</p>
   823→        <p className="mt-2 text-sm text-stone">
   824→          There are no published postings at the moment — check back soon.
   825→        </p>
   826→      </div>
   827→    );
   828→  } else if (filtered.length === 0) {
   829→    content = (
   830→      <div className="dlg-card p-10 text-center">
   831→        <p className="font-display text-xl text-carbon">No positions match your filters</p>
   832→        <p className="mt-2 text-sm text-stone">Try a different search term or clear the filters.</p>
   833→        <button onClick={clearFilters} className="dlg-ghost mt-5 min-h-[44px] px-6 text-sm">
   834→          Clear filters
   835→        </button>
   836→      </div>
   837→    );
   838→  } else {
   839→    content = (
   840→      <>
   841→        <div className="space-y-4">
   842→          {pageItems.map((job) => (
   843→            <JobCard
   844→              key={job.id}
   845→              job={job}
   846→              expanded={expandedId === job.id}
   847→              onToggle={toggleExpanded}
   848→              onApply={applyTo}
   849→              onDetail={openDetail}
   850→            />
   851→          ))}
   852→        </div>
   853→
   854→        {totalPages > 1 && (
   855→          <nav
   856→            className="mt-6 flex flex-wrap items-center justify-center gap-1 sm:gap-1.5"
   857→            aria-label="Pagination"
   858→          >
   859→            <button
   860→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   861→              disabled={current === 1}
   862→              aria-label="Previous page"
   863→              className="dlg-ghost grid h-11 w-11 place-items-center disabled:opacity-40"
   864→            >
   865→              <ChevronLeft className="h-4 w-4" aria-hidden="true" />
   866→            </button>
   867→            {pageWindow(current, totalPages).map((p, i) =>
   868→              p === "…" ? (
   869→                <span key={`ellipsis-${i}`} className="grid h-11 w-8 place-items-center text-sm text-pebble">
   870→                  …
   871→                </span>
   872→              ) : (
   873→                <button
   874→                  key={p}
   875→                  onClick={() => setPage(p)}
   876→                  aria-current={p === current ? "page" : undefined}
   877→                  className={
   878→                    p === current
   879→                      ? "grid h-11 min-w-11 place-items-center bg-ink px-3 text-sm font-medium text-white"
   880→                      : "grid h-11 min-w-11 place-items-center px-3 text-sm text-stone hover:bg-fog hover:text-ink"
   881→                  }
   882→                >
   883→                  {p}
   884→                </button>
   885→              )
   886→            )}
   887→            <button
   888→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   889→              disabled={current === totalPages}
   890→              aria-label="Next page"
   891→              className="dlg-ghost grid h-11 w-11 place-items-center disabled:opacity-40"
   892→            >
   893→              <ChevronRight className="h-4 w-4" aria-hidden="true" />
   894→            </button>
   895→          </nav>
   896→        )}
   897→      </>
   898→    );
   899→  }
   900→
   901→  return (
   902→    <div className={shellPad}>
   903→      <div className="mb-5 max-w-2xl">
   904→        <h1 className="text-heading-lg">Positions</h1>
   905→        <p className="mt-2 text-sm leading-relaxed text-stone">
   906→          Browse published vacancies at DOST-MIRDC and submit your application online.
   907→        </p>
   908→      </div>
   909→
   910→      <div className="items-start lg:flex lg:gap-6">
   911→        {/* Search rail — sticky on desktop, collapsible on mobile */}
   912→        <aside
   913→          className={`mb-4 w-full lg:sticky ${user ? "lg:top-6" : "lg:top-20"} lg:mb-0 lg:w-72 lg:shrink-0`}
   914→        >
   915→          <div className="lg:hidden">
   916→            <Collapsible open={filtersOpen} onOpenChange={setFiltersOpen}>
   917→              <CollapsibleTrigger asChild>
   918→                <button className="dlg-ghost flex min-h-[44px] w-full items-center justify-between px-5 text-sm">
   919→                  <span className="flex items-center gap-2">
   920→                    <SlidersHorizontal className="h-4 w-4" aria-hidden="true" />
   921→                    Filters
   922→                    {activeFilterCount > 0 && (
   923→                      <span className="grid h-5 min-w-5 place-items-center bg-ink px-1.5 text-[11px] font-medium text-white">
   924→                        {activeFilterCount}
   925→                      </span>
   926→                    )}
   927→                  </span>
   928→                  <ChevronDown
   929→                    className={`h-4 w-4 transition-transform ${filtersOpen ? "rotate-180" : ""}`}
   930→                    aria-hidden="true"
   931→                  />
   932→                </button>
   933→              </CollapsibleTrigger>
   934→              <CollapsibleContent>
   935→                <div className="pt-3">{filterRail("m")}</div>
   936→              </CollapsibleContent>
   937→            </Collapsible>
   938→          </div>
   939→          <div className="hidden lg:block">{filterRail("d")}</div>
   940→        </aside>
   941→
   942→        {/* Results */}
   943→        <div className="min-w-0 flex-1">
   944→          <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   945→            <p className="text-sm text-stone">
   946→              {filtered.length} Result{filtered.length === 1 ? "" : "s"}
   947→            </p>
   948→            <div className="flex items-center gap-2">
   949→              <span className="hidden text-sm text-stone sm:inline">Sort</span>
   950→              <Select
   951→                value={sort}
   952→                onValueChange={(v) => {
   953→                  setSort(v as SortKey);
   954→                  setPage(1);
   955→                }}
   956→              >
   957→                <SelectTrigger
   958→                  className="dlg-input min-h-[44px] w-[140px] sm:w-[170px]"
   959→                  aria-label="Sort results"
   960→                >
   961→                  <SelectValue />
   962→                </SelectTrigger>
   963→                <SelectContent className="rounded-none">
   964→                  <SelectItem value="newest">Newest</SelectItem>
   965→                  <SelectItem value="deadline">Deadline</SelectItem>
   966→                  <SelectItem value="salary">Salary desc</SelectItem>
   967→                </SelectContent>
   968→              </Select>
   969→            </div>
   970→          </div>
   971→
   972→          {/* Active-filter chips — presentation of existing filter state */}
   973→          {activeFilterCount > 0 && (
   974→            <div className="mb-4 flex flex-wrap items-center gap-2">
   975→              {query.trim() && (
   976→                <span className="dlg-pill inline-flex items-center gap-1.5 bg-white py-1.5 pl-3 pr-2 text-xs font-medium text-graphite">
   977→                  &ldquo;{query.trim()}&rdquo;
   978→                  <button
   979→                    onClick={() => changeQuery("")}
   980→                    aria-label="Clear search filter"
   981→                    className="focus-ring -mr-2 grid h-7 w-7 place-items-center text-stone hover:bg-fog hover:text-ink"
   982→                  >
   983→                    <X className="h-3 w-3" aria-hidden="true" />
   984→                  </button>
   985→                </span>
   986→              )}
   987→              {divisionFilter.map((code) => (
   988→                <span
   989→                  key={code}
   990→                  className="dlg-pill inline-flex items-center gap-1.5 bg-white py-1.5 pl-3 pr-2 text-xs font-medium text-graphite"
   991→                >
   992→                  {divisionShort(code)}
   993→                  <button
   994→                    onClick={() => toggleDivision(code)}
   995→                    aria-label={`Remove ${divisionShort(code)} filter`}
   996→                    className="focus-ring -mr-2 grid h-7 w-7 place-items-center text-stone hover:bg-fog hover:text-ink"
   997→                  >
   998→                    <X className="h-3 w-3" aria-hidden="true" />
   999→                  </button>
  1000→                </span>
  1001→              ))}
  1002→              <button
  1003→                onClick={clearFilters}
  1004→                className="text-xs text-stone underline underline-offset-4 hover:text-ink"
  1005→              >
  1006→                Clear all
  1007→              </button>
  1008→            </div>
  1009→          )}
  1010→
  1011→          {content}
  1012→        </div>
  1013→      </div>
  1014→
  1015→      {dialogs}
  1016→    </div>
  1017→  );
  1018→}
  1019→
