    71→const SAVED_KEY = "rmis-frontpage-saved-jobs";

    73→/** Client-persisted bookmark list for the frontpage explorer. */
    74→function useSavedJobs() {
    75→  const [saved, setSaved] = useState<number[]>(() => {
    76→    if (typeof window === "undefined") return [];
    77→    try {
    78→      const raw = window.localStorage.getItem(SAVED_KEY);
    79→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
    80→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
    81→    } catch {
    82→      return [];
    83→    }
    84→  });
    85→
    86→  const toggle = useCallback((id: number) => {
    87→    setSaved((prev) => {
    88→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
    89→      try {
    90→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
    91→      } catch {
    92→        /* storage unavailable — in-session state still works */
    93→      }
    94→      return next;
    95→    });
    96→  }, []);
    97→
    98→  return { saved, toggle };
    99→}

   280→/** Plain-text posting section rendered as a disc bullet list. */
   281→function TextBullets({ text }: { text: string }) {
   282→  const items = text
   283→    .split(/\r?\n/)
   284→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   285→    .filter(Boolean);
   286→  if (items.length === 0) return null;
   287→  return (
   288→    <ul className="list-disc space-y-2 pl-5 text-[15px] leading-relaxed text-body-gray">
   289→      {items.map((item, i) => (
   290→        <li key={i}>{item}</li>
   291→      ))}
   292→    </ul>
   293→  );
   294→}

   296→/** Posting section: sanitized rich HTML when present, else text bullets. */
   297→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {
   298→  if (html && html.trim()) {
   299→    return (
   300→      <section className="space-y-3">
   301→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   302→        <div className="rich-text text-[15px] leading-relaxed text-body-gray" dangerouslySetInnerHTML={{ __html: html }} />
   303→      </section>
   304→    );
   305→  }
   306→  if (text && text.trim()) {
   307→    return (
   308→      <section className="space-y-3">
   309→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   310→        <TextBullets text={text} />
   311→      </section>
   312→    );
   313→  }
   314→  return null;
   315→}

   335→

   446→export default function PublicLanding() {
   447→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   448→  const [error, setError] = useState<string | null>(null);
   449→  const [selectedId, setSelectedId] = useState<number | null>(null);
   450→  const [expandedId, setExpandedId] = useState<number | null>(null);
   451→  const [isDesktop, setIsDesktop] = useState(false);
   452→  const lastJson = useRef("");
   453→  const paneRef = useRef<HTMLDivElement>(null);
   454→  const { saved, toggle: toggleSave } = useSavedJobs();
   455→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   456→  // applicants get the confirm flow (same as the board), incomplete profiles
   457→  // are sent to finish their profile first (the fast-track was retired).
   458→  const { requestApply, dialogs } = useApplyFlow();
   459→
   460→  const load = useCallback(async (silent: boolean) => {
   461→    try {
   462→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   463→      const json = JSON.stringify(data);
   464→      if (!silent || json !== lastJson.current) {
   465→        lastJson.current = json;
   466→        setJobs(data);
   467→      }
   468→      setError(null);
   469→    } catch (e) {
   470→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   471→    }
   472→  }, []);
   473→
   474→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   475→  useEffect(() => {
   476→    void load(false);
   477→    const timer = setInterval(() => {
   478→      if (!document.hidden) void load(true);
   479→    }, 30_000);
   480→    const onFocus = () => void load(true);
   481→    window.addEventListener("focus", onFocus);
   482→    return () => {
   483→      clearInterval(timer);
   484→      window.removeEventListener("focus", onFocus);
   485→    };
   486→  }, [load]);
   487→
   488→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   489→  useEffect(() => {
   490→    const mq = window.matchMedia("(min-width: 1024px)");
   491→    const sync = () => setIsDesktop(mq.matches);
   492→    sync();
   493→    mq.addEventListener("change", sync);
   494→    return () => mq.removeEventListener("change", sync);
   495→  }, []);
   496→
   497→  const openJobs = useMemo(
   498→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   499→    [jobs]
   500→  );
   501→
   502→  const selectedJob = useMemo(
   503→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   504→    [openJobs, selectedId]
   505→  );
   506→
   507→  // Desktop opens with the first posting pre-selected (reference-board
   508→  // behavior); a stale selection after a poll snaps back to the top card.
   509→  useEffect(() => {
   510→    if (isDesktop && selectedId === null && openJobs.length > 0) {
   511→      setSelectedId(openJobs[0].id);
   512→    }
   513→  }, [isDesktop, openJobs, selectedId]);
   514→
   515→  useEffect(() => {
   516→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   517→      setSelectedId(openJobs[0]?.id ?? null);
   518→    }
   519→  }, [openJobs, selectedId]);
   520→
   521→  // Reference-board behavior — a freshly picked posting opens from the top,
   522→  // even when the previous posting was scrolled deep into its body.
   523→  useEffect(() => {
   524→    if (paneRef.current) paneRef.current.scrollTop = 0;
   525→  }, [selectedId]);
   526→
   527→  const select = useCallback(
   528→    (id: number) => {
   529→      setSelectedId(id);
   530→      if (!window.matchMedia("(min-width: 1024px)").matches) {
   531→        window.scrollTo({ top: 0, behavior: "smooth" });
   532→      }
   533→    },
   534→    []
   535→  );
   536→
   537→  const saveAndToast = useCallback(
   538→    (job: JobWire) => {
   539→      const wasSaved = saved.includes(job.id);
   540→      toggleSave(job.id);
   541→      toast.success(
   542→        wasSaved
   543→          ? "Removed from your saved jobs"
   544→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   545→      );
   546→    },
   547→    [saved, toggleSave]
   548→  );
   549→
   550→  const showList = isDesktop || selectedId === null;
   551→  const showDetail = isDesktop || selectedId !== null;
   552→
   553→  return (
   554→    <div className="flex min-h-screen flex-col lg:h-dvh">
   555→      <SiteHeader />
   556→
   557→      <main className="mx-auto w-full max-w-[1600px] flex-1 px-4 pb-10 pt-6 sm:px-6 lg:flex lg:min-h-0 lg:flex-col lg:px-8">
   558→        {/* Hero — Bevel "Hero Sky" field (rounded 32) anchoring the explorer,
   559→            arranged as a split: display headline + copy + single charcoal
   560→            capsule CTA left; official seal lockup + authority line right as
   561→            the quiet institutional proof strip (real marks only). */}
   562→        <section className="hero-sky rounded-[32px] p-6 sm:p-8 lg:px-10 lg:py-10">
   563→          <div className="grid items-center gap-8 lg:grid-cols-[minmax(0,1fr)_auto]">
   564→            {/* QA 2026-02: header renamed to "Job Opportunities" to match the
   565→                Positions tab; the live "N open" pill was removed — it read as
   566→                an unclickable button. */}
   567→            <div className="min-w-0">
   568→              <h1 className="text-4xl font-semibold leading-[1.02] tracking-[-0.03em] text-ink sm:text-6xl">
   569→                Job Opportunities
   570→              </h1>
   571→              <p className="mt-3 max-w-xl text-base leading-relaxed text-body-gray">
   572→                Current vacancies at DOST-MIRDC. Pick a role to read the full posting.
   573→              </p>
   574→              <div className="mt-6">
   575→                <button
   576→                  onClick={() => navigate("vacant-positions")}
   577→                  className="dlg-cta inline-flex min-h-[44px] items-center gap-2 px-6 text-sm font-medium"
   578→                >
   579→                  Browse the full board
   580→                  <ArrowRight className="h-4 w-4" aria-hidden="true" />
   581→                </button>
   582→              </div>
   583→            </div>
   584→            {/* Institutional proof — the same emblems and authority line the
   585→                footer carries, moved into the hero as the trust anchor. */}
   586→            <div className="flex items-center gap-4 lg:flex-col lg:items-end lg:gap-3 lg:text-right">
   587→              <BrandLogo logoSize={36} tone="light" />
   588→              <div className="min-w-0">
   589→                <p className="text-sm font-semibold leading-snug text-ink">
   590→                  Metals Industry Research and Development Center
   591→                </p>
   592→                <p className="mt-0.5 text-xs leading-relaxed text-body-gray">
   593→                  An agency of the Department of Science and Technology, Republic of the Philippines
   594→                </p>
   595→              </div>
   596→            </div>
   597→          </div>
   598→        </section>
   599→
   600→        {error !== null ? (
   601→          <div className="mt-6 rounded-2xl bg-[var(--bad-bg)] p-5">
   602→            <p className="text-sm text-ink">{error}</p>
   603→            <button
   604→              onClick={() => void load(false)}
   605→              className="mt-2 text-sm text-metric-blue underline underline-offset-2"
   606→            >
   607→              Try again
   608→            </button>
   609→          </div>
   610→        ) : jobs === null ? (
   611→          <div className="mt-6 space-y-3.5" aria-hidden="true">
   612→            {[0, 1, 2].map((i) => (
   613→              <div key={i} className="skel h-28" />
   614→            ))}
   615→          </div>
   616→        ) : openJobs.length === 0 ? (
   617→          <div className="dlg-card mt-6">
   618→            <EmptyState
   619→              icon={Inbox}
   620→              title="No open positions right now"
   621→              description="Check back soon or explore the full board for upcoming announcements."
   622→            />
   623→            <div className="pb-8 text-center">
   624→              <button
   625→                onClick={() => navigate("vacant-positions")}
   626→                className="inline-flex min-h-[44px] items-center gap-2 text-sm text-metric-blue underline underline-offset-2 hover:text-ink"
   627→              >
   628→                Go to the full positions board
   629→                <ArrowRight className="h-4 w-4" aria-hidden="true" />
   630→              </button>
   631→            </div>
   632→          </div>
   633→        ) : (
   634→          // Board height is derived, not a magic viewport constant: on lg the
   635→          // grid flex-fills whatever is left between the real header and
   636→          // footer (main is a min-h-0 flex column on lg), so the page itself
   637→          // never scrolls and the open posting stays pinned in view — the
   638→          // reference board's "sticky description" behavior.
   639→          <div className="mt-6 grid gap-8 sm:mt-8 lg:min-h-0 lg:flex-1 lg:grid-cols-[minmax(340px,430px)_minmax(0,1fr)]">
   640→            {/* Left · the register — 8px of ring room on every side (pl/py)
   641→                with matching negative margins so the cards, the column
   642→                alignment, and the scrollbar stay exactly where they were;
   643→                without it the overflow-y padding box clips the selected
   644→                card's outside ring on the left/top/bottom. */}
   645→            {showList && (
   646→              <div
   647→                aria-label="Job opportunities list"
   648→                className="scroll-thin min-h-0 space-y-3.5 lg:-ml-2 lg:-my-2 lg:overflow-y-auto lg:overscroll-contain lg:py-2 lg:pl-2 lg:pr-1.5"
   649→              >
   650→                {openJobs.map((job) => (
   651→                  <JobListCard
   652→                    key={job.id}
   653→                    job={job}
   654→                    selected={selectedId === job.id}
   655→                    expanded={expandedId === job.id}
   656→                    saved={saved.includes(job.id)}
   657→                    onSelect={() => select(job.id)}
   658→                    onToggle={() => setExpandedId(expandedId === job.id ? null : job.id)}
   659→                    onSave={() => saveAndToast(job)}
   660→                  />
   661→                ))}
   662→              </div>
   663→            )}
   664→
   665→            {/* Right · the posting */}
   666→            {showDetail && (
   667→              <div ref={paneRef} className="scroll-thin min-h-0 lg:overscroll-contain lg:overflow-y-auto lg:pr-2">
   668→                {selectedJob ? (
   669→                  <div className="dlg-card p-6 sm:p-8">
   670→                    <JobDetail
   671→                      job={selectedJob}
   672→                      saved={saved.includes(selectedJob.id)}
   673→                      onApply={() => requestApply(selectedJob)}
   674→                      onSave={() => saveAndToast(selectedJob)}
   675→                      onBack={() => setSelectedId(null)}
   676→                    />
   677→                  </div>
   678→                ) : (
   679→                  <div className="dlg-card">
   680→                    <EmptyState
   681→                      icon={Briefcase}
   682→                      title="Select a position"
   683→                      description="Choose a role from the list to read the full posting."
   684→                    />
   685→                  </div>
   686→                )}
   687→              </div>
   688→            )}
   689→          </div>
   690→        )}
   691→      </main>
   692→
   693→      <Footer />
   694→      {/* Shared apply dialog (confirm / MQR / done), mounted once. */}
   695→      {dialogs}
   696→    </div>
   697→  );
   698→}
