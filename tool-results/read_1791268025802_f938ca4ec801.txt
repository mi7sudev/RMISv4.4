   111→/* Progress-timeline palette, sampled from the approved status spec: jade for
   112→   achieved/live steps, quiet gray for steps ahead, bad-red for a failed one. */
   113→const TL_JADE = "#00ae75";
   114→const TL_JADE_TEXT = "#0a9565";
   115→const TL_GRAY_DISC = "#e8eaec";
   116→const TL_GRAY_RAIL = "#ededed";
   117→
   118→/** Timeline checkpoint marker (approved status spec): achieved steps are a
   119→ *  solid jade disc with a white glyph, the live step is a thick jade ring,
   120→ *  steps ahead are a quiet gray disc, and a failed decision is the app's
   121→ *  bad-red disc with a white ×. Glyph/shape carries the state so it never
   122→ *  rides on hue alone; the ring keeps a pulsing halo — motion-safe only. */
   123→function CheckpointDot({ state, doneGlyph }: { state: CheckpointState; doneGlyph?: ReactNode }) {
   124→  if (state === "current") {
   125→    return (
   126→      <span
   127→        aria-hidden="true"
   128→        className="motion-safe:animate-pulse inline-flex h-5 w-5 shrink-0 rounded-full border-[7px] bg-white"
   129→        style={{ borderColor: TL_JADE }}
   130→      />
   131→    );
   132→  }
   133→  if (state === "failed") {
   134→    return (
   135→      <span
   136→        aria-hidden="true"
   137→        className="inline-flex h-5 w-5 shrink-0 items-center justify-center rounded-full bg-[var(--bad)] text-white"
   138→      >
   139→        <X aria-hidden="true" className="h-[11px] w-[11px]" strokeWidth={3} />
   140→      </span>
   141→    );
   142→  }
   143→  if (state === "done") {
   144→    return (
   145→      <span
   146→        aria-hidden="true"
   147→        className="inline-flex h-5 w-5 shrink-0 items-center justify-center rounded-full text-white"
   148→        style={{ backgroundColor: TL_JADE }}
   149→      >
   150→        {doneGlyph ?? <Check aria-hidden="true" className="h-[11px] w-[11px]" strokeWidth={3} />}
   151→      </span>
   152→    );
   153→  }
   154→  return (
   155→    <span
   156→      aria-hidden="true"
   157→      className="inline-flex h-5 w-5 shrink-0 rounded-full"
   158→      style={{ backgroundColor: TL_GRAY_DISC }}
   159→    />
   160→  );
   161→}

   163→/** 3-checkpoint tracking timeline: Submitted → Under Review → Decision (§7.3),
   164→ *  styled to the approved status spec — jade rails, a bold step label with a
   165→ *  one-line caption under every checkpoint, the live step captioned in jade. */
   166→function JourneyTimeline({ status }: { status: string }) {
   167→  const stage = stageForStatus(status);
   168→  const decided = stage === "Shortlisted" || stage === "Rejected";
   169→
   170→  const steps: { label: string; state: CheckpointState; desc: string; doneGlyph?: ReactNode }[] = [
   171→    {
   172→      label: "Submitted",
   173→      state: "done",
   174→      desc: "Your application has been successfully submitted",
   175→    },
   176→    {
   177→      label: "Under Review",
   178→      state: stage === "Under Review" ? "current" : decided ? "done" : "pending",
   179→      desc:
   180→        stage === "Under Review"
   181→          ? "HR is currently reviewing your application"
   182→          : decided
   183→            ? "Your application has passed the initial review"
   184→            : "HR will evaluate your application next",
   185→    },
   186→    {
   187→      // Neutral pending label — never presumes the shortlist outcome, since
   188→      // the decision can go either way.
   189→      label: stage === "Shortlisted" ? "Shortlisted" : stage === "Rejected" ? "Not Shortlisted" : "Awaiting Decision",
   190→      state: stage === "Shortlisted" ? "done" : stage === "Rejected" ? "failed" : "pending",
   191→      desc:
   192→        stage === "Shortlisted"
   193→          ? "You've been shortlisted for the next step"
   194→          : stage === "Rejected"
   195→            ? "Your application was not shortlisted"
   196→            : "You will be notified once a decision is made",
   197→      // Proper glyph for the achieved decision — a star, not a generic check.
   198→      doneGlyph: <Star aria-hidden="true" className="h-[11px] w-[11px] fill-current" strokeWidth={0} />,
   199→    },
   200→  ];
   201→
   202→  return (
   203→    <div className="mt-3 grid grid-cols-3 gap-1">
   204→      {steps.map((s, i) => {
   205→        const reached = s.state !== "pending";
   206→        return (
   207→          <div key={`${s.label}-${i}`} className="flex flex-col items-center text-center">
   208→            <div className="flex w-full items-center">
   209→              {/* Traversed connectors render in the timeline's jade green.
   210→                  Negative outer margins bridge the grid's gap-1 so each rail
   211→                  reads as one continuous line, not segments with gaps. */}
   212→              <div
   213→                className={cn("h-[2px] flex-1", i > 0 && "-ml-1")}
   214→                style={{ backgroundColor: i === 0 ? "transparent" : reached ? TL_JADE : TL_GRAY_RAIL }}
   215→              />
   216→              <CheckpointDot state={s.state} doneGlyph={s.doneGlyph} />
   217→              <div
   218→                className={cn("h-[2px] flex-1", i < steps.length - 1 && "-mr-1")}
   219→                style={{
   220→                  backgroundColor:
   221→                    i === steps.length - 1 ? "transparent" : steps[i + 1].state !== "pending" ? TL_JADE : TL_GRAY_RAIL,
   222→                }}
   223→              />
   224→            </div>
   225→            <div className="mt-2 flex flex-col gap-0.5">
   226→              <span
   227→                className="text-[10px] font-semibold leading-tight text-ink"
   228→                style={
   229→                  s.state === "current"
   230→                    ? { color: TL_JADE_TEXT }
   231→                    : s.state === "failed"
   232→                      ? { color: "var(--bad)" }
   233→                      : undefined
   234→                }
   235→              >
   236→                {s.label}
   237→              </span>
   238→              <span
   239→                className="text-[9px] leading-snug text-stone"
   240→                style={s.state === "current" ? { color: TL_JADE_TEXT } : undefined}
   241→              >
   242→                {s.desc}
   243→              </span>
   244→            </div>
   245→          </div>
   246→        );
   247→      })}
   248→    </div>
   249→  );
   250→}

   257→// Shared with the frontpage explorer — bookmarks saved here show as saved
   258→// there and vice versa.
   259→const SAVED_KEY = "rmis-frontpage-saved-jobs";
   260→
   261→/** Client-persisted bookmark list (same store as the frontpage). */
   262→function useSavedJobs() {
   263→  const [saved, setSaved] = useState<number[]>(() => {
   264→    if (typeof window === "undefined") return [];
   265→    try {
   266→      const raw = window.localStorage.getItem(SAVED_KEY);
   267→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
   268→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
   269→    } catch {
   270→      return [];
   271→    }
   272→  });
   273→
   274→  const toggle = useCallback((id: number) => {
   275→    setSaved((prev) => {
   276→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
   277→      try {
   278→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
   279→      } catch {
   280→        /* storage unavailable — in-session state still works */
   281→      }
   282→      return next;
   283→    });
   284→  }, []);
   285→
   286→  return { saved, toggle };
   287→}

   389→function ApplicationDetailModal({
   390→  application,
   391→  onClose,
   392→  onCancelled,
   393→}: {
   394→  application: ApplicationWire;
   395→  onClose: () => void;
   396→  onCancelled: () => void;
   397→}) {
   398→  const [confirmOpen, setConfirmOpen] = useState(false);
   399→  const [cancelling, setCancelling] = useState(false);
   400→  const job = application.job ?? null;
   401→  const position = job?.position ?? null;
   402→  const label = currentStageLabel(application.status);
   403→  const cancellable = label === "Submitted";
   404→  const dl = deadlineState(job?.deadlineDate ?? null);
   405→
   406→  const cancelApplication = async () => {
   407→    setCancelling(true);
   408→    try {
   409→      await apiFetch(`/api/applications/${application.id}`, { method: "DELETE" });
   410→      toast.success("Application cancelled", { description: "Your application has been withdrawn." });
   411→      setConfirmOpen(false);
   412→      onClose();
   413→      onCancelled();
   414→    } catch (e) {
   415→      // Progressed applications: surface the server's guidance (e.g. "contact HR") as-is.
   416→      toast.error(e instanceof Error ? e.message : "Unable to cancel this application");
   417→      setConfirmOpen(false);
   418→    } finally {
   419→      setCancelling(false);
   420→    }
   421→  };
   422→
   423→  const facts: [string, string][] = [
   424→    ["Place of Assignment", position?.placeOfAssignment ?? "-"],
   425→    ["Employment Type", job?.positionType ? humanize(job.positionType) : position?.positionType ? humanize(position.positionType) : "-"],
   426→    ["Vacancies", job?.numberOfVacancy == null ? "-" : String(job.numberOfVacancy)],
   427→    ["Monthly Salary", formatCurrency(position?.salaryAmount ?? null)],
   428→    ["Salary Grade", position?.salaryGrade ? `SG-${position.salaryGrade}${position.salaryStep ? ` Step ${position.salaryStep}` : ""}` : "-"],
   429→    ["Item No.", position?.itemNumber ?? "-"],
   430→    ["Published", formatDate(job?.publishDate ?? null)],
   431→    ["Deadline", job ? dl.label : "-"],
   432→  ];
   433→
   434→  return (
   435→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   436→      <DialogContent className="dlg-card-plain max-h-[90dvh] w-full overflow-y-auto scroll-thin p-0 sm:max-w-3xl">
   437→        <div className="p-6">
   438→          <DialogHeader className="space-y-2 text-left">
   439→            <div className="flex flex-wrap items-center gap-2">
   440→              <StatusPill status={label} variant={stageVariant(label)} />
   441→              <span className="dlg-pill num rounded-full bg-fog px-3 py-1 text-xs font-medium text-stone">
   442→                Application #{application.id}
   443→              </span>
   444→            </div>
   445→            <DialogTitle className="font-display text-2xl leading-tight tracking-[-0.02em] text-ink">
   446→              {job?.title ?? "Application"}
   447→            </DialogTitle>
   448→          </DialogHeader>
   449→
   450→          {/* Successfully-applied strip (§7.7) */}
   451→          <div className="mt-4 flex items-center gap-3 rounded-lg bg-cloud/60 p-3">
   452→            <IconChip icon={BadgeCheck} tone="emerald" size={36} iconSize={17} />
   453→            <div>
   454→              <p className="text-sm font-medium text-ink">Successfully Applied</p>
   455→              <p className="text-xs text-stone">
   456→                Applied <span className="num">{formatDate(application.dateApplied)}</span>
   457→              </p>
   458→            </div>
   459→          </div>
   460→
   461→          {/* Vitals summary grid */}
   462→          <div className="mt-5 grid grid-cols-2 gap-2 sm:grid-cols-4">
   463→            {facts.map(([k, v]) => (
   464→              <div key={k} className="rounded-lg bg-cloud/60 p-3">
   465→                <p className="text-[11px] text-stone">{k}</p>
   466→                <p className="num mt-0.5 truncate text-sm text-ink" title={v}>
   467→                  {v}
   468→                </p>
   469→              </div>
   470→            ))}
   471→          </div>
   472→
   473→          <div className="mt-6 space-y-6">
   474→            <MqrLedger position={position} />
   475→            <RichSection title="Brief Description" text={job?.briefDescription} html={job?.briefDescriptionHtml} />
   476→            <RichSection title="Duties & Responsibilities" text={job?.dutiesResponsibilities} html={job?.dutiesHtml} />
   477→            <RichSection title="Compensation Package" text={job?.compensationPackage} html={job?.compensationHtml} />
   478→            <RichSection title="Other Qualifications" text={job?.otherQualifications} html={job?.otherQualificationsHtml} />
   479→          </div>
   480→
   481→          <div className="mt-8 flex flex-col gap-3 border-t border-border pt-5 sm:flex-row sm:justify-end">
   482→            {cancellable && (
   483→              <button
   484→                type="button"
   485→                onClick={() => setConfirmOpen(true)}
   486→                disabled={cancelling}
   487→                className="inline-flex min-h-[44px] items-center justify-center rounded-full border border-[var(--bad)]/30 bg-white px-6 py-2.5 text-sm font-medium text-[var(--bad)] transition-colors hover:bg-[var(--bad-bg)] disabled:opacity-50"
   488→              >
   489→                Cancel Application
   490→              </button>
   491→            )}
   492→            <button type="button" onClick={onClose} className="dlg-ghost min-h-[44px] px-6 py-2.5 text-sm">
   493→              Close
   494→            </button>
   495→          </div>
   496→        </div>
   497→
   498→        {/* Nested destructive confirm (§7.7) */}
   499→        <AlertDialog open={confirmOpen} onOpenChange={setConfirmOpen}>
   500→          <AlertDialogContent className="dlg-card-plain">
   501→            <AlertDialogHeader>
   502→              <AlertDialogTitle className="text-2xl font-semibold leading-[1.1] tracking-[-0.01em] text-ink">Cancel this application?</AlertDialogTitle>
   503→              <AlertDialogDescription className="text-sm leading-relaxed text-body-gray">
   504→                Your application for {job?.title ?? "this position"} will be withdrawn. You may apply again while the
