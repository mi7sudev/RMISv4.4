     1→// ============================================================================
     2→// email.ts — provider-agnostic EMAIL sender for RMIS notifications.
     3→//
     4→// Mirrors the audit-trail design of the notification surfaces:
     5→//
     6→//   1. "mock" (default, dev) — FREE forever. No external service: every send
     7→//      is logged to the `email_logs` table so the whole flow (shortlist
     8→//      notice included) is testable end-to-end and auditable in the admin
     9→//      Email panel.
    10→//   2. "resend" — production-grade HTTP API. Free tier (100 emails/day),
    11→//      no SDK needed — plain fetch. Set RESEND_API_KEY + (optional)
    12→//      EMAIL_FROM. Perfect fit: shortlist notices are low-volume, so the
    13→//      free tier realistically costs ₱0 for this office.
    14→//   3. "middleware" — the office's OWN email gateway (mirdc_middleware, an
    15→//      Express+nodemailer service on the MIRDC VM). Sends through the
    16→//      office mailbox systems@mirdc.dost.gov.ph over office SMTP. Set
    17→//      EMAIL_MIDDLEWARE_URL (+ EMAIL_MIDDLEWARE_API_KEY). ₱0 cost, no
    18→//      external vendor, works on the intranet.
    19→//
    20→// CONFIG (env vars):
    21→//   EMAIL_PROVIDER=mock|resend|middleware
    22→//   # resend:
    23→//   RESEND_API_KEY=<key from resend.com>
    24→//   EMAIL_FROM="DOST-MIRDC Recruitment <recruitment@mirdc.gov.ph>"
    25→//   # middleware:
    26→//   EMAIL_MIDDLEWARE_URL=http://<vm-host>:3232
    27→//   EMAIL_MIDDLEWARE_API_KEY=<shared middleware token>
    28→//   EMAIL_MIDDLEWARE_SYSTEM="DOST-MIRDC Recruitment"  (From display name)
    29→//
    30→// Design rules:
    31→//   - sendEmail() NEVER throws — callers fire it after their own success
    32→//     path and a provider outage must never break a status update.
    33→//     Results are always persisted to `email_logs`.
    34→//   - Templates: the SHORTLIST notice is the flagship (rich HTML + text);
    35→//     every other status change gets a simple, honest plain-text notice.
    36→// ============================================================================
    37→

   104→export type SendEmailInput = {
   105→  to: string | null | undefined;
   106→  subject: string;
   107→  /** plain-text body — always required (email clients + deliverability) */
   108→  text: string;
   109→  /** optional HTML body (inline styles only — no external assets) */
   110→  html?: string;
   111→  /** optional linkage for the audit trail, e.g. { type: "application", id: 12 } */
   112→  related?: { type: string; id: number };
   113→  /**
   114→   * Optional file attachments (base64 content). Only the ACTIVE provider
   115→   * receives the bytes — email_logs persists metadata (name + size) only,
   116→   * never the file content (PII / DB-size discipline).
   117→   */
   118→  attachments?: EmailAttachment[];
   119→};
   120→
   121→/** A single email attachment. `content` is base64-encoded file bytes. */
   122→export type EmailAttachment = {
   123→  filename: string;
   124→  content: string;
   125→  contentType?: string;
   126→};
   127→
   128→export type SendEmailResult = {
   129→  attempted: boolean;
   130→  status: "sent" | "failed" | "mock" | "skipped";
   131→  provider: EmailProviderId;
   132→  providerRef?: string;
   133→  error?: string;
   134→  to?: string;
   135→};
   136→
   137→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;
   138→
   139→/** Fire an email through the active provider. Never throws. */
   140→export async function sendEmail(input: SendEmailInput): Promise<SendEmailResult> {
   141→  const provider = getEmailProviderId();
   142→  const to = (input.to ?? "").trim();
   143→
   144→  // Unreachable/absent address — record the skip so admins can see why an
   145→  // applicant got no email.
   146→  if (!to || !EMAIL_RE.test(to)) {
   147→    await logEmail({
   148→      to: to || "(none)",
   149→      subject: input.subject,
   150→      bodyText: input.text,
   151→      bodyHtml: input.html ?? null,
   152→      provider,
   153→      status: "skipped",
   154→      error: "No valid email address on record",
   155→      relatedType: input.related?.type,
   156→      relatedId: input.related?.id,
   157→      attachmentsMeta: attachmentsMeta(input.attachments),
   158→    }).catch(() => {});
   159→    return { attempted: false, status: "skipped", provider };
   160→  }
   161→
   162→  let result: { ok: boolean; ref?: string; error?: string; status: "sent" | "failed" | "mock" };
   163→  try {
   164→    switch (provider) {
   165→      case "resend":
   166→        result = await sendViaResend(
   167→          to,
   168→          input.subject,
   169→          input.text,
   170→          input.html,
   171→          input.attachments
   172→        );
   173→        break;
   174→      case "middleware":
   175→        result = await sendViaMiddleware(
   176→          to,
   177→          input.subject,
   178→          input.text,
   179→          input.html,
   180→          input.attachments
   181→        );
   182→        break;
   183→      default:
   184→        // mock: nothing is delivered — attachment metadata is still logged
   185→        // below so the audit trail shows what WOULD have been sent.
   186→        result = { ok: true, status: "mock", ref: "mock" };
   187→    }
   188→  } catch (e) {
   189→    result = {
   190→      ok: false,
   191→      status: "failed",
   192→      error: e instanceof Error ? e.message : String(e),
   193→    };
   194→  }
   195→
   196→  await logEmail({
   197→    to,
   198→    subject: input.subject,
   199→    bodyText: input.text,
   200→    bodyHtml: input.html ?? null,
   201→    provider,
   202→    status: result.status,
   203→    providerRef: result.ref,
   204→    error: result.error,
   205→    relatedType: input.related?.type,
   206→    relatedId: input.related?.id,
   207→    attachmentsMeta: attachmentsMeta(input.attachments),
   208→  }).catch(() => {});
   209→
   210→  return {
   211→    attempted: true,
   212→    status: result.status,
   213→    provider,
   214→    providerRef: result.ref,
   215→    error: result.error,
   216→    to,
   217→  };
   218→}
   219→
   220→/** JSON metadata persisted to email_logs.attachments — name + size only. */
   221→function attachmentsMeta(attachments?: EmailAttachment[]): string | null {
   222→  if (!attachments || attachments.length === 0) return null;
   223→  try {
   224→    return JSON.stringify(
   225→      attachments.map((a) => ({
   226→        name: a.filename,
   227→        bytes: Math.floor((a.content.length * 3) / 4),
   228→      }))
   229→    );
   230→  } catch {
   231→    return null;
   232→  }
   233→}

   235→async function logEmail(row: {
   236→  to: string;
   237→  subject: string;
   238→  bodyText: string;
   239→  bodyHtml: string | null;
   240→  provider: EmailProviderId;
   241→  status: string;
   242→  providerRef?: string;
   243→  error?: string;
   244→  relatedType?: string;
   245→  relatedId?: number;
   246→  attachmentsMeta?: string | null;
   247→}) {
   248→  await db.emailLog.create({
   249→    data: {
   250→      to: row.to,
   251→      subject: row.subject,
   252→      bodyText: row.bodyText,
   253→      bodyHtml: row.bodyHtml,
   254→      provider: row.provider,
   255→      status: row.status,
   256→      providerRef: row.providerRef,
   257→      error: row.error,
   258→      relatedType: row.relatedType,
   259→      relatedId: row.relatedId,
   260→      attachments: row.attachmentsMeta ?? null,
   261→    },
   262→  });
   263→}

   265→// --- Driver: Resend (HTTP API — no SDK) ------------------------------------
   266→
   267→async function sendViaResend(
   268→  to: string,
   269→  subject: string,
   270→  text: string,
   271→  html?: string,
   272→  attachments?: EmailAttachment[]
   273→): Promise<{ ok: boolean; ref?: string; error?: string; status: "sent" | "failed" }> {
   274→  const key = process.env.RESEND_API_KEY;
   275→  if (!key) return { ok: false, status: "failed", error: "RESEND_API_KEY not set" };
   276→
   277→  const res = await fetch("https://api.resend.com/emails", {
   278→    method: "POST",
   279→    headers: {
   280→      "Content-Type": "application/json",
   281→      Authorization: `Bearer ${key}`,
   282→    },
   283→    body: JSON.stringify({
   284→      from: DEFAULT_FROM,
   285→      to: [to],
   286→      subject,
   287→      text,
   288→      ...(html ? { html } : {}),
   289→      ...(attachments && attachments.length > 0
   290→        ? {
   291→            attachments: attachments.map((a) => ({
   292→              filename: a.filename,
   293→              content: a.content,
   294→              ...(a.contentType ? { content_type: a.contentType } : {}),
   295→            })),
   296→          }
   297→        : {}),
   298→    }),
   299→  });
   300→  if (!res.ok) {
   301→    return {
   302→      ok: false,
   303→      status: "failed",
   304→      error: `resend ${res.status}: ${await safeBody(res)}`,
   305→    };
   306→  }
   307→  const data = (await res.json().catch(() => ({}))) as { id?: string };
   308→  return { ok: true, status: "sent", ref: data?.id };
   309→}
   310→
   311→async function safeBody(res: Response): Promise<string> {
   312→  try {
   313→    return (await res.text()).slice(0, 200);
   314→  } catch {
   315→    return "";
   316→  }
   317→}

   319→// --- Driver: MIRDC Mail Middleware (office VM, mirdc_middleware) ------------
   320→// POST {EMAIL_MIDDLEWARE_URL}/sendMail — Express+nodemailer service run by
   321→// the office. The From header is ALWAYS the office mailbox
   322→// systems@mirdc.dost.gov.ph (display name = EMAIL_MIDDLEWARE_SYSTEM, default
   323→// "DOST-MIRDC Recruitment"); the middleware's own default Reply-To applies.
   324→// The updated middleware also accepts a plain-text part and base64
   325→// attachments; an OLDER middleware simply ignores those fields — emails
   326→// still deliver (HTML-only), so this driver is safe against both versions.
   327→
   328→async function sendViaMiddleware(
   329→  to: string,
   330→  subject: string,
   331→  text: string,
   332→  html?: string,
   333→  attachments?: EmailAttachment[]
   334→): Promise<{ ok: boolean; ref?: string; error?: string; status: "sent" | "failed" }> {
   335→  const base = env.EMAIL_MIDDLEWARE_URL;
   336→  if (!base) return { ok: false, status: "failed", error: "EMAIL_MIDDLEWARE_URL not set" };
   337→
   338→  // The middleware delivers an HTML body only — always give it one. When a
   339→  // template didn't build HTML, wrap the plain text minimally so the
   340→  // message is never an empty body.
   341→  const message = html ?? textToSimpleHtml(text);
   342→
   343→  const res = await fetch(`${base}/sendMail`, {
   344→    method: "POST",
   345→    headers: { "Content-Type": "application/json" },
   346→    // Never hang an API route on a stalled VM — the middleware normally
   347→    // answers in well under a second.
   348→    signal: AbortSignal.timeout(15_000),
   349→    body: JSON.stringify({
   350→      // Shared middleware token (validated only when the VM opts into
   351→      // MIDDLEWARE_API_KEY strict mode; harmless otherwise).
   352→      api: env.EMAIL_MIDDLEWARE_API_KEY,
   353→      // From display name — becomes "…" <systems@mirdc.dost.gov.ph>.
   354→      system: env.EMAIL_MIDDLEWARE_SYSTEM || "DOST-MIRDC Recruitment",
   355→      to,
   356→      subject,
   357→      message,
   358→      // Additive fields — newer middleware adds the text part / attachments;
   359→      // older middleware ignores them.
   360→      text,
   361→      ...(attachments && attachments.length > 0
   362→        ? {
   363→            attachments: attachments.map((a) => ({
   364→              filename: a.filename,
   365→              content: a.content,
   366→              encoding: "base64",
   367→              ...(a.contentType ? { contentType: a.contentType } : {}),
   368→            })),
   369→          }
   370→        : {}),
   371→    }),
   372→  });
   373→
   374→  if (!res.ok) {
   375→    return {
   376→      ok: false,
   377→      status: "failed",
   378→      error: `middleware ${res.status}: ${await safeBody(res)}`,
   379→    };
   380→  }
   381→  const data = (await res.json().catch(() => ({}))) as { messageId?: string | null };
   382→  return { ok: true, status: "sent", ref: data?.messageId || "middleware" };
   383→}
   384→
   385→/** Minimal escaped-HTML wrapper for plain-text bodies (middleware has no text-only mode). */
   386→function textToSimpleHtml(text: string): string {
   387→  const paras = text
   388→    .split(/\n{2,}/)
   389→    .map(
   390→      (p) =>
   391→        `<p style="margin:0 0 12px;">${esc(p).replace(/\n/g, "<br/>")}</p>`
   392→    )
   393→    .join("");
   394→  return `<div style="font-family:Arial,Helvetica,sans-serif;font-size:14px;line-height:1.6;color:#1a1a1a;">${paras}</div>`;
   395→}

   429→function esc(s: string): string {
   430→  return s
   431→    .replace(/&/g, "&amp;")
   432→    .replace(/</g, "&lt;")
   433→    .replace(/>/g, "&gt;")
   434→    .replace(/"/g, "&quot;");
