   424→export default function Recruitment() {
   425→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   426→  const [error, setError] = useState<string | null>(null);
   427→  const [search, setSearch] = useState("");
   428→  const [debouncedSearch, setDebouncedSearch] = useState("");
   429→  const [statusFilter, setStatusFilter] = useState<"all" | "open" | "closed">("all");
   430→  const [sortBy, setSortBy] = useState<"recent" | "deadline" | "applications">("recent");
   431→  const [page, setPage] = useState(1);
   432→  const [formOpen, setFormOpen] = useState(false);
   433→  const [editJob, setEditJob] = useState<JobWire | null>(null);
   434→  const [spin, setSpin] = useState(false);
   435→
   436→  const openCreate = () => {
   437→    setEditJob(null);
   438→    setFormOpen(true);
   439→  };
   440→  const openEdit = (j: JobWire) => {
   441→    setEditJob(j);
   442→    setFormOpen(true);
   443→  };
   444→
   445→  const load = useCallback(async (silent = false) => {
   446→    if (!silent) setError(null);
   447→    try {
   448→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   449→      setJobs(data);
   450→    } catch (e) {
   451→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load job postings");
   452→    }
   453→  }, []);
   454→
   455→  useEffect(() => {
   456→    void load();
   457→  }, [load]);
   458→
   459→  // 30 s silent poll + refetch-on-focus.
   460→  useEffect(() => {
   461→    const id = setInterval(() => {
   462→      if (!document.hidden) void load(true);
   463→    }, 30_000);
   464→    return () => clearInterval(id);
   465→  }, [load]);
   466→  useEffect(() => {
   467→    const onFocus = () => void load(true);
   468→    window.addEventListener("focus", onFocus);
   469→    document.addEventListener("visibilitychange", onFocus);
   470→    return () => {
   471→      window.removeEventListener("focus", onFocus);
   472→      document.removeEventListener("visibilitychange", onFocus);
   473→    };
   474→  }, [load]);
   475→
   476→  // Debounced search (350 ms).
   477→  useEffect(() => {
   478→    const t = setTimeout(() => {
   479→      setDebouncedSearch(search.trim().toLowerCase());
   480→      setPage(1);
   481→    }, 350);
   482→    return () => clearTimeout(t);
   483→  }, [search]);
   484→
   485→  const filtered = useMemo(() => {
   486→    let rows = jobs ?? [];
   487→    if (debouncedSearch) {
   488→      rows = rows.filter((j) => {
   489→        const hay = [
   490→          j.title,
   491→          j.position?.itemNumber,
   492→          j.position?.placeOfAssignment,
   493→          j.positionType,
   494→        ]
   495→          .filter(Boolean)
   496→          .join(" ")
   497→          .toLowerCase();
   498→        return hay.includes(debouncedSearch);
   499→      });
   500→    }
   501→    if (statusFilter === "open") rows = rows.filter(isOpenJob);
   502→    else if (statusFilter === "closed") rows = rows.filter((j) => !isOpenJob(j));
   503→    const sorted = [...rows];
   504→    if (sortBy === "recent") {
   505→      sorted.sort((a, b) => new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime());
   506→    } else if (sortBy === "deadline") {
   507→      sorted.sort((a, b) => {
   508→        const av = a.deadlineDate ? new Date(a.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   509→        const bv = b.deadlineDate ? new Date(b.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   510→        return av - bv;
   511→      });
   512→    } else {
   513→      sorted.sort((a, b) => b.applicationCount - a.applicationCount);
   514→    }
   515→    return sorted;
   516→  }, [jobs, debouncedSearch, statusFilter, sortBy]);
   517→
   518→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   519→  const safePage = Math.min(page, totalPages);
   520→  const pageRows = filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE);
   521→
   522→  const manualRefresh = useCallback(() => {
   523→    setSpin(true);
   524→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));
   525→  }, [load]);
   526→
   527→  return (
   528→    <div className="space-y-6">
   529→      <PageHeader
   530→        title="Recruitment"
   531→        description={`${filtered.length} posting${filtered.length === 1 ? "" : "s"}`}
   532→        actions={
   533→          <>
   534→            <button type="button" className={ghostBtn} onClick={manualRefresh}>
