    89→/** §7.3 Next Step hint paragraph per pipeline stage. */
    90→function nextStepHint(stage: string): string {
    91→  if (stage === "Rejected")
    92→    return "Your application was not shortlisted. You may apply for other open positions.";
    93→  if (stage === "Shortlisted")
    94→    return "A notice has been sent to your registered email. HR will contact you for the next steps.";

   132→/** 3-checkpoint tracking timeline: Submitted → Review → Decision (§7.3). */
   133→function JourneyTimeline({ status }: { status: string }) {
   134→  const stage = stageForStatus(status);
   135→  const decided = stage === "Shortlisted" || stage === "Rejected";
   136→
   137→  const steps: { label: string; short?: string; state: CheckpointState }[] = [
   138→    { label: "Submitted", state: "done" },
   139→    {
   140→      label: "Review",
   141→      state: stage === "Under Review" ? "current" : decided ? "done" : "pending",
   142→    },
   143→    {
   144→      label:
   145→        stage === "Shortlisted" ? "Shortlisted" : stage === "Rejected" ? "Not Shortlisted" : "Awaiting the shortlist decision",
   146→      // Responsive copy swap (3-d, presentation-only): the full label wraps to 4+ lines
   147→      // inside a 3-up timeline at 320px; "Shortlisted"/"Not Shortlisted" stay verbatim.
   148→      short: stage === "Shortlisted" || stage === "Rejected" ? undefined : "Awaiting decision",
   149→      state: stage === "Shortlisted" ? "done" : stage === "Rejected" ? "failed" : "pending",
   150→    },
   151→  ];
   152→
   153→  return (
   154→    <div className="mt-3 grid grid-cols-3 gap-1">
   155→      {steps.map((s, i) => {
   156→        const reached = s.state !== "pending";
   157→        return (
   158→          <div key={`${s.label}-${i}`} className="flex flex-col items-center gap-1.5 text-center">
   159→            <div className="flex w-full items-center">
   160→              {/* Traversed connectors pick up a subtle ok→warn gradient tint. */}
   161→              <div
   162→                className={cn(
   163→                  "h-px flex-1",
   164→                  i === 0 ? "bg-transparent" : reached ? "bg-gradient-to-r from-[var(--ok)]/60 to-[var(--warn)]/45" : "bg-divider"
   165→                )}
   166→              />
   167→              <CheckpointDot state={s.state} />
   168→              <div
   169→                className={cn(
   170→                  "h-px flex-1",
   171→                  i === steps.length - 1
   172→                    ? "bg-transparent"
   173→                    : steps[i + 1].state !== "pending"
   174→                      ? "bg-gradient-to-r from-[var(--ok)]/60 to-[var(--warn)]/45"
   175→                      : "bg-divider"
   176→                )}
   177→              />
   178→            </div>
   179→            <span className={cn("text-[10px] leading-tight", s.state === "current" ? "font-medium text-ink" : "text-stone")}>
   180→              {s.short ? (
   181→                <>
   182→                  <span className="sm:hidden">{s.short}</span>
   183→                  <span className="hidden sm:inline">{s.label}</span>
   184→                </>
   185→              ) : (
   186→                s.label
   187→              )}
   188→            </span>
   189→          </div>
   190→        );
   191→      })}
   192→    </div>
   193→  );
   194→}

   296→function ApplicationDetailModal({
   297→  application,
   298→  onClose,
   299→  onCancelled,
   300→}: {
   301→  application: ApplicationWire;
   302→  onClose: () => void;
   303→  onCancelled: () => void;
   304→}) {
   305→  const [confirmOpen, setConfirmOpen] = useState(false);
   306→  const [cancelling, setCancelling] = useState(false);
   307→  const job = application.job ?? null;
   308→  const position = job?.position ?? null;
   309→  const label = currentStageLabel(application.status);
   310→  const cancellable = label === "Submitted";
   311→  const dl = deadlineState(job?.deadlineDate ?? null);
   312→
   313→  const cancelApplication = async () => {
   314→    setCancelling(true);
   315→    try {
   316→      await apiFetch(`/api/applications/${application.id}`, { method: "DELETE" });
   317→      toast.success("Application cancelled", { description: "Your application has been withdrawn." });
   318→      setConfirmOpen(false);
   319→      onClose();
   320→      onCancelled();
   321→    } catch (e) {
   322→      // Progressed applications: surface the server's guidance (e.g. "contact HR") as-is.
   323→      toast.error(e instanceof Error ? e.message : "Unable to cancel this application");
   324→      setConfirmOpen(false);
   325→    } finally {
   326→      setCancelling(false);
   327→    }
   328→  };
   329→
   330→  const facts: [string, string][] = [
   331→    ["Place of Assignment", position?.placeOfAssignment ?? "-"],
   332→    ["Employment Type", job?.positionType ? humanize(job.positionType) : position?.positionType ? humanize(position.positionType) : "-"],
   333→    ["Vacancies", job?.numberOfVacancy == null ? "-" : String(job.numberOfVacancy)],
   334→    ["Monthly Salary", formatCurrency(position?.salaryAmount ?? null)],
   335→    ["Salary Grade", position?.salaryGrade ? `SG-${position.salaryGrade}${position.salaryStep ? ` Step ${position.salaryStep}` : ""}` : "-"],
   336→    ["Item No.", position?.itemNumber ?? "-"],
   337→    ["Published", formatDate(job?.publishDate ?? null)],
   338→    ["Deadline", job ? dl.label : "-"],
   339→  ];
   340→
   341→  return (
   342→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   343→      <DialogContent className="dlg-card-plain max-h-[90dvh] w-full overflow-y-auto scroll-thin p-0 sm:max-w-3xl">
   344→        <div className="p-6">
   345→          <DialogHeader className="space-y-2 text-left">
   346→            <div className="flex flex-wrap items-center gap-2">
   347→              <StatusPill status={label} variant={stageVariant(label)} />
   348→              <span className="dlg-pill num rounded-full bg-fog px-3 py-1 text-xs font-medium text-stone">
   349→                Application #{application.id}
   350→              </span>
   351→            </div>
   352→            <DialogTitle className="font-display text-2xl leading-tight tracking-[-0.02em] text-ink">
   353→              {job?.title ?? "Application"}
   354→            </DialogTitle>
   355→          </DialogHeader>
   356→
   357→          {/* Successfully-applied strip (§7.7) */}
   358→          <div className="mt-4 flex items-center gap-3 rounded-lg bg-cloud/60 p-3">
   359→            <IconChip icon={BadgeCheck} tone="emerald" size={36} iconSize={17} />
   360→            <div>
   361→              <p className="text-sm font-medium text-ink">Successfully Applied</p>
   362→              <p className="text-xs text-stone">
   363→                Applied <span className="num">{formatDate(application.dateApplied)}</span>
   364→              </p>
   365→            </div>
   366→          </div>
   367→
   368→          {/* Vitals summary grid */}
   369→          <div className="mt-5 grid grid-cols-2 gap-2 sm:grid-cols-4">
   370→            {facts.map(([k, v]) => (
   371→              <div key={k} className="rounded-lg bg-cloud/60 p-3">
   372→                <p className="text-[11px] text-stone">{k}</p>
   373→                <p className="num mt-0.5 truncate text-sm text-ink" title={v}>
   374→                  {v}
   375→                </p>
   376→              </div>
   377→            ))}
   378→          </div>
   379→
   380→          <div className="mt-6 space-y-6">
   381→            <MqrLedger position={position} />
   382→            <RichSection title="Brief Description" text={job?.briefDescription} html={job?.briefDescriptionHtml} />
   383→            <RichSection title="Duties & Responsibilities" text={job?.dutiesResponsibilities} html={job?.dutiesHtml} />
   384→            <RichSection title="Compensation Package" text={job?.compensationPackage} html={job?.compensationHtml} />
   385→            <RichSection title="Other Qualifications" text={job?.otherQualifications} html={job?.otherQualificationsHtml} />
   386→          </div>
   387→
   388→          <div className="mt-8 flex flex-col gap-3 border-t border-border pt-5 sm:flex-row sm:justify-end">
   389→            {cancellable && (
   390→              <button
   391→                type="button"
   392→                onClick={() => setConfirmOpen(true)}
   393→                disabled={cancelling}
   394→                className="inline-flex min-h-[44px] items-center justify-center rounded-full border border-[var(--bad)]/30 bg-white px-6 py-2.5 text-sm font-medium text-[var(--bad)] transition-colors hover:bg-[var(--bad-bg)] disabled:opacity-50"
   395→              >
   396→                Cancel Application
   397→              </button>
   398→            )}
   399→            <button type="button" onClick={onClose} className="dlg-ghost min-h-[44px] px-6 py-2.5 text-sm">
   400→              Close

   402→          </div>

   408→            <AlertDialogHeader>

   413→              </AlertDialogDescription>
   414→            </AlertDialogHeader>
   415→            <AlertDialogFooter>
   416→              <AlertDialogCancel className="dlg-ghost min-h-[44px] px-5 py-2.5 text-sm">Keep it</AlertDialogCancel>

   418→                onClick={(e) => {
   419→                  e.preventDefault();

   421→                }}
   422→                disabled={cancelling}
   423→                className="inline-flex min-h-[44px] items-center justify-center rounded-full border border-[var(--bad)]/30 bg-white px-5 py-2.5 text-sm font-medium text-[var(--bad)] transition-colors hover:bg-[var(--bad-bg)] disabled:opacity-50"
   424→              >

   427→            </AlertDialogFooter>

   429→        </AlertDialog>

   455→/** The board card register, ported: momentum rule, humanized title + gold
   456→ *  Applied pill, pipe-separated vitals (place | type | pay | closes), square
   457→ *  ember "+/−" toggle and the in-place quick view (clamped description ·
   458→ *  six QuickFacts · actions). "Apply now" and "Read full description" both
   459→ *  deep-link into the job's full posting on the Positions board. */
   460→function JobListCard({
   461→  job,
   462→  applied,
   463→  expanded,
   464→  onToggle,
   465→  onOpen,
   466→}: {
   467→  job: JobWire;
   468→  applied: boolean;
   469→  expanded: boolean;
   470→  onToggle: () => void;
   471→  onOpen: () => void;
   472→}) {
   473→  const reduced = useReducedMotion();
   474→  const pos = job.position;
   475→  const title = pos?.positionTitle || job.title || "Position Title Unavailable";
   476→  const place = jobPlace(job);
   477→  const salary = pos?.salaryAmount ?? null;
   478→  const deadline = job.deadlineDate;
   479→  const overdue = deadline ? new Date(deadline).getTime() < Date.now() : false;
   480→
   481→  // Meta — pipe-separated vitals, the card register (place | type | pay |
   482→  // closes); overdue flips the closer to red.
   483→  const meta: { label: string; className?: string }[] = [{ label: place }];
   484→  if (job.positionType) meta.push({ label: humanize(job.positionType) });
   485→  if (salary != null) meta.push({ label: `${formatCurrency(salary)} monthly`, className: "num" });
   486→  meta.push({
   487→    label: overdue ? "Closed" : deadline ? `Closes ${formatDate(deadline)}` : "Open until filled",
   488→    className: overdue ? "font-semibold text-[var(--bad)]" : "num",
   489→  });
   490→
   491→  return (
   492→    <div className="dlg-card lift group relative overflow-hidden">
   493→      {/* Metric-blue rule — wipes in across the card on hover */}
   494→      <span
   495→        aria-hidden="true"

   602→                  </span>
   603→                )}

   605→            </div>
   606→          </motion.div>

   667→        apiFetch<JobWire[]>("/api/jobs"),

   681→      if (!document.hidden) void reload(true);

   687→    window.addEventListener("focus", onFocus);
   688→    document.addEventListener("visibilitychange", onVisible);
