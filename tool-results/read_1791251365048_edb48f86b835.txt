   127→   127→/** 3-checkpoint tracking timeline: Submitted → Review → Decision (§7.3). */
   128→   128→function JourneyTimeline({ status }: { status: string }) {
   129→   129→  const stage = stageForStatus(status);
   130→   130→  const decided = stage === "Shortlisted" || stage === "Rejected";
   131→   131→
   132→   132→  const steps: { label: string; short?: string; state: CheckpointState }[] = [
   133→   133→    { label: "Submitted", state: "done" },
   134→   134→    {
   135→   135→      label: "Review",
   136→   136→      state: stage === "Under Review" ? "current" : decided ? "done" : "pending",
   137→   137→    },
   138→   138→    {
   139→   139→      label:
   140→   140→        stage === "Shortlisted" ? "Shortlisted" : stage === "Rejected" ? "Not Shortlisted" : "Awaiting the shortlist decision",
   141→   141→      // Responsive copy swap (3-d, presentation-only): the full label wraps to 4+ lines
   142→   142→      // inside a 3-up timeline at 320px; "Shortlisted"/"Not Shortlisted" stay verbatim.
   143→   143→      short: stage === "Shortlisted" || stage === "Rejected" ? undefined : "Awaiting decision",
   144→   144→      state: stage === "Shortlisted" ? "done" : stage === "Rejected" ? "failed" : "pending",
   145→   145→    },
   146→   146→  ];
   147→   147→
   148→   148→  return (
   149→   149→    <div className="mt-3 grid grid-cols-3 gap-1">
   150→   150→      {steps.map((s, i) => {
   151→   151→        const reached = s.state !== "pending";
   152→   152→        return (
   153→   153→          <div key={`${s.label}-${i}`} className="flex flex-col items-center gap-1.5 text-center">
   154→   154→            <div className="flex w-full items-center">
   155→   155→              {/* Traversed connectors render in solid brand blue. Negative
   156→   156→                  outer margins bridge the grid's gap-1 so each rail reads
   157→   157→                  as one continuous line, not segments with center gaps. */}
   158→   158→              <div
   159→   159→                className={cn(
   160→   160→                  "h-px flex-1",
   161→   161→                  i > 0 && "-ml-1",
   162→   162→                  i === 0 ? "bg-transparent" : reached ? "bg-primary" : "bg-divider"
   163→   163→                )}
   164→   164→              />
   165→   165→              <CheckpointDot state={s.state} />
   166→   166→              <div
   167→   167→                className={cn(
   168→   168→                  "h-px flex-1",
   169→   169→                  i < steps.length - 1 && "-mr-1",
   170→   170→                  i === steps.length - 1
   171→   171→                    ? "bg-transparent"
   172→   172→                    : steps[i + 1].state !== "pending"
   173→   173→                      ? "bg-primary"
   174→   174→                      : "bg-divider"
   175→   175→                )}
   176→   176→              />
   177→   177→            </div>
   178→   178→            <span className={cn("text-[10px] leading-tight", s.state === "current" ? "font-medium text-ink" : "text-stone")}>
   179→   179→              {s.short ? (
   180→   180→                <>
   181→   181→                  <span className="sm:hidden">{s.short}</span>
   182→   182→                  <span className="hidden sm:inline">{s.label}</span>
   183→   183→                </>
   184→   184→              ) : (
   185→   185→                s.label
   186→   186→              )}
   187→   187→            </span>
   188→   188→          </div>
   189→   189→        );
   190→   190→      })}
   191→   191→    </div>
   192→   192→  );
   193→   193→}

   202→   202→const SAVED_KEY = "rmis-frontpage-saved-jobs";
   203→   203→
   204→   204→/** Client-persisted bookmark list (same store as the frontpage). */
   205→   205→function useSavedJobs() {
   206→   206→  const [saved, setSaved] = useState<number[]>(() => {
   207→   207→    if (typeof window === "undefined") return [];
   208→   208→    try {
   209→   209→      const raw = window.localStorage.getItem(SAVED_KEY);
   210→   210→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
   211→   211→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
   212→   212→    } catch {
   213→   213→      return [];
   214→   214→    }
   215→   215→  });
   216→   216→
   217→   217→  const toggle = useCallback((id: number) => {
   218→   218→    setSaved((prev) => {
   219→   219→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
   220→   220→      try {
   221→   221→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
   222→   222→      } catch {
   223→   223→        /* storage unavailable — in-session state still works */
   224→   224→      }
   225→   225→      return next;
   226→   226→    });
   227→   227→  }, []);
   228→   228→
   229→   229→  return { saved, toggle };
   230→   230→}
