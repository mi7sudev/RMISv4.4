    78→const SAVED_KEY = "rmis-frontpage-saved-jobs";

    80→/** Client-persisted bookmark list for the frontpage explorer. */
    81→function useSavedJobs() {
    82→  const [saved, setSaved] = useState<number[]>(() => {
    83→    if (typeof window === "undefined") return [];
    84→    try {
    85→      const raw = window.localStorage.getItem(SAVED_KEY);
    86→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
    87→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
    88→    } catch {
    89→      return [];
    90→    }
    91→  });
    92→
    93→  const toggle = useCallback((id: number) => {
    94→    setSaved((prev) => {
    95→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
    96→      try {
    97→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
    98→      } catch {
    99→        /* storage unavailable — in-session state still works */
   100→      }
   101→      return next;
   102→    });
   103→  }, []);
   104→
   105→  return { saved, toggle };
   106→}

   287→/** Plain-text posting section rendered as a disc bullet list. */
   288→function TextBullets({ text }: { text: string }) {
   289→  const items = text
   290→    .split(/\r?\n/)
   291→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   292→    .filter(Boolean);
   293→  if (items.length === 0) return null;
   294→  return (
   295→    <ul className="list-disc space-y-2 pl-5 text-base leading-relaxed text-stone">
   296→      {items.map((item, i) => (
   297→        <li key={i}>{item}</li>
   298→      ))}
   299→    </ul>
   300→  );
   301→}

   303→/** Posting section: sanitized rich HTML when present, else text bullets. */
   304→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {
   305→  if (html && html.trim()) {
   306→    return (
   307→      <section className="space-y-3">
   308→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   309→        <div className="rich-text text-base leading-relaxed text-stone" dangerouslySetInnerHTML={{ __html: html }} />
   310→      </section>
   311→    );
   312→  }
   313→  if (text && text.trim()) {
   314→    return (
   315→      <section className="space-y-3">
   316→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   317→        <TextBullets text={text} />
   318→      </section>
   319→    );
   320→  }
   321→  return null;
   322→}

   333→  onApply: () => void;
   334→  onSave: () => void;
   335→  onBack: () => void;

   338→  const title = pos?.positionTitle || job.title || "Position Title Unavailable";

   341→  const overdue = deadline ? new Date(deadline).getTime() < Date.now() : false;
   342→

   453→export default function PublicLanding() {
   454→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   455→  const [error, setError] = useState<string | null>(null);
   456→  const [selectedId, setSelectedId] = useState<number | null>(null);
   457→  const [expandedId, setExpandedId] = useState<number | null>(null);
   458→  const [isDesktop, setIsDesktop] = useState(false);
   459→  const lastJson = useRef("");
   460→  const paneRef = useRef<HTMLDivElement>(null);
   461→  const { saved, toggle: toggleSave } = useSavedJobs();
   462→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   463→  // applicants get the confirm flow (same as the board), incomplete profiles
   464→  // are sent to finish their profile first (the fast-track was retired).
   465→  const { requestApply, dialogs } = useApplyFlow();
   466→
   467→  const load = useCallback(async (silent: boolean) => {
   468→    try {
   469→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   470→      const json = JSON.stringify(data);
   471→      if (!silent || json !== lastJson.current) {
   472→        lastJson.current = json;
   473→        setJobs(data);
   474→      }
   475→      setError(null);
   476→    } catch (e) {
   477→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   478→    }
   479→  }, []);
   480→
   481→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   482→  useEffect(() => {
   483→    void load(false);
   484→    const timer = setInterval(() => {
   485→      if (!document.hidden) void load(true);
   486→    }, 30_000);
   487→    const onFocus = () => void load(true);
   488→    window.addEventListener("focus", onFocus);
   489→    return () => {
   490→      clearInterval(timer);
   491→      window.removeEventListener("focus", onFocus);
   492→    };
   493→  }, [load]);
   494→
   495→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   496→  useEffect(() => {
   497→    const mq = window.matchMedia("(min-width: 1024px)");
   498→    const sync = () => setIsDesktop(mq.matches);
   499→    sync();
   500→    mq.addEventListener("change", sync);
   501→    return () => mq.removeEventListener("change", sync);
   502→  }, []);
   503→
   504→  const openJobs = useMemo(
   505→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   506→    [jobs]
   507→  );
   508→
   509→  const selectedJob = useMemo(
   510→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   511→    [openJobs, selectedId]
   512→  );
   513→
   514→  // Desktop opens with the first posting pre-selected (reference-board
   515→  // behavior); a stale selection after a poll snaps back to the top card.
   516→  useEffect(() => {
   517→    if (isDesktop && selectedId === null && openJobs.length > 0) {
   518→      setSelectedId(openJobs[0].id);
   519→    }
   520→  }, [isDesktop, openJobs, selectedId]);
   521→
   522→  useEffect(() => {
   523→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   524→      setSelectedId(openJobs[0]?.id ?? null);
   525→    }
   526→  }, [openJobs, selectedId]);
   527→
   528→  // Reference-board behavior — a freshly picked posting opens from the top,
   529→  // even when the previous posting was scrolled deep into its body.
   530→  useEffect(() => {
   531→    if (paneRef.current) paneRef.current.scrollTop = 0;
   532→  }, [selectedId]);
   533→
   534→  const select = useCallback(
   535→    (id: number) => {
   536→      setSelectedId(id);
   537→      if (!window.matchMedia("(min-width: 1024px)").matches) {
   538→        window.scrollTo({ top: 0, behavior: "smooth" });
   539→      }
   540→    },
   541→    []
   542→  );
   543→
   544→  const saveAndToast = useCallback(
   545→    (job: JobWire) => {
   546→      const wasSaved = saved.includes(job.id);
   547→      toggleSave(job.id);
   548→      toast.success(
   549→        wasSaved
   550→          ? "Removed from your saved jobs"
   551→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   552→      );
   553→    },
   554→    [saved, toggleSave]
   555→  );
   556→
   557→  const showList = isDesktop || selectedId === null;
   558→  const showDetail = isDesktop || selectedId !== null;
   559→
   560→  return (
   561→    // Viewport-adaptive board (no app pinning): a normal min-h-screen flex
   562→    // column whose footer always sits at the real bottom — pinned to the
   563→    // viewport on short pages (mt-auto), pushed down naturally on tall ones.
   564→    // Each board column (list + posting) is its own sticky, viewport-capped
   565→    // scroller — see the grid below.
   566→    <div className="flex min-h-screen flex-col">
   567→      <SiteHeader />
   568→
   569→      <main className="mx-auto w-full max-w-[1600px] flex-1 px-4 pb-10 pt-6 sm:px-6 lg:px-8">
   570→        {/* Hero — Bevel "Hero Sky" field (rounded 32) anchoring the explorer,
   571→            arranged as a split: display headline + copy + single charcoal
   572→            capsule CTA left; official brand mark + authority line right as
   573→            the quiet institutional proof strip (real marks only). */}
   574→        <section className="hero-sky rounded-[32px] p-6 sm:p-8 lg:px-10 lg:py-10 lg:[@media(max-height:860px)]:py-8">
   575→          <div className="grid items-center gap-8 lg:grid-cols-[minmax(0,1fr)_auto]">
   576→            {/* QA 2026-02: header renamed to "Job Opportunities" to match the
   577→                Positions tab; the live "N open" pill was removed — it read as
   578→                an unclickable button. */}
   579→            <div className="min-w-0">
   580→              <h1 className="text-display-hero">Job Opportunities</h1>
   581→              <p className="text-body mt-3 max-w-xl text-body-gray">
   582→                Current vacancies at DOST-MIRDC. Pick a role to read the full posting.
   583→              </p>
   584→              <div className="mt-6">
   585→                <button
   586→                  onClick={() => navigate("vacant-positions")}
   587→                  className="dlg-cta text-nav inline-flex min-h-[44px] items-center gap-2 px-4 font-medium"
   588→                >
   589→                  Browse the full board
   590→                  <ArrowRight className="h-4 w-4" aria-hidden="true" />
   591→                </button>
   592→              </div>
   593→            </div>
   594→            {/* Institutional proof — the same emblems and authority line the
   595→                footer carries, moved into the hero as the trust anchor. */}
   596→            <div className="flex items-center gap-4 lg:flex-col lg:items-end lg:gap-3 lg:text-right">
   597→              <BrandLogo logoSize={36} tone="light" />
   598→              <div className="min-w-0">
   599→                <p className="text-sm font-semibold leading-snug text-ink">
   600→                  Metals Industry Research and Development Center
   601→                </p>
   602→                <p className="mt-0.5 text-xs leading-relaxed text-stone">
   603→                  An agency of the Department of Science and Technology, Republic of the Philippines
   604→                </p>
   605→              </div>
   606→            </div>
   607→          </div>
   608→        </section>
   609→
   610→        {error !== null ? (
   611→          <div className="mt-6 rounded-2xl bg-[var(--bad-bg)] p-5">
   612→            <p className="text-sm text-ink">{error}</p>
   613→            <button
   614→              onClick={() => void load(false)}
   615→              className="mt-2 text-sm text-metric-blue underline underline-offset-2"
   616→            >
   617→              Try again
   618→            </button>
   619→          </div>
   620→        ) : jobs === null ? (
   621→          <div className="mt-6 space-y-3.5" aria-hidden="true">
   622→            {[0, 1, 2].map((i) => (
   623→              <div key={i} className="skel h-28" />
   624→            ))}
   625→          </div>
   626→        ) : openJobs.length === 0 ? (
   627→          <div className="dlg-card mt-6">
   628→            <EmptyState
   629→              icon={Inbox}
   630→              title="No open positions right now"
   631→              description="Check back soon or explore the full board for upcoming announcements."
   632→            />
   633→            <div className="pb-8 text-center">
   634→              <button
   635→                onClick={() => navigate("vacant-positions")}
   636→                className="inline-flex min-h-[44px] items-center gap-2 text-sm text-metric-blue underline underline-offset-2 hover:text-ink"
   637→              >
   638→                Go to the full positions board
   639→                <ArrowRight className="h-4 w-4" aria-hidden="true" />
   640→              </button>
   641→            </div>
   642→          </div>
   643→        ) : (
   644→          // Viewport-adaptive two-pane board. The posting pane sizes to its
   645→          // own content and contributes that height to the row (a short list
   646→          // no longer squeezes it), capped at the LIVE viewport height
   647→          // (100dvh − 88px sticky offset − 24px breathing room — no fixed
   648→          // pixel constant), so:
   649→          //   • tall screens — the complete description renders in full with
   650→          //     no internal scrolling; the footer only ever sits BELOW it;
   651→          //   • short screens — the pane caps at the available space and
   652→          //     scrolls internally while sticky under the header (the
   653→          //     reference board's "sticky description"), and the grid-row
   654→          //     clamp keeps it above the footer at every scroll position.
   655→          // The job list is deliberately NOT sticky — a plain block that
   656→          // scrolls with the page (user clarification). The 400px floor
   657→          // keeps the board usable when both columns are short.
   658→          <div className="mt-6 grid gap-8 sm:mt-8 lg:min-h-[400px] lg:grid-cols-[minmax(340px,430px)_minmax(0,1fr)]">
   659→            {/* Left · the register — plain block, natural height, scrolls
   660→                with the page (NOT sticky, per user clarification). */}
   661→            {showList && (
   662→              <div
   663→                aria-label="Job opportunities list"
   664→                className="scroll-thin space-y-3.5"
   665→              >
   666→                {openJobs.map((job) => (
   667→                  <JobListCard
   668→                    key={job.id}
   669→                    job={job}
   670→                    selected={selectedId === job.id}
   671→                    expanded={expandedId === job.id}
   672→                    saved={saved.includes(job.id)}
   673→                    onSelect={() => select(job.id)}
   674→                    onToggle={() => setExpandedId(expandedId === job.id ? null : job.id)}
   675→                    onSave={() => saveAndToast(job)}
   676→                  />
   677→                ))}
   678→              </div>
   679→            )}
   680→
   681→            {/* Right · the posting — ONE sticky, viewport-capped scroller
   682→                (paneRef sits on the scroller itself, so the select-reset
   683→                actually zeroes it). Its natural height is the full posting,
   684→                capped at the live viewport: complete content with no inner
   685→                scroll when the screen has room; capped + internally
   686→                scrollable when it doesn't. Sticky keeps it in view while the
   687→                list scrolls with the page; the grid-row clamp keeps it above
   688→                the footer. On mobile it is a plain block (single-pane swap,
   689→                page scroll). */}
   690→            {showDetail && (
   691→              <div
   692→                ref={paneRef}
   693→                className="scroll-thin lg:sticky lg:top-[88px] lg:self-start lg:max-h-[calc(100dvh-112px)] lg:overflow-y-auto lg:pr-2"
   694→              >
   695→                {selectedJob ? (
   696→                  <div className="dlg-card p-6 sm:p-8">
   697→                    <JobDetail
   698→                      job={selectedJob}
   699→                      saved={saved.includes(selectedJob.id)}
   700→                      onApply={() => requestApply(selectedJob)}
   701→                      onSave={() => saveAndToast(selectedJob)}
   702→                      onBack={() => setSelectedId(null)}
   703→                    />
   704→                  </div>
   705→                ) : (
   706→                  <div className="dlg-card">
   707→                    <EmptyState
   708→                      icon={Briefcase}
   709→                      title="Select a position"
   710→                      description="Choose a role from the list to read the full posting."
   711→                    />
   712→                  </div>
   713→                )}
   714→              </div>
   715→            )}
   716→          </div>
   717→        )}
   718→      </main>
   719→
   720→      <Footer />
   721→      {/* Shared apply dialog (confirm / MQR / done), mounted once. */}
   722→      {dialogs}
   723→    </div>
   724→  );
   725→}
