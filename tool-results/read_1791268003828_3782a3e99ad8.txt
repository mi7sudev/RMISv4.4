    28→// ── PageHeader ──────────────────────────────────────────────────────────────
    29→// The single enterprise page-header anatomy used by EVERY view:
    30→// optional labelled back link (detail views) → eyebrow → title row
    31→// (title + titleAside badge) → description → meta row, with the action
    32→// cluster right-aligned and stacked below the title below xl.
    33→// ────────────────────────────────────────────────────────────────────────────
    34→
    35→export function PageHeader({
    36→  title,
    37→  description,
    38→  actions,
    39→  eyebrow,
    40→  back,
    41→  leading,
    42→  titleAside,
    43→  meta,
    44→  className,
    45→}: {
    46→  title: React.ReactNode;
    47→  description?: React.ReactNode;
    48→  actions?: React.ReactNode;
    49→  eyebrow?: React.ReactNode;
    50→  /** Detail-view affordance: labelled back link rendered above the title. */
    51→  back?: { label: string; onClick: () => void };
    52→  /** Avatar / icon lockup rendered left of the title block. */
    53→  leading?: React.ReactNode;
    54→  /** Badge / status pill rendered inline after the title. */
    55→  titleAside?: React.ReactNode;
    56→  /** Secondary row under the description (pills, stamps, counts). */
    57→  meta?: React.ReactNode;
    58→  className?: string;
    59→}) {
    60→  // NOTE: no self animation — AppShell owns the view mount transition, and a
    61→  // second fade inside the header produced a double-fade on every navigation.
    62→  return (
    63→    <header className={className}>
    64→      {back ? (
    65→        <button
    66→          type="button"
    67→          onClick={back.onClick}
    68→          className="focus-ring -ml-3 mb-3 inline-flex min-h-[36px] items-center gap-1.5 rounded-full px-3 text-sm font-medium text-stone transition-colors hover:bg-cloud hover:text-ink"
    69→        >
    70→          <ArrowLeft className="h-4 w-4" aria-hidden="true" />
    71→          {back.label}
    72→        </button>
    73→      ) : null}
    74→      {/* Stacks below xl so wide action clusters (filters, toggles, refresh) can
    75→          never squeeze the title into a truncated stub on tablet widths. */}
    76→      <div className="flex flex-col gap-4 xl:flex-row xl:items-end xl:justify-between">
    77→        <div className="flex min-w-0 items-center gap-3 sm:gap-4">
    78→          {leading ? <div className="shrink-0">{leading}</div> : null}
    79→          <div className="min-w-0">
    80→            {eyebrow ? (
    81→              <p className="eyebrow mb-1.5">{eyebrow}</p>
    82→            ) : null}
    83→            <div className="flex flex-wrap items-center gap-x-3 gap-y-2">
    84→              <h1 className="font-display truncate text-2xl font-semibold tracking-[-0.02em] text-ink sm:text-[28px]">
    85→                {title}
    86→              </h1>
    87→              {titleAside}
    88→            </div>
    89→            {description ? (
    90→              typeof description === "string" ? (
    91→                <p className="mt-1.5 text-sm leading-relaxed text-stone">{description}</p>
    92→              ) : (
    93→                <div className="mt-1.5 text-sm leading-relaxed text-stone">{description}</div>
    94→              )
    95→            ) : null}
    96→            {meta ? (
    97→              <div className="mt-2 flex flex-wrap items-center gap-x-2 gap-y-1">{meta}</div>
    98→            ) : null}
    99→          </div>
   100→        </div>
   101→        {actions ? <div className="flex shrink-0 flex-wrap items-center gap-2">{actions}</div> : null}
   102→      </div>
   103→    </header>
   104→  );
   105→}

   109→// THE one status pill. Displays the normalized status label ("UNDER_REVIEW"
   110→// renders as "Under Review"); the variant auto-maps from the status tone
   111→// unless explicitly overridden (used for non-application labels such as
   112→// verdicts, completion or document states).
   113→export function StatusPill({
   114→  status,
   115→  variant,
   116→  className,
   117→}: {
   118→  /** Raw stored status spelling (normalized internally). */
   119→  status?: string | null;
   120→  /** Explicit variant overrides the status→tone mapping. */
   121→  variant?: StatusVariant;
   122→  className?: string;
   123→}) {
   124→  const meta = status ? getStatusMeta(status) : null;
   125→  const v: StatusVariant = variant ?? (meta ? variantForStatus(status) : "neutral");
   126→  return (
   127→    <span className={cn(pillClass(v), "shrink-0", className)}>
   128→      {meta ? meta.label : (status ?? "")}
   129→    </span>
   130→  );
   131→}

   133→// ── VerdictPill ─────────────────────────────────────────────────────────────
   134→
   135→function verdictMeta(verdict: string): { chip: string; variant: StatusVariant } {
   136→  switch (verdict) {
   137→    case "ALL_MET":
   138→      return { chip: "Qualified", variant: variantForVerdict("Qualified") };
   139→    case "PARTIAL":
   140→      return { chip: "Partial", variant: variantForVerdict("Partial") };
   141→    case "NONE_MET":
   142→      return { chip: "Not qualified", variant: variantForVerdict("Not qualified") };
   143→    case "NEEDS_REVIEW":
   144→      return { chip: "Verify", variant: variantForVerdict("Verify") };
   145→    case "NO_REQUIREMENTS":
   146→      return { chip: "No reqs", variant: variantForVerdict("No reqs") };
   147→    default:
   148→      return { chip: verdict, variant: variantForVerdict(verdict) };
   149→  }
   150→}

   153→export function VerdictPill({
   154→  verdict,
   155→  className,
   156→}: {
   157→  verdict: string | null | undefined;

   167→/** First letters of the first two words, uppercase ("Juan Cruz" → "JC"). */
   168→export function nameInitials(name: string | null | undefined): string {
   169→  return (
   170→    (name || "")
   171→      .split(/\s+/)
   172→      .filter(Boolean)
   173→      .slice(0, 2)
   174→      .map((w) => w[0]?.toUpperCase() ?? "")
   175→      .join("") || "?"
   176→  );
   177→}

   204→export function SegmentedControl<T extends string>({
   205→  value,
   206→  onChange,
   207→  options,
   208→  ariaLabel,
   209→  className,
   210→}: {
   211→  value: T;
   212→  onChange: (next: T) => void;
   213→  options: { value: T; label: string; icon?: LucideIcon }[];
   214→  ariaLabel: string;
   215→  className?: string;
   216→}) {

   218→    <div
   219→      role="group"
   220→      aria-label={ariaLabel}
   221→      className={cn("inline-flex items-center rounded-full bg-fog p-1", className)}
   222→    >
   223→      {options.map((opt) => {
   224→        const active = opt.value === value;
   225→        const Icon = opt.icon;
   226→        return (
   227→          <button
   228→            key={opt.value}
   229→            type="button"
   230→            aria-pressed={active}
   231→            onClick={() => onChange(opt.value)}
   232→            className={cn(
   233→              "focus-ring inline-flex min-h-[36px] items-center justify-center gap-1.5 rounded-full px-3 text-xs font-medium transition-colors",
   234→              active ? "bg-white text-ink shadow-e1" : "text-stone hover:text-ink"
   235→            )}
   236→          >
   237→            {Icon ? <Icon className="h-3.5 w-3.5" aria-hidden="true" /> : null}
   238→            {opt.label}
   239→          </button>
   240→        );
   241→      })}
   242→    </div>
   243→  );
   244→}

   395→export function KpiCard({
   396→  label,
   397→  value,
   398→  hint,
   399→  icon: Icon,
   400→  tone = "neutral",
   401→  onClick,
   402→  ariaLabel,
   403→  className,

   410→        {Icon ? <IconChip icon={Icon} tone={TONE_CHIP[tone]} size={40} /> : null}

   417→        {aside}
   418→      </div>
   419→    </>

   423→    "dlg-card p-6",
   424→    onClick && "group focus-ring cursor-pointer lift",
   425→    className
   426→  );
   427→
   428→  if (onClick) {
   429→    return (
   430→      <button type="button" onClick={onClick} aria-label={ariaLabel ?? label} className={cn(base, "block w-full text-left")}>
   431→        {body}
   432→      </button>
   433→    );
   434→  }
   435→  return <div className={base}>{body}</div>;

   569→// ── SectionCard ─────────────────────────────────────────────────────────────
   570→// Bevel Cloud Feature Card: #ebf0f8 surface, 24px radius, no border, no
   571→// shadow, 32px padding (p-6 keeps in-app density; p-8 on wide screens).
   572→
   573→export function SectionCard({
   574→  title,
   575→  description,
   576→  actions,
   577→  icon: Icon,
   578→  chipTone,
   579→  children,
   580→  className,
   581→  bodyClassName,
   582→  /** "cloud" = pale Cloud Card (default) · "plain" = white panel for content
   583→   *  sitting inside dialogs / on cloud surfaces (profile section anatomy). */
   584→  variant = "cloud",
   585→}: {
   586→  title?: string;
   587→  description?: string;
   588→  actions?: React.ReactNode;
   589→  icon?: LucideIcon;
   590→  chipTone?: ChipTone;
   591→  children: React.ReactNode;
   592→  className?: string;
   593→  bodyClassName?: string;
   594→  variant?: "cloud" | "plain";
   595→}) {
   596→  // min-w-0: as a CSS-grid child the card must be allowed to shrink below the
   597→  // min-content floor of its nowrap description, or the grid track blows out
   598→  // and the card overflows its viewport (observed at 320–1279px).
   599→  return (
   600→    <section
   601→      className={cn(
   602→        variant === "plain" ? "dlg-card-plain min-w-0 p-5 sm:p-6" : "dlg-card min-w-0 p-6",
   603→        className
   604→      )}
   605→      aria-label={title}
   606→    >
   607→      {title || actions ? (
   608→        <div className="mb-5 flex flex-wrap items-start justify-between gap-3">
   609→          <div className="flex min-w-0 items-center gap-3">
   610→            {Icon ? (
   611→              <IconChip
   612→                icon={Icon}
   613→                tone={chipTone ?? "slate"}
   614→                size={36}
   615→                iconSize={16}
   616→              />
   617→            ) : null}
   618→            <div className="min-w-0">
   619→              <h2 className="font-display truncate text-lg leading-7 text-ink">{title}</h2>
   620→              {/* Wraps below sm, truncates from sm up. Full-width nowrap (truncate
   621→                  everywhere) gave the card a huge min-content floor that blew out
   622→                  auto grid tracks on phones; wrapping text has none. */}
   623→              {description ? <p className="text-xs text-stone sm:truncate">{description}</p> : null}
   624→            </div>
   625→          </div>
   626→          {actions ? <div className="flex shrink-0 flex-wrap items-center gap-2">{actions}</div> : null}
   627→        </div>
   628→      ) : null}
   629→      <div className={bodyClassName}>{children}</div>
   630→    </section>
   631→  );
   632→}
