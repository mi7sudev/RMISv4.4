     1→     1→// ============================================================================
     2→     2→// email.ts — provider-agnostic EMAIL sender for RMIS notifications.
     3→     3→//
     4→     4→// Mirrors the audit-trail design of the notification surfaces:
     5→     5→//
     6→     6→//   1. "mock" (default, dev) — FREE forever. No external service: every send
     7→     7→//      is logged to the `email_logs` table so the whole flow (shortlist
     8→     8→//      notice included) is testable end-to-end and auditable in the admin
     9→     9→//      Email panel.
    10→    10→//   2. "resend" — production-grade HTTP API. Free tier (100 emails/day),
    11→    11→//      no SDK needed — plain fetch. Set RESEND_API_KEY + (optional)
    12→    12→//      EMAIL_FROM. Perfect fit: shortlist notices are low-volume, so the
    13→    13→//      free tier realistically costs ₱0 for this office.
    14→    14→//   3. "middleware" — the office's OWN email gateway (mirdc_middleware, an
    15→    15→//      Express+nodemailer service on the MIRDC VM). Sends through the
    16→    16→//      office mailbox systems@mirdc.dost.gov.ph over office SMTP. Set
    17→    17→//      EMAIL_MIDDLEWARE_URL (+ EMAIL_MIDDLEWARE_API_KEY). ₱0 cost, no
    18→    18→//      external vendor, works on the intranet.
    19→    19→//
    20→    20→// CONFIG (env vars):
    21→    21→//   EMAIL_PROVIDER=mock|resend|middleware
    22→    22→//   # resend:
    23→    23→//   RESEND_API_KEY=<key from resend.com>
    24→    24→//   EMAIL_FROM="DOST-MIRDC Recruitment <recruitment@mirdc.gov.ph>"
    25→    25→//   # middleware:
    26→    26→//   EMAIL_MIDDLEWARE_URL=http://<vm-host>:3232
    27→    27→//   EMAIL_MIDDLEWARE_API_KEY=<shared middleware token>
    28→    28→//   EMAIL_MIDDLEWARE_SYSTEM="DOST-MIRDC Recruitment"  (From display name)
    29→    29→//
    30→    30→// Design rules:
    31→    31→//   - sendEmail() NEVER throws — callers fire it after their own success
    32→    32→//     path and a provider outage must never break a status update.
    33→    33→//     Results are always persisted to `email_logs`.
    34→    34→//   - Templates: the SHORTLIST notice is the flagship (rich HTML + text);
    35→    35→//     every other status change gets a simple, honest plain-text notice.

   137→   137→const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;
   138→   138→
   139→   139→/** Fire an email through the active provider. Never throws. */
   140→   140→export async function sendEmail(input: SendEmailInput): Promise<SendEmailResult> {
   141→   141→  const provider = getEmailProviderId();
   142→   142→  const to = (input.to ?? "").trim();
   143→   143→
   144→   144→  // Unreachable/absent address — record the skip so admins can see why an
   145→   145→  // applicant got no email.
   146→   146→  if (!to || !EMAIL_RE.test(to)) {
   147→   147→    await logEmail({
   148→   148→      to: to || "(none)",
   149→   149→      subject: input.subject,
   150→   150→      bodyText: input.text,
   151→   151→      bodyHtml: input.html ?? null,
   152→   152→      provider,
   153→   153→      status: "skipped",
   154→   154→      error: "No valid email address on record",
   155→   155→      relatedType: input.related?.type,
   156→   156→      relatedId: input.related?.id,
   157→   157→      attachmentsMeta: attachmentsMeta(input.attachments),
   158→   158→    }).catch(() => {});
   159→   159→    return { attempted: false, status: "skipped", provider };
   160→   160→  }
   161→   161→
   162→   162→  let result: { ok: boolean; ref?: string; error?: string; status: "sent" | "failed" | "mock" };
   163→   163→  try {
   164→   164→    switch (provider) {
   165→   165→      case "resend":
   166→   166→        result = await sendViaResend(
   167→   167→          to,
   168→   168→          input.subject,
   169→   169→          input.text,
   170→   170→          input.html,
   171→   171→          input.attachments
   172→   172→        );
   173→   173→        break;
   174→   174→      case "middleware":
   175→   175→        result = await sendViaMiddleware(
   176→   176→          to,
   177→   177→          input.subject,
   178→   178→          input.text,
   179→   179→          input.html,
   180→   180→          input.attachments
   181→   181→        );
   182→   182→        break;
   183→   183→      default:
   184→   184→        // mock: nothing is delivered — attachment metadata is still logged
   185→   185→        // below so the audit trail shows what WOULD have been sent.
   186→   186→        result = { ok: true, status: "mock", ref: "mock" };
   187→   187→    }
   188→   188→  } catch (e) {
   189→   189→    result = {
   190→   190→      ok: false,
   191→   191→      status: "failed",
   192→   192→      error: e instanceof Error ? e.message : String(e),
   193→   193→    };
   194→   194→  }
   195→   195→
   196→   196→  await logEmail({
   197→   197→    to,
   198→   198→    subject: input.subject,
   199→   199→    bodyText: input.text,
   200→   200→    bodyHtml: input.html ?? null,
   201→   201→    provider,
   202→   202→    status: result.status,
   203→   203→    providerRef: result.ref,
   204→   204→    error: result.error,
   205→   205→    relatedType: input.related?.type,
   206→   206→    relatedId: input.related?.id,
   207→   207→    attachmentsMeta: attachmentsMeta(input.attachments),
   208→   208→  }).catch(() => {});
   209→   209→
   210→   210→  return {
   211→   211→    attempted: true,
   212→   212→    status: result.status,
   213→   213→    provider,
   214→   214→    providerRef: result.ref,
   215→   215→    error: result.error,
   216→   216→    to,
   217→   217→  };
   218→   218→}

   220→   220→/** JSON metadata persisted to email_logs.attachments — name + size only. */
   221→   221→function attachmentsMeta(attachments?: EmailAttachment[]): string | null {
   222→   222→  if (!attachments || attachments.length === 0) return null;
   223→   223→  try {
   224→   224→    return JSON.stringify(
   225→   225→      attachments.map((a) => ({
   226→   226→        name: a.filename,
   227→   227→        bytes: Math.floor((a.content.length * 3) / 4),
   228→   228→      }))
   229→   229→    );
   230→   230→  } catch {
   231→   231→    return null;
   232→   232→  }
   233→   233→}

   262→   262→  });
   263→   263→}

   265→   265→// --- Driver: Resend (HTTP API — no SDK) ------------------------------------
   266→   266→
   267→   267→async function sendViaResend(
   268→   268→  to: string,
   269→   269→  subject: string,
   270→   270→  text: string,
   271→   271→  html?: string,
   272→   272→  attachments?: EmailAttachment[]
   273→   273→): Promise<{ ok: boolean; ref?: string; error?: string; status: "sent" | "failed" }> {
   274→   274→  const key = process.env.RESEND_API_KEY;
   275→   275→  if (!key) return { ok: false, status: "failed", error: "RESEND_API_KEY not set" };
   276→   276→
   277→   277→  const res = await fetch("https://api.resend.com/emails", {
   278→   278→    method: "POST",
   279→   279→    headers: {
   280→   280→      "Content-Type": "application/json",
   281→   281→      Authorization: `Bearer ${key}`,
   282→   282→    },
   283→   283→    body: JSON.stringify({
   284→   284→      from: DEFAULT_FROM,
   285→   285→      to: [to],
   286→   286→      subject,
   287→   287→      text,
   288→   288→      ...(html ? { html } : {}),
   289→   289→      ...(attachments && attachments.length > 0
   290→   290→        ? {
   291→   291→            attachments: attachments.map((a) => ({
   292→   292→              filename: a.filename,
   293→   293→              content: a.content,
   294→   294→              ...(a.contentType ? { content_type: a.contentType } : {}),
   295→   295→            })),
   296→   296→          }
   297→   297→        : {}),
   298→   298→    }),
   299→   299→  });
   300→   300→  if (!res.ok) {
   301→   301→    return {
   302→   302→      ok: false,
   303→   303→      status: "failed",
   304→   304→      error: `resend ${res.status}: ${await safeBody(res)}`,
   305→   305→    };
   306→   306→  }
   307→   307→  const data = (await res.json().catch(() => ({}))) as { id?: string };
   308→   308→  return { ok: true, status: "sent", ref: data?.id };
   309→   309→}
   310→   310→
   311→   311→async function safeBody(res: Response): Promise<string> {
   312→   312→  try {
   313→   313→    return (await res.text()).slice(0, 200);
   314→   314→  } catch {
   315→   315→    return "";
   316→   316→  }
   317→   317→}

   319→   319→// --- Driver: MIRDC Mail Middleware (office VM, mirdc_middleware) ------------
   320→   320→// POST {EMAIL_MIDDLEWARE_URL}/sendMail — Express+nodemailer service run by
   321→   321→// the office. The From header is ALWAYS the office mailbox
   322→   322→// systems@mirdc.dost.gov.ph (display name = EMAIL_MIDDLEWARE_SYSTEM, default
   323→   323→// "DOST-MIRDC Recruitment"); the middleware's own default Reply-To applies.
   324→   324→// The updated middleware also accepts a plain-text part and base64
   325→   325→// attachments; an OLDER middleware simply ignores those fields — emails
   326→   326→// still deliver (HTML-only), so this driver is safe against both versions.
   327→   327→
   328→   328→async function sendViaMiddleware(
   329→   329→  to: string,
   330→   330→  subject: string,
   331→   331→  text: string,
   332→   332→  html?: string,
   333→   333→  attachments?: EmailAttachment[]
   334→   334→): Promise<{ ok: boolean; ref?: string; error?: string; status: "sent" | "failed" }> {
   335→   335→  const base = env.EMAIL_MIDDLEWARE_URL;
   336→   336→  if (!base) return { ok: false, status: "failed", error: "EMAIL_MIDDLEWARE_URL not set" };
   337→   337→
   338→   338→  // The middleware delivers an HTML body only — always give it one. When a
   339→   339→  // template didn't build HTML, wrap the plain text minimally so the
   340→   340→  // message is never an empty body.
   341→   341→  const message = html ?? textToSimpleHtml(text);
   342→   342→
   343→   343→  const res = await fetch(`${base}/sendMail`, {
   344→   344→    method: "POST",
   345→   345→    headers: { "Content-Type": "application/json" },
   346→   346→    // Never hang an API route on a stalled VM — the middleware normally
   347→   347→    // answers in well under a second.
   348→   348→    signal: AbortSignal.timeout(15_000),
   349→   349→    body: JSON.stringify({
   350→   350→      // Shared middleware token (validated only when the VM opts into
   351→   351→      // MIDDLEWARE_API_KEY strict mode; harmless otherwise).
   352→   352→      api: env.EMAIL_MIDDLEWARE_API_KEY,
   353→   353→      // From display name — becomes "…" <systems@mirdc.dost.gov.ph>.
   354→   354→      system: env.EMAIL_MIDDLEWARE_SYSTEM || "DOST-MIRDC Recruitment",
   355→   355→      to,
   356→   356→      subject,
   357→   357→      message,
   358→   358→      // Additive fields — newer middleware adds the text part / attachments;
   359→   359→      // older middleware ignores them.
   360→   360→      text,
   361→   361→      ...(attachments && attachments.length > 0
   362→   362→        ? {
   363→   363→            attachments: attachments.map((a) => ({
   364→   364→              filename: a.filename,
   365→   365→              content: a.content,
   366→   366→              encoding: "base64",
   367→   367→              ...(a.contentType ? { contentType: a.contentType } : {}),
   368→   368→            })),
   369→   369→          }
   370→   370→        : {}),
   371→   371→    }),
   372→   372→  });
   373→   373→
   374→   374→  if (!res.ok) {
   375→   375→    return {
   376→   376→      ok: false,
   377→   377→      status: "failed",
   378→   378→      error: `middleware ${res.status}: ${await safeBody(res)}`,
   379→   379→    };
   380→   380→  }
   381→   381→  const data = (await res.json().catch(() => ({}))) as { messageId?: string | null };
   382→   382→  return { ok: true, status: "sent", ref: data?.messageId || "middleware" };
   383→   383→}

   385→   385→/** Minimal escaped-HTML wrapper for plain-text bodies (middleware has no text-only mode). */
   386→   386→function textToSimpleHtml(text: string): string {
   387→   387→  const paras = text
   388→   388→    .split(/\n{2,}/)
   389→   389→    .map(
   390→   390→      (p) =>
   391→   391→        `<p style="margin:0 0 12px;">${esc(p).replace(/\n/g, "<br/>")}</p>`
   392→   392→    )
   393→   393→    .join("");
   394→   394→  return `<div style="font-family:Arial,Helvetica,sans-serif;font-size:14px;line-height:1.6;color:#1a1a1a;">${paras}</div>`;
   395→   395→}

   429→   429→function esc(s: string): string {
   430→   430→  return s
   431→   431→    .replace(/&/g, "&amp;")
   432→   432→    .replace(/</g, "&lt;")
   433→   433→    .replace(/>/g, "&gt;")
   434→   434→    .replace(/"/g, "&quot;");
   435→   435→}
