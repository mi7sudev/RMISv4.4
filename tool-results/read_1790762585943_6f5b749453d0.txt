   484→   484→
   485→   485→  const filtered = useMemo(() => {

   487→   487→    if (debouncedSearch) {

   490→   490→          j.title,
   491→   491→          j.position?.itemNumber,
   492→   492→          j.position?.placeOfAssignment,
   493→   493→          j.positionType,
   494→   494→        ]
   495→   495→          .filter(Boolean)
   496→   496→          .join(" ")
   497→   497→          .toLowerCase();
   498→   498→        return hay.includes(debouncedSearch);
   499→   499→      });
   500→   500→    }
   501→   501→    if (statusFilter === "open") rows = rows.filter(isOpenJob);
   502→   502→    else if (statusFilter === "closed") rows = rows.filter((j) => !isOpenJob(j));
   503→   503→    const sorted = [...rows];
   504→   504→    if (sortBy === "recent") {
   505→   505→      sorted.sort((a, b) => new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime());
   506→   506→    } else if (sortBy === "deadline") {
   507→   507→      sorted.sort((a, b) => {
   508→   508→        const av = a.deadlineDate ? new Date(a.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   509→   509→        const bv = b.deadlineDate ? new Date(b.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   510→   510→        return av - bv;
   511→   511→      });
   512→   512→    } else {
   513→   513→      sorted.sort((a, b) => b.applicationCount - a.applicationCount);
   514→   514→    }
   515→   515→    return sorted;
   516→   516→  }, [jobs, debouncedSearch, statusFilter, sortBy]);
   517→   517→
   518→   518→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));

   521→   521→

   523→   523→    setSpin(true);
   524→   524→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));

   527→   527→  return (

   531→   531→        description={`${filtered.length} posting${filtered.length === 1 ? "" : "s"}`}
   532→   532→        actions={

   535→   535→              <RefreshCw className={`h-4 w-4 ${spin ? "animate-spin" : ""}`} aria-hidden />
   536→   536→              Refresh
   537→   537→            </button>
   538→   538→            <button type="button" className={ctaBtn} onClick={openCreate}>
   539→   539→              <Plus className="h-4 w-4" aria-hidden />
   540→   540→              Create Job
   541→   541→            </button>
   542→   542→          </>
   543→   543→        }
   544→   544→      />
   545→   545→
   546→   546→      {/* Filter bar — one card row: chip + search grows, selects fixed */}

   553→   553→            value={search}

   555→   555→            aria-label="Search job postings"

   558→   558→        <Select
   559→   559→          value={statusFilter}
   560→   560→          onValueChange={(v) => {

   562→   562→            setPage(1);
   563→   563→          }}

   565→   565→          <SelectTrigger className="dlg-input min-h-[44px] w-full sm:w-44" aria-label="Status filter">
   566→   566→            <SelectValue />
   567→   567→          </SelectTrigger>
   568→   568→          <SelectContent>
   569→   569→            <SelectItem value="all">All statuses</SelectItem>
   570→   570→            <SelectItem value="open">Open</SelectItem>
   571→   571→            <SelectItem value="closed">Closed</SelectItem>
   572→   572→          </SelectContent>
   573→   573→        </Select>
   574→   574→        <Select
   575→   575→          value={sortBy}
   576→   576→          onValueChange={(v) => {
   577→   577→            setSortBy(v as typeof sortBy);
   578→   578→            setPage(1);
   579→   579→          }}
   580→   580→        >
   581→   581→          <SelectTrigger className="dlg-input min-h-[44px] w-full sm:w-44" aria-label="Sort">
   582→   582→            <SelectValue />
   583→   583→          </SelectTrigger>
   584→   584→          <SelectContent>
   585→   585→            <SelectItem value="recent">Recently posted</SelectItem>

   614→   614→                Create Job

   617→   617→          />

   621→   621→          <div className="flex items-center gap-3 border-b border-[#ececec] px-5 py-4">

   641→   641→                </tr>
   642→   642→              </thead>

   644→   644→                {pageRows.map((j) => {
   645→   645→                  const dl = deadlineState(j.deadlineDate);
   646→   646→                  const urgent = isOpenJob(j) && dl.closingSoon;
   647→   647→                  return (
   648→   648→                    <tr

   651→   651→                      onClick={() => navigate("job", { id: String(j.id) })}
   652→   652→                    >
   653→   653→                      <td className="px-5 py-3.5">
   654→   654→                        <div className="flex items-center gap-3">
   655→   655→                          <span aria-hidden="true" className="contents">
   656→   656→                            <BrandLogo logoSize={20} />

   659→   659→                            <p className="truncate text-sm font-medium text-ink">{j.title}</p>
   660→   660→                            <p className="mt-0.5 text-xs text-stone">
   661→   661→                              {[j.position?.itemNumber, j.position?.placeOfAssignment, j.position?.division].filter(Boolean).join(" · ") || "—"}
   662→   662→                            </p>

   693→   693→                          </button>

   696→   696→                            className={iconBtn}
   697→   697→                            aria-label={`Edit ${j.title}`}
   698→   698→                            title="Edit"
   699→   699→                            onClick={(e) => {
   700→   700→                              e.stopPropagation();
   701→   701→                              openEdit(j);

   704→   704→                            <Pencil className="h-4 w-4" aria-hidden />

   709→   709→                  );

   711→   711→              </tbody>
   712→   712→            </table>
   713→   713→          </div>
   714→   714→        </div>
   715→   715→      )}
   716→   716→
   717→   717→      {/* Pagination */}
   718→   718→      {jobs !== null && filtered.length > 0 && (

   723→   723→          <div className="flex items-center gap-2">

   731→   731→              Next
   732→   732→            </button>
   733→   733→          </div>
   734→   734→        </div>
   735→   735→      )}
   736→   736→
   737→   737→      <JobFormDialog open={formOpen} onOpenChange={setFormOpen} job={editJob} onSaved={() => void load(true)} />
   738→   738→    </div>
   739→   739→  );
   740→   740→}
