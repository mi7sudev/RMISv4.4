    50→/** First-letter monogram text ("?" when empty). */
    51→function monogramOf(name: string): string {
    52→  return (
    53→    name
    54→      .split(/\s+/)
    55→      .filter(Boolean)
    56→      .slice(0, 2)
    57→      .map((w) => w[0]?.toUpperCase() ?? "")
    58→      .join("") || "?"
    59→  );

   100→      if (!silent) toast.error(e instanceof Error ? e.message : "Failed to load users");
   101→      setRows([]);
   102→    }

   104→

   121→      toast.error(e instanceof Error ? e.message : "Action failed");

   378→    setBusy(true);
   379→    try {
   380→      const body: Record<string, unknown> = { firstName: form.firstName, lastName: form.lastName, role: form.role };
