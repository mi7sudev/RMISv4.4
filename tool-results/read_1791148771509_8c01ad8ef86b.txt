    73→const SAVED_KEY = "rmis-frontpage-saved-jobs";

    75→/** Client-persisted bookmark list for the frontpage explorer. */
    76→function useSavedJobs() {
    77→  const [saved, setSaved] = useState<number[]>(() => {
    78→    if (typeof window === "undefined") return [];
    79→    try {
    80→      const raw = window.localStorage.getItem(SAVED_KEY);
    81→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
    82→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
    83→    } catch {
    84→      return [];
    85→    }
    86→  });
    87→
    88→  const toggle = useCallback((id: number) => {
    89→    setSaved((prev) => {
    90→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
    91→      try {
    92→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
    93→      } catch {
    94→        /* storage unavailable — in-session state still works */
    95→      }
    96→      return next;
    97→    });
    98→  }, []);
    99→
   100→  return { saved, toggle };
   101→}

   282→/** Plain-text posting section rendered as a disc bullet list. */
   283→function TextBullets({ text }: { text: string }) {
   284→  const items = text
   285→    .split(/\r?\n/)
   286→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   287→    .filter(Boolean);
   288→  if (items.length === 0) return null;
   289→  return (
   290→    <ul className="list-disc space-y-2 pl-5 text-base leading-relaxed text-stone">
   291→      {items.map((item, i) => (
   292→        <li key={i}>{item}</li>
   293→      ))}
   294→    </ul>
   295→  );
   296→}
   297→
   298→/** Posting section: sanitized rich HTML when present, else text bullets. */
   299→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {
   300→  if (html && html.trim()) {
   301→    return (
   302→      <section className="space-y-3">
   303→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   304→        <div className="rich-text text-base leading-relaxed text-stone" dangerouslySetInnerHTML={{ __html: html }} />
   305→      </section>
   306→    );
   307→  }
   308→  if (text && text.trim()) {
   309→    return (
   310→      <section className="space-y-3">
   311→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   312→        <TextBullets text={text} />
   313→      </section>
   314→    );
   315→  }
   316→  return null;
   317→}

   448→export default function PublicLanding() {
   449→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   450→  const [error, setError] = useState<string | null>(null);
   451→  const [selectedId, setSelectedId] = useState<number | null>(null);
   452→  const [expandedId, setExpandedId] = useState<number | null>(null);
   453→  const [isDesktop, setIsDesktop] = useState(false);
   454→  const lastJson = useRef("");
   455→  const paneRef = useRef<HTMLDivElement>(null);
   456→  const { saved, toggle: toggleSave } = useSavedJobs();
   457→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   458→  // applicants get the confirm flow (same as the board), incomplete profiles
   459→  // are sent to finish their profile first (the fast-track was retired).
   460→  const { requestApply, dialogs } = useApplyFlow();
   461→
   462→  const load = useCallback(async (silent: boolean) => {
   463→    try {
   464→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   465→      const json = JSON.stringify(data);
   466→      if (!silent || json !== lastJson.current) {
   467→        lastJson.current = json;
   468→        setJobs(data);
   469→      }
   470→      setError(null);
   471→    } catch (e) {
   472→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   473→    }
   474→  }, []);
   475→
   476→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   477→  useEffect(() => {
   478→    void load(false);
   479→    const timer = setInterval(() => {
   480→      if (!document.hidden) void load(true);
   481→    }, 30_000);
   482→    const onFocus = () => void load(true);
   483→    window.addEventListener("focus", onFocus);
   484→    return () => {
   485→      clearInterval(timer);
   486→      window.removeEventListener("focus", onFocus);
   487→    };
   488→  }, [load]);
   489→
   490→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   491→  useEffect(() => {
   492→    const mq = window.matchMedia("(min-width: 1024px)");
   493→    const sync = () => setIsDesktop(mq.matches);
   494→    sync();
   495→    mq.addEventListener("change", sync);
   496→    return () => mq.removeEventListener("change", sync);
   497→  }, []);
   498→
   499→  const openJobs = useMemo(
   500→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   501→    [jobs]
   502→  );
   503→
   504→  const selectedJob = useMemo(
   505→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   506→    [openJobs, selectedId]
   507→  );
   508→
   509→  // Desktop opens with the first posting pre-selected (reference-board
   510→  // behavior); a stale selection after a poll snaps back to the top card.
   511→  useEffect(() => {
   512→    if (isDesktop && selectedId === null && openJobs.length > 0) {
   513→      setSelectedId(openJobs[0].id);
   514→    }
   515→  }, [isDesktop, openJobs, selectedId]);
   516→
   517→  useEffect(() => {
   518→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   519→      setSelectedId(openJobs[0]?.id ?? null);
   520→    }
   521→  }, [openJobs, selectedId]);
   522→
   523→  // Reference-board behavior — a freshly picked posting opens from the top,
   524→  // even when the previous posting was scrolled deep into its body.
   525→  useEffect(() => {
   526→    if (paneRef.current) paneRef.current.scrollTop = 0;
   527→  }, [selectedId]);
   528→
   529→  const select = useCallback(
   530→    (id: number) => {
   531→      setSelectedId(id);
   532→      if (!window.matchMedia("(min-width: 1024px)").matches) {
   533→        window.scrollTo({ top: 0, behavior: "smooth" });
   534→      }
   535→    },
   536→    []
   537→  );
   538→
   539→  const saveAndToast = useCallback(
   540→    (job: JobWire) => {
   541→      const wasSaved = saved.includes(job.id);
   542→      toggleSave(job.id);
   543→      toast.success(
   544→        wasSaved
   545→          ? "Removed from your saved jobs"
   546→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   547→      );
   548→    },
   549→    [saved, toggleSave]
   550→  );
   551→
   552→  const showList = isDesktop || selectedId === null;
   553→  const showDetail = isDesktop || selectedId !== null;
   554→
   555→  return (
   556→    <div className="flex min-h-screen flex-col lg:h-dvh">
   557→      <SiteHeader />
   558→
   559→      <main className="mx-auto w-full max-w-[1600px] flex-1 px-4 pb-10 pt-6 sm:px-6 lg:flex lg:min-h-0 lg:flex-col lg:px-8">
   560→        {/* Hero — Bevel "Hero Sky" field (rounded 32) anchoring the explorer,
   561→            arranged as a split: display headline + copy + single charcoal
   562→            capsule CTA left; official seal lockup + authority line right as
   563→            the quiet institutional proof strip (real marks only). */}
   564→        <section className="hero-sky rounded-[32px] p-6 sm:p-8 lg:px-10 lg:py-10">
   565→          <div className="grid items-center gap-8 lg:grid-cols-[minmax(0,1fr)_auto]">
   566→            {/* QA 2026-02: header renamed to "Job Opportunities" to match the
   567→                Positions tab; the live "N open" pill was removed — it read as
   568→                an unclickable button. */}
   569→            <div className="min-w-0">
   570→              <h1 className="text-display-hero">Job Opportunities</h1>
   571→              <p className="text-body mt-3 max-w-xl text-body-gray">
   572→                Current vacancies at DOST-MIRDC. Pick a role to read the full posting.
   573→              </p>
   574→              <div className="mt-6">
   575→                <button
   576→                  onClick={() => navigate("vacant-positions")}
   577→                  className="dlg-cta text-nav inline-flex min-h-[44px] items-center gap-2 px-4 font-medium"
   578→                >
   579→                  Browse the full board
   580→                  <ArrowRight className="h-4 w-4" aria-hidden="true" />
   581→                </button>
   582→              </div>
   583→            </div>
   584→            {/* Institutional proof — the same emblems and authority line the
   585→                footer carries, moved into the hero as the trust anchor. */}
   586→            <div className="flex items-center gap-4 lg:flex-col lg:items-end lg:gap-3 lg:text-right">
   587→              <BrandLogo logoSize={36} tone="light" />
   588→              <div className="min-w-0">
   589→                <p className="text-sm font-semibold leading-snug text-ink">
   590→                  Metals Industry Research and Development Center
   591→                </p>
   592→                <p className="mt-0.5 text-xs leading-relaxed text-stone">
   593→                  An agency of the Department of Science and Technology, Republic of the Philippines
   594→                </p>
   595→              </div>
   596→            </div>
   597→          </div>
   598→        </section>
   599→
   600→        {error !== null ? (
   601→          <div className="mt-6 rounded-2xl bg-[var(--bad-bg)] p-5">
   602→            <p className="text-sm text-ink">{error}</p>
   603→            <button
   604→              onClick={() => void load(false)}
   605→              className="mt-2 text-sm text-metric-blue underline underline-offset-2"
   606→            >
   607→              Try again
   608→            </button>
   609→          </div>
   610→        ) : jobs === null ? (
   611→          <div className="mt-6 space-y-3.5" aria-hidden="true">
   612→            {[0, 1, 2].map((i) => (
   613→              <div key={i} className="skel h-28" />
   614→            ))}
   615→          </div>
   616→        ) : openJobs.length === 0 ? (
   617→          <div className="dlg-card mt-6">
   618→            <EmptyState
   619→              icon={Inbox}
   620→              title="No open positions right now"
   621→              description="Check back soon or explore the full board for upcoming announcements."
   622→            />
   623→            <div className="pb-8 text-center">
   624→              <button
   625→                onClick={() => navigate("vacant-positions")}
   626→                className="inline-flex min-h-[44px] items-center gap-2 text-sm text-metric-blue underline underline-offset-2 hover:text-ink"
   627→              >
   628→                Go to the full positions board
   629→                <ArrowRight className="h-4 w-4" aria-hidden="true" />
   630→              </button>
   631→            </div>
   632→          </div>
   633→        ) : (
   634→          // Board height is derived, not a magic viewport constant: on lg the
   635→          // grid flex-fills whatever is left between the real header and
   636→          // footer (main is a min-h-0 flex column on lg), so the page itself
   637→          // never scrolls and the open posting stays pinned in view — the
   638→          // reference board's "sticky description" behavior.
   639→          <div className="mt-6 grid gap-8 sm:mt-8 lg:min-h-0 lg:flex-1 lg:grid-cols-[minmax(340px,430px)_minmax(0,1fr)]">
   640→            {/* Left · the register — 8px of ring room on every side (pl/py)
   641→                with matching negative margins so the cards, the column
   642→                alignment, and the scrollbar stay exactly where they were;
   643→                without it the overflow-y padding box clips the selected
   644→                card's outside ring on the left/top/bottom. */}
   645→            {showList && (
   646→              <div
   647→                aria-label="Job opportunities list"
   648→                className="scroll-thin min-h-0 space-y-3.5 lg:-ml-2 lg:-my-2 lg:overflow-y-auto lg:overscroll-contain lg:py-2 lg:pl-2 lg:pr-1.5"
   649→              >
   650→                {openJobs.map((job) => (
   651→                  <JobListCard
   652→                    key={job.id}
   653→                    job={job}
   654→                    selected={selectedId === job.id}
   655→                    expanded={expandedId === job.id}
   656→                    saved={saved.includes(job.id)}
   657→                    onSelect={() => select(job.id)}
   658→                    onToggle={() => setExpandedId(expandedId === job.id ? null : job.id)}
   659→                    onSave={() => saveAndToast(job)}
   660→                  />
   661→                ))}
   662→              </div>
   663→            )}
   664→
   665→            {/* Right · the posting */}
   666→            {showDetail && (
   667→              <div ref={paneRef} className="scroll-thin min-h-0 lg:overscroll-contain lg:overflow-y-auto lg:pr-2">
   668→                {selectedJob ? (
   669→                  <div className="dlg-card p-6 sm:p-8">
   670→                    <JobDetail
   671→                      job={selectedJob}
   672→                      saved={saved.includes(selectedJob.id)}
   673→                      onApply={() => requestApply(selectedJob)}
   674→                      onSave={() => saveAndToast(selectedJob)}
   675→                      onBack={() => setSelectedId(null)}
   676→                    />
   677→                  </div>
   678→                ) : (
   679→                  <div className="dlg-card">
   680→                    <EmptyState
   681→                      icon={Briefcase}
   682→                      title="Select a position"
   683→                      description="Choose a role from the list to read the full posting."
   684→                    />
   685→                  </div>
   686→                )}
   687→              </div>
   688→            )}
   689→          </div>
   690→        )}
   691→      </main>
   692→
   693→      <Footer />
   694→      {/* Shared apply dialog (confirm / MQR / done), mounted once. */}
   695→      {dialogs}
   696→    </div>
   697→  );
   698→}
