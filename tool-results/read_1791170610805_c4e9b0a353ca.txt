    75→    75→/** Client-persisted bookmark list for the frontpage explorer. */
    76→    76→function useSavedJobs() {
    77→    77→  const [saved, setSaved] = useState<number[]>(() => {
    78→    78→    if (typeof window === "undefined") return [];
    79→    79→    try {
    80→    80→      const raw = window.localStorage.getItem(SAVED_KEY);
    81→    81→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
    82→    82→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
    83→    83→    } catch {
    84→    84→      return [];
    85→    85→    }
    86→    86→  });
    87→    87→
    88→    88→  const toggle = useCallback((id: number) => {
    89→    89→    setSaved((prev) => {
    90→    90→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
    91→    91→      try {
    92→    92→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
    93→    93→      } catch {
    94→    94→        /* storage unavailable — in-session state still works */
    95→    95→      }
    96→    96→      return next;
    97→    97→    });
    98→    98→  }, []);
    99→    99→
   100→   100→  return { saved, toggle };
   101→   101→}

   282→   282→/** Plain-text posting section rendered as a disc bullet list. */
   283→   283→function TextBullets({ text }: { text: string }) {
   284→   284→  const items = text
   285→   285→    .split(/\r?\n/)
   286→   286→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   287→   287→    .filter(Boolean);
   288→   288→  if (items.length === 0) return null;
   289→   289→  return (
   290→   290→    <ul className="list-disc space-y-2 pl-5 text-base leading-relaxed text-stone">
   291→   291→      {items.map((item, i) => (
   292→   292→        <li key={i}>{item}</li>
   293→   293→      ))}
   294→   294→    </ul>
   295→   295→  );
   296→   296→}

   299→   299→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {

   305→   305→      </section>
   306→   306→    );
   307→   307→  }
   308→   308→  if (text && text.trim()) {
   309→   309→    return (
   310→   310→      <section className="space-y-3">
   311→   311→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>

   313→   313→      </section>
   314→   314→    );
   315→   315→  }
   316→   316→  return null;
   317→   317→}
