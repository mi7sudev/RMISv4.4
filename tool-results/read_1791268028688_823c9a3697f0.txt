   131→/** Legacy tone→class bridge (kept for existing importers). */
   132→export function toneClass(tone: Tone): string {
   133→  const map: Record<Tone, StatusVariant> = {

   135→    warning: "warn",
   136→    danger: "bad",
   137→    info: "info",

   144→/**
   145→ * Canonical stage/status pill. Displays the normalized status label; the
   146→ * variant comes from variantForStatus unless explicitly overridden (used for
   147→ * completion / verdict labels that are not application statuses).
   148→ */
   149→export function StatusPill({
   150→  status,
   151→  variant,
   152→  className = "",
   153→}: {
   154→  status: string | null | undefined;
   155→  variant?: StatusVariant;
   156→  className?: string;
   157→}) {
   158→  const label = getStatusMeta(status).label;
   159→  const v: StatusVariant = variant ?? variantForStatus(status);
   160→  return <span className={cn(pillClass(v), "shrink-0", className)}>{label}</span>;

   163→function verdictMeta(verdict: string): { chip: string; variant: StatusVariant; meaning: string } {
   164→  switch (verdict) {
   165→    case "ALL_MET":
   166→      return { chip: "Qualified", variant: variantForVerdict("Qualified"), meaning: "Meets the minimum requirements" };
   167→    case "PARTIAL":
   168→      return { chip: "Partial", variant: variantForVerdict("Partial"), meaning: "Partially meets the minimum requirements" };
   169→    case "NONE_MET":
   170→      return { chip: "Not qualified", variant: variantForVerdict("Not qualified"), meaning: "Does not meet the minimum requirements" };
   171→    case "NEEDS_REVIEW":
   172→      return { chip: "Verify", variant: variantForVerdict("Verify"), meaning: "Needs manual verification" };
   173→    case "NO_REQUIREMENTS":
   174→      return { chip: "No reqs", variant: variantForVerdict("No reqs"), meaning: "No published requirements" };
   175→    default:
   176→      return { chip: verdict, variant: variantForVerdict(verdict), meaning: "" };
   177→  }
   178→}

   194→/** Gradient monogram avatar (wave-3) with initials derived from the name. */
   195→function MonogramAvatar({ name, size = 36, warm = false }: { name: string; size?: number; warm?: boolean }) {
   196→  const initials = name
   197→    .split(/\s+/)
   198→    .filter(Boolean)
   199→    .slice(0, 2)
   200→    .map((w) => w[0]?.toUpperCase() ?? "")
   201→    .join("");
   202→  return (
   203→    <span aria-hidden="true">
   204→      <Monogram size={size} warm={warm} className="rounded-full">{initials || "?"}</Monogram>
   205→    </span>
   206→  );

   759→function parseAttachments(json: string | null): { name: string; bytes: number }[] {
   760→  if (!json) return [];
   761→  try {
   762→    const parsed: unknown = JSON.parse(json);
   763→    return Array.isArray(parsed) ? (parsed as { name: string; bytes: number }[]) : [];
   764→  } catch {
   765→    return [];
   766→  }

   773→/** Composer dialog: subject + content, recipient auto-retrieved from the
   774→    applicant profile (the server re-resolves it on send). Send is guarded by
   775→    a confirmation dialog. */
   776→function DirectEmailDialog({
   777→  applicationId,
   778→  to,
   779→  onClose,
   780→  onSent,
   781→}: {
   782→  applicationId: number;
   783→  to: string;
   784→  onClose: () => void;
   785→  onSent: () => void;
   786→}) {
   787→  const [subject, setSubject] = useState("");
   788→  const [message, setMessage] = useState("");
   789→  const [files, setFiles] = useState<File[]>([]);
   790→  const [busy, setBusy] = useState(false);
   791→  const [confirmOpen, setConfirmOpen] = useState(false);
   792→  const fileRef = useRef<HTMLInputElement>(null);
   793→
   794→  const pickFiles = (list: FileList | null) => {
   795→    if (!list) return;
   796→    const next = Array.from(list);
   797→    if (next.length + files.length > 3) {
   798→      toast.error("A maximum of 3 attachments is allowed");
   799→      return;
   800→    }
   801→    if (next.some((f) => f.size > 5 * 1024 * 1024)) {
   802→      toast.error("Each attachment must be 5 MB or smaller");
   803→      return;
   804→    }
   805→    setFiles((prev) => [...prev, ...next].slice(0, 3));
   806→    if (fileRef.current) fileRef.current.value = "";
   807→  };
   808→
   809→  const send = async () => {
   810→    if (!message.trim()) return;
   811→    setBusy(true);
   812→    try {
   813→      const fd = new FormData();
   814→      fd.set("subject", subject.trim());
   815→      fd.set("message", message.trim());
   816→      for (const f of files) fd.append("attachments", f);
   817→      const res = await apiFetch<{ email: { status: string; error?: string | null }; attachments: number }>(
   818→        `/api/evaluator/applications/${applicationId}/email`,
   819→        { method: "POST", formData: fd }
   820→      );
   821→      if (res.email?.status === "failed") {
   822→        toast.error("Email delivery failed", { description: res.email.error ?? undefined });
   823→      } else {
   824→        toast.success(`Direct email ${res.email?.status}`, {
   825→          description: res.attachments ? `${res.attachments} attachment(s) included.` : undefined,
   826→        });
   827→      }
   828→      setConfirmOpen(false);
   829→      onClose();
   830→      onSent();
   831→    } catch (e) {
   832→      setConfirmOpen(false);
   833→      toast.error(e instanceof Error ? e.message : "Failed to send the email");
   834→    } finally {
   835→      setBusy(false);
   836→    }
   837→  };
   838→
   839→  return (
   840→    <>
   841→      <Dialog open onOpenChange={(v) => !v && onClose()}>
   842→        <DialogContent className="max-h-[92dvh] overflow-y-auto scroll-thin sm:max-w-lg">
   843→          <DialogHeader>
   844→            <DialogTitle className="text-2xl font-semibold leading-[1.1] tracking-[-0.01em]">Send a direct email</DialogTitle>
   845→            <DialogDescription>
   846→              The recipient is retrieved automatically from the email on the applicant's profile.
   847→            </DialogDescription>
   848→          </DialogHeader>
   849→          <div className="space-y-4">
   850→            <div className="space-y-1.5">
   851→              <Label htmlFor="direct-email-to" className="text-xs text-stone">
   852→                To
   853→              </Label>
   854→              <Input id="direct-email-to" className="dlg-input" value={to} readOnly aria-label="Recipient email" />
   855→            </div>
   856→            <div className="space-y-1.5">
   857→              <Label htmlFor="direct-email-subject" className="text-xs text-stone">
   858→                Subject
   859→              </Label>
   860→              <Input
   861→                id="direct-email-subject"
   862→                className="dlg-input"
   863→                placeholder="Subject"
   864→                value={subject}
   865→                maxLength={200}
   866→                onChange={(e) => setSubject(e.target.value)}
   867→              />
   868→            </div>
   869→            <div className="space-y-1.5">
   870→              <Label htmlFor="direct-email-message" className="text-xs text-stone">
   871→                Message *
   872→              </Label>

   896→                <ul className="mt-2 space-y-1">

   898→                    <li key={f.name} className="flex items-center justify-between gap-2 text-xs text-stone">
