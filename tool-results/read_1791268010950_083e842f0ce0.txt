   133→   133→// ── VerdictPill ─────────────────────────────────────────────────────────────
   134→   134→
   135→   135→function verdictMeta(verdict: string): { chip: string; variant: StatusVariant } {
   136→   136→  switch (verdict) {
   137→   137→    case "ALL_MET":
   138→   138→      return { chip: "Qualified", variant: variantForVerdict("Qualified") };
   139→   139→    case "PARTIAL":
   140→   140→      return { chip: "Partial", variant: variantForVerdict("Partial") };
   141→   141→    case "NONE_MET":
   142→   142→      return { chip: "Not qualified", variant: variantForVerdict("Not qualified") };
   143→   143→    case "NEEDS_REVIEW":
   144→   144→      return { chip: "Verify", variant: variantForVerdict("Verify") };
   145→   145→    case "NO_REQUIREMENTS":
   146→   146→      return { chip: "No reqs", variant: variantForVerdict("No reqs") };
   147→   147→    default:
   148→   148→      return { chip: verdict, variant: variantForVerdict(verdict) };
   149→   149→  }
   150→   150→}

   167→   167→/** First letters of the first two words, uppercase ("Juan Cruz" → "JC"). */
   168→   168→export function nameInitials(name: string | null | undefined): string {
   169→   169→  return (
   170→   170→    (name || "")
   171→   171→      .split(/\s+/)
   172→   172→      .filter(Boolean)
   173→   173→      .slice(0, 2)
   174→   174→      .map((w) => w[0]?.toUpperCase() ?? "")
   175→   175→      .join("") || "?"
   176→   176→  );
   177→   177→}

   573→   573→export function SectionCard({
   574→   574→  title,
   575→   575→  description,
   576→   576→  actions,

   578→   578→  chipTone,
   579→   579→  children,
   580→   580→  className,
   581→   581→  bodyClassName,
   582→   582→  /** "cloud" = pale Cloud Card (default) · "plain" = white panel for content
   583→   583→   *  sitting inside dialogs / on cloud surfaces (profile section anatomy). */
   584→   584→  variant = "cloud",
   585→   585→}: {
   586→   586→  title?: string;

   588→   588→  actions?: React.ReactNode;

   595→   595→}) {

   619→   619→              <h2 className="font-display truncate text-lg leading-7 text-ink">{title}</h2>
   620→   620→              {/* Wraps below sm, truncates from sm up. Full-width nowrap (truncate
   621→   621→                  everywhere) gave the card a huge min-content floor that blew out
   622→   622→                  auto grid tracks on phones; wrapping text has none. */}

   630→   630→    </section>
   631→   631→  );
   632→   632→}
