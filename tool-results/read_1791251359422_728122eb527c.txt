   127→/** 3-checkpoint tracking timeline: Submitted → Review → Decision (§7.3). */
   128→function JourneyTimeline({ status }: { status: string }) {
   129→  const stage = stageForStatus(status);
   130→  const decided = stage === "Shortlisted" || stage === "Rejected";
   131→
   132→  const steps: { label: string; short?: string; state: CheckpointState }[] = [
   133→    { label: "Submitted", state: "done" },
   134→    {
   135→      label: "Review",
   136→      state: stage === "Under Review" ? "current" : decided ? "done" : "pending",
   137→    },
   138→    {
   139→      label:
   140→        stage === "Shortlisted" ? "Shortlisted" : stage === "Rejected" ? "Not Shortlisted" : "Awaiting the shortlist decision",
   141→      // Responsive copy swap (3-d, presentation-only): the full label wraps to 4+ lines
   142→      // inside a 3-up timeline at 320px; "Shortlisted"/"Not Shortlisted" stay verbatim.
   143→      short: stage === "Shortlisted" || stage === "Rejected" ? undefined : "Awaiting decision",
   144→      state: stage === "Shortlisted" ? "done" : stage === "Rejected" ? "failed" : "pending",
   145→    },
   146→  ];
   147→
   148→  return (
   149→    <div className="mt-3 grid grid-cols-3 gap-1">
   150→      {steps.map((s, i) => {
   151→        const reached = s.state !== "pending";
   152→        return (
   153→          <div key={`${s.label}-${i}`} className="flex flex-col items-center gap-1.5 text-center">
   154→            <div className="flex w-full items-center">
   155→              {/* Traversed connectors render in solid brand blue. Negative
   156→                  outer margins bridge the grid's gap-1 so each rail reads
   157→                  as one continuous line, not segments with center gaps. */}
   158→              <div
   159→                className={cn(
   160→                  "h-px flex-1",
   161→                  i > 0 && "-ml-1",
   162→                  i === 0 ? "bg-transparent" : reached ? "bg-primary" : "bg-divider"
   163→                )}
   164→              />
   165→              <CheckpointDot state={s.state} />
   166→              <div
   167→                className={cn(
   168→                  "h-px flex-1",
   169→                  i < steps.length - 1 && "-mr-1",
   170→                  i === steps.length - 1
   171→                    ? "bg-transparent"
   172→                    : steps[i + 1].state !== "pending"
   173→                      ? "bg-primary"
   174→                      : "bg-divider"
   175→                )}
   176→              />
   177→            </div>
   178→            <span className={cn("text-[10px] leading-tight", s.state === "current" ? "font-medium text-ink" : "text-stone")}>
   179→              {s.short ? (
   180→                <>
   181→                  <span className="sm:hidden">{s.short}</span>
   182→                  <span className="hidden sm:inline">{s.label}</span>
   183→                </>
   184→              ) : (
   185→                s.label
   186→              )}
   187→            </span>
   188→          </div>
   189→        );
   190→      })}
   191→    </div>
   192→  );
   193→}

   200→// Shared with the frontpage explorer — bookmarks saved here show as saved
   201→// there and vice versa.
   202→const SAVED_KEY = "rmis-frontpage-saved-jobs";
   203→
   204→/** Client-persisted bookmark list (same store as the frontpage). */
   205→function useSavedJobs() {
   206→  const [saved, setSaved] = useState<number[]>(() => {
   207→    if (typeof window === "undefined") return [];
   208→    try {
   209→      const raw = window.localStorage.getItem(SAVED_KEY);
   210→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
   211→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
   212→    } catch {
   213→      return [];
   214→    }
   215→  });
   216→
   217→  const toggle = useCallback((id: number) => {
   218→    setSaved((prev) => {
   219→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
   220→      try {
   221→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
   222→      } catch {
   223→        /* storage unavailable — in-session state still works */
   224→      }
   225→      return next;
   226→    });
   227→  }, []);
   228→
   229→  return { saved, toggle };
   230→}

   332→function ApplicationDetailModal({
   333→  application,
   334→  onClose,
   335→  onCancelled,
   336→}: {
   337→  application: ApplicationWire;
   338→  onClose: () => void;
   339→  onCancelled: () => void;
   340→}) {
   341→  const [confirmOpen, setConfirmOpen] = useState(false);
   342→  const [cancelling, setCancelling] = useState(false);
   343→  const job = application.job ?? null;
   344→  const position = job?.position ?? null;
   345→  const label = currentStageLabel(application.status);
   346→  const cancellable = label === "Submitted";
   347→  const dl = deadlineState(job?.deadlineDate ?? null);
   348→
   349→  const cancelApplication = async () => {
   350→    setCancelling(true);
   351→    try {
   352→      await apiFetch(`/api/applications/${application.id}`, { method: "DELETE" });
   353→      toast.success("Application cancelled", { description: "Your application has been withdrawn." });
   354→      setConfirmOpen(false);
   355→      onClose();
   356→      onCancelled();
   357→    } catch (e) {
   358→      // Progressed applications: surface the server's guidance (e.g. "contact HR") as-is.
   359→      toast.error(e instanceof Error ? e.message : "Unable to cancel this application");
   360→      setConfirmOpen(false);
   361→    } finally {
   362→      setCancelling(false);
   363→    }
   364→  };
   365→
   366→  const facts: [string, string][] = [
   367→    ["Place of Assignment", position?.placeOfAssignment ?? "-"],
   368→    ["Employment Type", job?.positionType ? humanize(job.positionType) : position?.positionType ? humanize(position.positionType) : "-"],
   369→    ["Vacancies", job?.numberOfVacancy == null ? "-" : String(job.numberOfVacancy)],
   370→    ["Monthly Salary", formatCurrency(position?.salaryAmount ?? null)],
   371→    ["Salary Grade", position?.salaryGrade ? `SG-${position.salaryGrade}${position.salaryStep ? ` Step ${position.salaryStep}` : ""}` : "-"],
   372→    ["Item No.", position?.itemNumber ?? "-"],
   373→    ["Published", formatDate(job?.publishDate ?? null)],
   374→    ["Deadline", job ? dl.label : "-"],
   375→  ];
   376→
   377→  return (
   378→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   379→      <DialogContent className="dlg-card-plain max-h-[90dvh] w-full overflow-y-auto scroll-thin p-0 sm:max-w-3xl">
   380→        <div className="p-6">
   381→          <DialogHeader className="space-y-2 text-left">
   382→            <div className="flex flex-wrap items-center gap-2">
   383→              <StatusPill status={label} variant={stageVariant(label)} />
   384→              <span className="dlg-pill num rounded-full bg-fog px-3 py-1 text-xs font-medium text-stone">
   385→                Application #{application.id}
   386→              </span>
   387→            </div>
   388→            <DialogTitle className="font-display text-2xl leading-tight tracking-[-0.02em] text-ink">
   389→              {job?.title ?? "Application"}
   390→            </DialogTitle>
   391→          </DialogHeader>
   392→
   393→          {/* Successfully-applied strip (§7.7) */}
   394→          <div className="mt-4 flex items-center gap-3 rounded-lg bg-cloud/60 p-3">
   395→            <IconChip icon={BadgeCheck} tone="emerald" size={36} iconSize={17} />
   396→            <div>
   397→              <p className="text-sm font-medium text-ink">Successfully Applied</p>
   398→              <p className="text-xs text-stone">
   399→                Applied <span className="num">{formatDate(application.dateApplied)}</span>
   400→              </p>
   401→            </div>
   402→          </div>
   403→
   404→          {/* Vitals summary grid */}
   405→          <div className="mt-5 grid grid-cols-2 gap-2 sm:grid-cols-4">
   406→            {facts.map(([k, v]) => (
   407→              <div key={k} className="rounded-lg bg-cloud/60 p-3">
   408→                <p className="text-[11px] text-stone">{k}</p>
   409→                <p className="num mt-0.5 truncate text-sm text-ink" title={v}>
   410→                  {v}
   411→                </p>
   412→              </div>
   413→            ))}
   414→          </div>
   415→
   416→          <div className="mt-6 space-y-6">
   417→            <MqrLedger position={position} />
   418→            <RichSection title="Brief Description" text={job?.briefDescription} html={job?.briefDescriptionHtml} />
   419→            <RichSection title="Duties & Responsibilities" text={job?.dutiesResponsibilities} html={job?.dutiesHtml} />
   420→            <RichSection title="Compensation Package" text={job?.compensationPackage} html={job?.compensationHtml} />

   422→          </div>

   424→          <div className="mt-8 flex flex-col gap-3 border-t border-border pt-5 sm:flex-row sm:justify-end">
   425→            {cancellable && (
   426→              <button
   427→                type="button"
   428→                onClick={() => setConfirmOpen(true)}
   429→                disabled={cancelling}
   430→                className="inline-flex min-h-[44px] items-center justify-center rounded-full border border-[var(--bad)]/30 bg-white px-6 py-2.5 text-sm font-medium text-[var(--bad)] transition-colors hover:bg-[var(--bad-bg)] disabled:opacity-50"
   431→              >
   432→                Cancel Application
   433→              </button>

   435→            <button type="button" onClick={onClose} className="dlg-ghost min-h-[44px] px-6 py-2.5 text-sm">

   438→          </div>

   446→              <AlertDialogDescription className="text-sm leading-relaxed text-body-gray">
   447→                Your application for {job?.title ?? "this position"} will be withdrawn. You may apply again while the
   448→                position is still open. Once review has started, applications can no longer be cancelled.
   449→              </AlertDialogDescription>
   450→            </AlertDialogHeader>
   451→            <AlertDialogFooter>
   452→              <AlertDialogCancel className="dlg-ghost min-h-[44px] px-5 py-2.5 text-sm">Keep it</AlertDialogCancel>

   454→                onClick={(e) => {
   455→                  e.preventDefault();

   458→                disabled={cancelling}
   459→                className="inline-flex min-h-[44px] items-center justify-center rounded-full border border-[var(--bad)]/30 bg-white px-5 py-2.5 text-sm font-medium text-[var(--bad)] transition-colors hover:bg-[var(--bad-bg)] disabled:opacity-50"
   460→              >
   461→                {cancelling ? "Cancelling…" : "Cancel Application"}
   462→              </AlertDialogAction>
   463→            </AlertDialogFooter>
   464→          </AlertDialogContent>
   465→        </AlertDialog>
   466→      </DialogContent>
   467→    </Dialog>
   468→  );
   469→}
