    58→// QA 2026-02: Position Category buckets. In the CSC vocabulary "Permanent" and
    59→// "Temporary" are plantilla items (1:1 with an item number), while "Contract
    60→// of Service" and "Job Order" are non-plantilla (COS) engagements.
    61→function matchesCategory(positionType: string | null, category: string): boolean {
    62→  const t = (positionType ?? "").toUpperCase();
    63→  const isCos = t.includes("COS") || t.includes("CONTRACT") || t.includes("JOB ORDER");
    64→  const isPlantilla = t.includes("PLANTILLA") || t.includes("PERMANENT") || t.includes("TEMPORARY");
    65→  if (category === "COS") return isCos;
    66→  return isPlantilla || (!isCos && t.length > 0); // anything else non-COS (e.g. Casual) is plantilla-graded
    67→}

    85→function pageWindow(current: number, total: number): (number | "…")[] {
    86→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1);
    87→  const out: (number | "…")[] = [1];
    88→  const lo = Math.max(2, current - 1);
    89→  const hi = Math.min(total - 1, current + 1);
    90→  if (lo > 2) out.push("…");
    91→  for (let i = lo; i <= hi; i++) out.push(i);
    92→  if (hi < total - 1) out.push("…");
    93→  out.push(total);
    94→  return out;
    95→}

    97→/** % of the application window elapsed between publish and deadline (0–100). */
    98→function deadlineProgress(job: JobWire): number {
    99→  const start = new Date(job.publishDate ?? job.publishedAt).getTime();
   100→  const end = job.deadlineDate ? new Date(job.deadlineDate).getTime() : Number.NaN;
   101→  if (Number.isNaN(start) || Number.isNaN(end) || end <= start) return job.deadlineDate ? 100 : 0;
   102→  const pct = ((Date.now() - start) / (end - start)) * 100;
   103→  return Math.min(100, Math.max(0, pct));
   104→}

   156→function QuickView({
   157→  job,
   158→  onApply,
   159→  onDetail,
   160→}: {
   161→  job: JobWire;
   162→  onApply: (job: JobWire) => void;
   163→  onDetail: (job: JobWire) => void;
   164→}) {
   165→  const pos = job.position;
   166→  const dl = deadlineState(job.deadlineDate);
   167→  const brief = job.briefDescription?.trim()
   168→    ? job.briefDescription
   169→    : job.briefDescriptionHtml
   170→      ? stripHtml(job.briefDescriptionHtml)
   171→      : "";
   172→  const facts: [string, string][] = [
   173→    ["Item No.", pos?.itemNumber || "-"],
   174→    ["Vacancies", job.numberOfVacancy == null ? "-" : String(job.numberOfVacancy)],
   175→    ["Salary Grade", sgLabel(pos?.salaryGrade, pos?.salaryStep)],
   176→    ["Monthly Salary", pos?.salaryAmount != null ? formatCurrency(pos.salaryAmount) : "-"],
   177→    ["Published", formatDate(job.publishDate ?? job.publishedAt)],
   178→    ["Deadline", formatDate(job.deadlineDate)],
   179→  ];
   180→  return (
   181→    <div className="mt-4 space-y-4 border-t border-divider pt-4">
   182→      {brief ? (
   183→        <p className="line-clamp-4 text-sm leading-relaxed text-stone">{brief}</p>
   184→      ) : (
   185→        <p className="text-sm text-stone">No brief description provided.</p>
   186→      )}
   187→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   188→        {facts.map(([label, value]) => (
   189→          <div key={label} className="rounded-xl bg-white p-3">
   190→            <p className="text-xs text-stone">{label}</p>
   191→            <p className="mt-0.5 truncate text-sm text-ink" title={value}>
   192→              {value}
   193→            </p>
   194→          </div>
   195→        ))}
   196→      </div>
   197→      <div className="flex flex-wrap gap-3">
   198→        <button
   199→          onClick={() => onApply(job)}
   200→          disabled={dl.overdue}
   201→          className="dlg-cta min-h-[44px] px-6 text-sm disabled:pointer-events-none disabled:opacity-50"
   202→        >
   203→          Apply now
   204→        </button>
   205→        <button onClick={() => onDetail(job)} className="dlg-ghost min-h-[44px] px-6 text-sm">
   206→          Read full description
   207→        </button>

   209→    </div>
