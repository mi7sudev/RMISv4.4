   127→   127→export function JobFormDialog({
   128→   128→  open,
   129→   129→  onOpenChange,
   130→   130→  job,
   131→   131→  onSaved,
   132→   132→}: {
   133→   133→  open: boolean;
   134→   134→  onOpenChange: (v: boolean) => void;
   135→   135→  job?: JobWire | null;
   136→   136→  onSaved: () => void;
   137→   137→}) {
   138→   138→  const editing = job != null;
   139→   139→  const [title, setTitle] = useState("");
   140→   140→  const [positionType, setPositionType] = useState("");
   141→   141→  const [vacancy, setVacancy] = useState("1");
   142→   142→  const [division, setDivision] = useState("");
   143→   143→  const [education, setEducation] = useState("");
   144→   144→  const [experience, setExperience] = useState("");
   145→   145→  const [training, setTraining] = useState("");
   146→   146→  const [eligibility, setEligibility] = useState("");
   147→   147→  const [license, setLicense] = useState("");
   148→   148→  const [brief, setBrief] = useState("");
   149→   149→  const [duties, setDuties] = useState("");
   150→   150→  const [compensation, setCompensation] = useState("");
   151→   151→  const [otherQuals, setOtherQuals] = useState("");
   152→   152→  const [publishDate, setPublishDate] = useState("");
   153→   153→  const [deadlineDate, setDeadlineDate] = useState("");
   154→   154→  const [processingDate, setProcessingDate] = useState("");
   155→   155→  const [errors, setErrors] = useState<{ title?: string | null; vacancy?: string | null }>({});
   156→   156→  const [busy, setBusy] = useState(false);
   157→   157→
   158→   158→  const prefill = useCallback(() => {
   159→   159→    setErrors({});
   160→   160→    if (!job) {
   161→   161→      setTitle("");
   162→   162→      setPositionType("");
   163→   163→      setVacancy("1");
   164→   164→      setDivision("");
   165→   165→      setEducation("");
   166→   166→      setExperience("");
   167→   167→      setTraining("");
   168→   168→      setEligibility("");
   169→   169→      setLicense("");
   170→   170→      setBrief("");
   171→   171→      setDuties("");
   172→   172→      setCompensation("");
   173→   173→      setOtherQuals("");
   174→   174→      setPublishDate(new Date().toISOString().slice(0, 10));
   175→   175→      setDeadlineDate("");
   176→   176→      setProcessingDate("");
   177→   177→      return;
   178→   178→    }
   179→   179→    setTitle(job.title ?? "");
   180→   180→    setPositionType(job.positionType ?? "");
   181→   181→    setVacancy(job.numberOfVacancy == null ? "" : String(job.numberOfVacancy));
   182→   182→    setDivision(job.position?.division ?? "");
   183→   183→    setEducation(job.position?.cscEducation ?? "");
   184→   184→    setExperience(job.position?.cscWorkExperience ?? "");
   185→   185→    setTraining(job.position?.cscTraining ?? "");
   186→   186→    setEligibility(job.position?.cscEligibilityGroup ?? job.position?.cscEligibility ?? "");
   187→   187→    setLicense(job.position?.license ?? "");
   188→   188→    setBrief(job.briefDescription ?? "");
   189→   189→    setDuties(job.dutiesResponsibilities ?? "");
   190→   190→    setCompensation(job.compensationPackage ?? "");
   191→   191→    setOtherQuals(job.otherQualifications ?? "");
   192→   192→    setPublishDate(job.publishDate ? job.publishDate.slice(0, 10) : "");
   193→   193→    setDeadlineDate(job.deadlineDate ? job.deadlineDate.slice(0, 10) : "");
   194→   194→    setProcessingDate(job.processingDate ? job.processingDate.slice(0, 10) : "");
   195→   195→  }, [job]);
   196→   196→
   197→   197→  useEffect(() => {
   198→   198→    if (open) prefill();
   199→   199→  }, [open, prefill]);
   200→   200→
   201→   201→  const opt = (v: string) => {
   202→   202→    const t = v.trim();
   203→   203→    return t === "" ? undefined : t;
   204→   204→  };
   205→   205→
   206→   206→  const submit = async () => {
   207→   207→    const nextErrors: { title?: string | null; vacancy?: string | null } = {};
   208→   208→    if (!title.trim()) nextErrors.title = "Title is required";
   209→   209→    // QA 2026-02: vacancy count is required for COS postings only —
   210→   210→    // Plantilla positions map 1:1 to a Plantilla Item Number.
   211→   211→    const typeUpper = (positionType ?? "").toUpperCase();
   212→   212→    const isCos = typeUpper.includes("COS") || typeUpper.includes("CONTRACT");
   213→   213→    let vacancyNum: number | null = null;
   214→   214→    if (vacancy.trim() !== "") {
   215→   215→      const n = Number(vacancy);
   216→   216→      if (!Number.isInteger(n) || n < 1) {
   217→   217→        nextErrors.vacancy = "Vacancies must be a whole number of at least 1";
   218→   218→      } else {
   219→   219→        vacancyNum = n;
   220→   220→      }
   221→   221→    } else if (isCos) {
   222→   222→      nextErrors.vacancy = "COS postings must specify the number of vacancies";
   223→   223→    }
   224→   224→    setErrors(nextErrors);
   225→   225→    if (Object.keys(nextErrors).length > 0) return;
   226→   226→
   227→   227→    setBusy(true);
   228→   228→    try {
   229→   229→      const body = {
   230→   230→        title: title.trim(),
   231→   231→        positionType: opt(positionType),
   232→   232→        numberOfVacancy: vacancyNum,
   233→   233→        division: opt(division),
   234→   234→        education: opt(education),
   235→   235→        experience: opt(experience),
   236→   236→        training: opt(training),
   237→   237→        eligibility: opt(eligibility),
   238→   238→        license: opt(license),
   239→   239→        briefDescription: opt(brief),
   240→   240→        briefDescriptionHtml: textToHtml(brief) || undefined,
   241→   241→        dutiesResponsibilities: opt(duties),
   242→   242→        dutiesHtml: textToHtml(duties) || undefined,
   243→   243→        compensationPackage: opt(compensation),
   244→   244→        compensationHtml: textToHtml(compensation) || undefined,
   245→   245→        otherQualifications: opt(otherQuals),
   246→   246→        otherQualificationsHtml: textToHtml(otherQuals) || undefined,
   247→   247→        publishDate: publishDate || undefined,
   248→   248→        deadlineDate: deadlineDate || undefined,
   249→   249→        processingDate: processingDate || undefined,
   250→   250→      };
   251→   251→      if (editing) {
   252→   252→        await apiFetch(`/api/jobs/${job.id}`, { method: "PATCH", body });
   253→   253→        toast.success("Job posting updated");
   254→   254→      } else {
   255→   255→        await apiFetch("/api/jobs", { method: "POST", body });
   256→   256→        toast.success("Job posting created and published");
   257→   257→      }
   258→   258→      onOpenChange(false);
   259→   259→      onSaved();
   260→   260→    } catch (e) {
   261→   261→      toast.error(e instanceof Error ? e.message : "Failed to save the job posting");
   262→   262→    } finally {
   263→   263→      setBusy(false);
   264→   264→    }
   265→   265→  };
   266→   266→
   267→   267→  return (
   268→   268→    <Dialog open={open} onOpenChange={onOpenChange}>
   269→   269→      <DialogContent className="sm:max-w-2xl max-h-[92dvh] overflow-y-auto scroll-thin">
   270→   270→        <DialogHeader>
   271→   271→          <DialogTitle className="text-2xl font-semibold leading-[1.1] tracking-[-0.01em]">{editing ? "Edit job posting" : "Create job posting"}</DialogTitle>
   272→   272→          <DialogDescription>
   273→   273→            {editing
   274→   274→              ? "Updates are saved to the posting; qualification vitals write through to the position master."
   275→   275→              : "The posting is published immediately; qualification vitals create or update the linked position."}
   276→   276→          </DialogDescription>
   277→   277→        </DialogHeader>
   278→   278→
   279→   279→        <div className="space-y-4">
   280→   280→          <FormPanel title="Position Details" icon={Briefcase} chipTone="slate">
   281→   281→            <div className="space-y-4">
   282→   282→              <Field id="job-title" label="Title" required error={errors.title}>
   283→   283→                <Input
   284→   284→                  id="job-title"
   285→   285→                  className="dlg-input"
   286→   286→                  value={title}
   287→   287→                  maxLength={200}
   288→   288→                  onChange={(e) => setTitle(e.target.value)}
   289→   289→                  placeholder="e.g. Science Research Specialist I"
   290→   290→                />
   291→   291→              </Field>
   292→   292→
   293→   293→              <Field id="job-type" label="Position type">
   294→   294→                <Input
   295→   295→                  id="job-type"
   296→   296→                  className="dlg-input"
   297→   297→                  list="rmis-position-types"
   298→   298→                  value={positionType}
   299→   299→                  maxLength={100}
   300→   300→                  onChange={(e) => setPositionType(e.target.value)}
   301→   301→                  placeholder="Select or type"
   302→   302→                />
   303→   303→                <datalist id="rmis-position-types">
   304→   304→                  {POSITION_TYPES.map((t) => (
   305→   305→                    <option key={t} value={t} />
   306→   306→                  ))}
   307→   307→                </datalist>
   308→   308→              </Field>
   309→   309→
   310→   310→              <Field id="job-division" label="Division">
   311→   311→                <Input

   313→   313→                  className="dlg-input"

   315→   315→                  value={division}
