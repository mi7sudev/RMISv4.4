    38→    38→// ── Deterministic PDS parsing (CS Form 212 — version-adaptive) ──────────────
    39→    39→
    40→    40→// Roman-numeral section titles ("IV. CIVIL SERVICE ELIGIBILITY", "VII.
    41→    41→// LEARNING AND DEVELOPMENT …") must never anchor a table — the real table
    42→    42→// header sits 1–2 rows below them. "(Continue on separate sheet …)" notes,
    43→    43→// signature rows, and template disclaimer banners are never data.
    44→    44→const NOISE = /cs\s*form|page\s*\d+\s*of|signature|do not abbreviate|^\s*[ivx]+\.\s*$|^\s*[ivxlc]+\s*\.\s|^section\b|continue\s*on\s*(a\s+)?(the\s+)?(separate\s+)?sheet|continue\s*on\s*separate|administering|e-?signature|over-\s*mark|demonstration\s*data|do\s*not\s*submit/i;
    45→    45→
    46→    46→// Template sub-captions printed inside value areas ("House/Block/Lot No.",
    47→    47→// "NAME EXTENSION (JR., SR)", "Pls. indicate country:") — never applicant
    48→    48→// data, so label scanners must hop over them.
    49→    49→const CAPTION_RE = /name\s*extension|house\/?block|^street$|subdivision\/?village|city\/?municipality|^barangay$|^province$|^zip\s*code|if\s*holder|pls\.?\s*indicate|dual\s*citizenship|by\s*birth|by\s*naturalization|inclusive\s*dates|write\s*in\s*full|continue\s*on|do\s*not\s*abbreviate/i;
    50→    50→
    51→    51→function cellText(v: unknown): string {
    52→    52→  if (v == null) return "";
    53→    53→  if (typeof v === "number") return String(v);
    54→    54→  if (typeof v === "boolean") return v ? "Yes" : "";
    55→    55→  if (typeof v === "object" && v !== null) {
    56→    56→    const obj = v as { text?: string; result?: unknown };
    57→    57→    if (typeof obj.text === "string") return obj.text;
    58→    58→    if (obj.result != null) return String(obj.result);
    59→    59→    return "";
    60→    60→  }
    61→    61→  return String(v).replace(/\s+/g, " ").trim();
    62→    62→}

    64→    64→// Percent-formatted numeric cells (CSC exam ratings are stored as 0.8853 with
    65→    65→// a "0.00%" display format) must surface as "88.53%", not a raw fraction.
    66→    66→function cellDisplay(ws: XLSX.WorkSheet, r: number, c: number): string {
    67→    67→  const cell = ws[XLSX.utils.encode_cell({ r, c })] as XLSX.CellObject | undefined;
    68→    68→  if (cell && typeof cell.v === "number" && typeof cell.z === "string" && cell.z.includes("%") && Math.abs(cell.v) <= 1.5) {
    69→    69→    return `${Math.round(cell.v * 10000) / 100}%`;
    70→    70→  }
    71→    71→  return cellText(cell?.v);
    72→    72→}

    74→    74→// PDS versions differ in how much boilerplate they embed inside label cells
    75→    75→// ("POSITION TITLE (Write in full/Do not abbreviate)", "(mm/dd/yyyy)",
    76→    76→// "(Continue on separate sheet if necessary)"). Probing the text with
    77→    77→// parentheticals stripped keeps real headers recognizable while page footers
    78→    78→// ("CS FORM 212 (Revised 2017), Page 1 of 4") are still rejected by NOISE.
    79→    79→function probeText(v: unknown): string {
    80→    80→  return cellText(v).replace(/\([^)]*\)/g, " ").replace(/\s+/g, " ").trim();
    81→    81→}

    83→    83→// "N/A" markers mean the applicant left the cell blank — treat as empty.
    84→    84→const isNA = (s: string) => !s || /^n\/?a\.?$/i.test(s.trim());
    85→    85→
    86→    86→// Real PDS cell values are short; anything beyond this is template
    87→    87→// boilerplate (disclaimer banners, instructions) that some versions embed in

    89→    89→const MAX_VALUE_LEN = 160;
    90→    90→const plausible = (v: string, min: number) => !isNA(v) && v.length >= min && v.length <= MAX_VALUE_LEN;

    92→    92→function findRow(ws: XLSX.WorkSheet, labelRe: RegExp, maxRow = 80, startRow?: number): { row: number; col: number } | null {
    93→    93→  const ref = ws["!ref"];
    94→    94→  if (!ref) return null;
    95→    95→  const range = XLSX.utils.decode_range(ref);
    96→    96→  const rEnd = Math.min(range.e.r, maxRow);
    97→    97→  for (let r = startRow ?? range.s.r; r <= rEnd; r++) {
    98→    98→    for (let c = range.s.c; c <= Math.min(range.e.c, range.s.c + 14); c++) {
    99→    99→      const raw = cellText(ws[XLSX.utils.encode_cell({ r, c })]?.v);
   100→   100→      if (!raw) continue;
   101→   101→      // Section banners ("VI. LEARNING AND DEVELOPMENT …", "WORK EXPERIENCE
   102→   102→      // (Continuation)") must never anchor a table — only column headers may.
   103→   103→      if (/^\s*[ivxlc]+\s*\.\s/i.test(raw) || /\(\s*continuation\s*\)\s*$/i.test(raw)) continue;
   104→   104→      const t = probeText(raw);
   105→   105→      if (t && labelRe.test(t) && !NOISE.test(t)) return { row: r, col: c };
   106→   106→    }
   107→   107→  }
   108→   108→  return null;
   109→   109→}
   110→   110→
   111→   111→// Agency templates hide dropdown option lists ("Single/Married/Separated",
   112→   112→// country names) in helper columns BESIDE the form. A label's value always
   113→   113→// sits within a cell or two, so the scan stops after a run of empty cells —
   114→   114→// it must never walk into those helper columns across a blank value area.
   115→   115→const SCAN_GAP = 3;
   116→   116→function scanRight(ws: XLSX.WorkSheet, row: number, startCol: number, maxScan = 12): string {
   117→   117→  const ref = ws["!ref"];
   118→   118→  if (!ref) return "";
   119→   119→  const range = XLSX.utils.decode_range(ref);
   120→   120→  let gap = 0;
   121→   121→  for (let c = startCol + 1; c <= Math.min(range.e.c, startCol + maxScan); c++) {
   122→   122→    const t = cellText(ws[XLSX.utils.encode_cell({ r: row, c })]?.v);
   123→   123→    if (!t || CAPTION_RE.test(t) || NOISE.test(t) || /^(yes|no)$/i.test(t) || /^\d{1,2}$/.test(t)) {
   124→   124→      if (++gap >= SCAN_GAP) break;
   125→   125→      continue;
   126→   126→    }
   127→   127→    return t;
   128→   128→  }
   129→   129→  return "";
   130→   130→}

   132→   132→// PDS address blocks print underline captions ("House/Block/Lot No.",
   133→   133→// "Subdivision/Village", …) directly BENEATH their value cell.
   134→   134→function scanAbove(ws: XLSX.WorkSheet, row: number, col: number, maxUp = 2): string {
   135→   135→  for (let r = row - 1; r >= Math.max(0, row - maxUp); r--) {
   136→   136→    const t = cellText(ws[XLSX.utils.encode_cell({ r, c: col })]?.v);
   137→   137→    if (!t) continue;
   138→   138→    if (CAPTION_RE.test(t) || NOISE.test(t)) return "";
   139→   139→    return t;
   140→   140→  }
   141→   141→  return "";
   142→   142→}

   144→   144→// The 2026 edition prints dates as dd/mm/yyyy (2017: mm/dd/yyyy). The flag is
   145→   145→// set per parse from the detected revision; ambiguous d/m strings resolve
   146→   146→// through it, unambiguous ones self-detect. Safe across awaits: the parse
   147→   147→// pipeline up to normDate use is fully synchronous.
   148→   148→let pdsDayFirst = false;
   149→   149→function normDate(v: string): string {
   150→   150→  if (!v) return "";
   151→   151→  // Excel serial?
   152→   152→  const num = Number(v);
   153→   153→  if (!Number.isNaN(num) && num > 20000 && num < 80000) {
   154→   154→    const d = new Date(Date.UTC(1899, 11, 30) as unknown as number);
   155→   155→    d.setUTCDate(d.getUTCDate() + Math.floor(num));
   156→   156→    return d.toISOString().slice(0, 10);
   157→   157→  }
   158→   158→  const m = v.match(/^(\d{1,2})[/\-.](\d{1,2})[/\-.](\d{4})$/);
   159→   159→  if (m) {
   160→   160→    const a = Number(m[1]);
   161→   161→    const b = Number(m[2]);
   162→   162→    let dayFirst = pdsDayFirst;
   163→   163→    if (a > 12) dayFirst = true;
   164→   164→    else if (b > 12) dayFirst = false;
   165→   165→    const mo = dayFirst ? b : a;
   166→   166→    const day = dayFirst ? a : b;
   167→   167→    return `${m[3]}-${String(mo).padStart(2, "0")}-${String(day).padStart(2, "0")}`;
   168→   168→  }
   169→   169→  const parsed = new Date(v);
   170→   170→  if (!Number.isNaN(parsed.getTime()) && /\d{4}/.test(v)) return parsed.toISOString().slice(0, 10);
   171→   171→  return "";
   172→   172→}

   174→   174→const year4 = (v: string): string => (/^\d{4}$/.test(v.trim()) ? v.trim() : (v.match(/\d{4}/)?.[0] ?? ""));

   176→   176→// ── Version-adaptive table reader ────────────────────────────────────────────
   177→   177→// Every PDS section table (education, eligibility, work, training, awards) is
   178→   178→// located by its HEADER ROW and each field's column is RESOLVED from the
   179→   179→// header labels (header row + up to 3 rows below, covering two-row headers
   180→   180→// with From/To sub-rows). Hardcoded column geometry from one template becomes
   181→   181→// only a fallback, so re-revisions and older/lower layouts that shifted
   182→   182→// columns still parse. Tables are collected from EVERY sheet (main pages and
   183→   183→// continuation sheets alike; Lookup sheets excluded).
   184→   184→
   185→   185→type TableRowGetter = (col: number) => string;
   186→   186→
   187→   187→type TableSpec = {
   188→   188→  key: "educations" | "workExperiences" | "trainings" | "eligibilities" | "awards";
   189→   189→  headerRe: RegExp;
   190→   190→  fallbackHeaderRe?: RegExp;
   191→   191→  fields: Record<string, { re?: RegExp; fallback: number[] }>;
   192→   192→  maxRows: number;
   193→   193→  build: (get: TableRowGetter, cols: Record<string, number>) => ExtractionSection | null;
   194→   194→};
   195→   195→
   196→   196→function resolveColumns(ws: XLSX.WorkSheet, headerRow: number, fields: TableSpec["fields"]): Record<string, number> {
   197→   197→  const ref = ws["!ref"] ? XLSX.utils.decode_range(ws["!ref"]) : null;
   198→   198→  const cStart = ref ? ref.s.c : 0;
   199→   199→  const cEnd = ref ? Math.min(ref.e.c, ref.s.c + 14) : 14;
   200→   200→  const cols: Record<string, number> = {};
   201→   201→  for (const [key, spec] of Object.entries(fields)) {
   202→   202→    cols[key] = -1;
   203→   203→    if (spec.re) {
   204→   204→      for (let r = headerRow; r <= headerRow + 3 && cols[key] === -1; r++) {
   205→   205→        for (let c = cStart; c <= cEnd; c++) {
   206→   206→          const t = probeText(ws[XLSX.utils.encode_cell({ r, c })]?.v);
   207→   207→          if (t && spec.re.test(t) && !NOISE.test(t)) {
   208→   208→            cols[key] = c;
   209→   209→            break;
   210→   210→          }
   211→   211→        }
   212→   212→      }
   213→   213→    }
   214→   214→    if (cols[key] === -1) cols[key] = spec.fallback[0] ?? -1;
   215→   215→  }
   216→   216→  return cols;
   217→   217→}
   218→   218→
   219→   219→// Pick the first candidate cell that satisfies pred, skipping columns that
   220→   220→// belong to other resolved fields (e.g. the To-date column next to a merged
   221→   221→// title) and the header column's merge-neighbors only when they are free.
   222→   222→function pickCell(cands: number[], exclude: number[], get: TableRowGetter, pred: (v: string) => boolean): string {
   223→   223→  for (const c of cands) {
   224→   224→    if (c < 0 || exclude.includes(c)) continue;
   225→   225→    const v = get(c);
   226→   226→    if (pred(v)) return v;
   227→   227→  }
   228→   228→  return "";
   229→   229→}

   231→   231→function parseTableAcrossSheets(wb: XLSX.WorkBook, spec: TableSpec): ExtractionSection[] {
   232→   232→  const entries: ExtractionSection[] = [];
   233→   233→  for (const name of wb.SheetNames) {
   234→   234→    if (/lookup/i.test(name)) continue;
   235→   235→    const ws = wb.Sheets[name];
   236→   236→    if (!ws) continue;
   237→   237→    const header = findRow(ws, spec.headerRe, 250) ?? (spec.fallbackHeaderRe ? findRow(ws, spec.fallbackHeaderRe, 250) : null);
   238→   238→    if (!header) continue;
   239→   239→    const cols = resolveColumns(ws, header.row, spec.fields);
   240→   240→    const ref = XLSX.utils.decode_range(ws["!ref"]!);
   241→   241→    // A section table ends at the next "(Continue on separate sheet …)" note,
   242→   242→    // the next roman-numeral section title, or a signature row — without this
   243→   243→    // clamp the scan bleeds into the FOLLOWING section's table (e.g. the
   244→   244→    // eligibility window reading the work-experience header row). The note is
   245→   245→    // tested on the RAW text because probeText strips its parentheses.
   246→   246→    let sectionEnd = Math.min(ref.e.r, header.row + spec.maxRows);
   247→   247→    for (let r = header.row + 1; r <= sectionEnd; r++) {
   248→   248→      const cells = (c: number) => cellText(ws[XLSX.utils.encode_cell({ r, c })]?.v);
   249→   249→      const raw = [0, 1, 2, 3, 4, 5, 6, 7].map(cells).join(" ");
   250→   250→      const probe = [0, 1, 2].map((c) => probeText(cells(c))).join(" ");
   251→   251→      const tail = [3, 4, 5, 6, 7].map(cells).join("");
   252→   252→      // "Continue on separate sheet …" (2017) and "Continue on sheet C5 …"
   253→   253→      // (2026) both close a section; so do signature rows and the next
   254→   254→      // roman-numeral section title.
   255→   255→      if (/continue\s+on\s+(a\s+)?(the\s+)?(separate|sheet)|signature|administering/i.test(raw) || (/^\s*[ivxlc]+\s*\.\s/i.test(probe) && tail === "")) {
   256→   256→        sectionEnd = r - 1;
   257→   257→        break;
   258→   258→      }
   259→   259→    }
   260→   260→    for (let r = header.row + 1; r <= sectionEnd; r++) {
   261→   261→      const get: TableRowGetter = (c) => (c >= 0 ? cellDisplay(ws, r, c) : "");
   262→   262→      try {
   263→   263→        const entry = spec.build(get, cols);
   264→   264→        // Never emit rows that resolved to nothing (blank templates leave
   265→   265→        // empty shells behind).
   266→   266→        if (entry && Object.values(entry).some((v) => v && v.value !== null && v.value !== "")) entries.push(entry);
   267→   267→      } catch {
   268→   268→        // malformed row — skip
   269→   269→      }
   270→   270→    }
   271→   271→  }
   272→   272→  return entries;
   273→   273→}

   442→   442→function detectRevision(wb: XLSX.WorkBook): { label: string; year: number | null } {
   443→   443→  let formOnly = false;
   444→   444→  for (const name of wb.SheetNames) {
   445→   445→    if (/lookup/i.test(name)) continue;
   446→   446→    const ws = wb.Sheets[name];
   447→   447→    const ref = ws["!ref"] ? XLSX.utils.decode_range(ws["!ref"]) : null;
   448→   448→    if (!ref) continue;
   449→   449→    for (let r = ref.s.r; r <= Math.min(ref.e.r, 120); r++) {
   450→   450→      for (let c = ref.s.c; c <= Math.min(ref.e.c, ref.s.c + 15); c++) {
   451→   451→        const t = cellText(ws[XLSX.utils.encode_cell({ r, c })]?.v);
   452→   452→        // The explicit "Revised <year>" footer wins over the bare form
   453→   453→        // number so the 2026 edition is never mislabelled "CS Form 212".
   454→   454→        const m = t.match(/revised\s*(20\d{2})/i);
   455→   455→        if (m) return { label: `CS Form 212 (Revised ${m[1]})`, year: Number(m[1]) };
   456→   456→        if (/cs\s*form\s*no\.?\s*\.?\s*212/i.test(t)) formOnly = true;
   457→   457→      }
   458→   458→    }
   459→   459→  }
   460→   460→  return { label: formOnly ? "CS Form 212" : "", year: null };
   461→   461→}

   463→   463→export function parsePdsWorkbook(filePath: string): ExtractionResult | null {
   464→   464→  let wb: XLSX.WorkBook;
   465→   465→  try {
   466→   466→    wb = XLSX.readFile(filePath, { cellDates: false, cellNF: true });
   467→   467→  } catch {
   468→   468→    return null;
   469→   469→  }
   470→   470→  const sheetName = wb.SheetNames.find((n) => /C1|Sheet1|personal/i.test(n)) || wb.SheetNames[0];
   471→   471→  if (!sheetName) return null;
   472→   472→  const ws = wb.Sheets[sheetName];
   473→   473→  const a1 = probeText(ws["A1"]?.v ?? ws["A2"]?.v ?? ws["A3"]?.v);
   474→   474→  const looksLikePds =
   475→   475→    /personal\s*data\s*sheet/i.test(a1) ||
   476→   476→    wb.SheetNames.some((n) => /C\d/i.test(n)) ||
   477→   477→    !!findRow(ws, /personal\s*data\s*sheet/i, 12);
   478→   478→  if (!looksLikePds) return null;
   479→   479→
   480→   480→  const result: ExtractionResult = { warnings: [] };
   481→   481→  const rev = detectRevision(wb);
   482→   482→  pdsDayFirst = (rev.year ?? 0) >= 2026;
   483→   483→  if (rev.label) result.warnings?.push(`Detected ${rev.label} layout.`);
   484→   484→
   485→   485→  const personal: ExtractionSection = {};
   486→   486→  const put = (key: string, v: string, conf: ConfidentValue["confidence"] = "high") => {
   487→   487→    const t = String(v).replace(/\s+/g, " ").trim();
   488→   488→    if (!t || isNA(t) || CAPTION_RE.test(t)) return;
   489→   489→    personal[key] = cv(t, conf);
   490→   490→  };
   491→   491→
   492→   492→  const surname = findRow(ws, /sur\s*name/i) || findRow(ws, /^surname/i);
   493→   493→  if (surname) put("lastName", scanRight(ws, surname.row, surname.col));
   494→   494→  const given = findRow(ws, /given\s*name|first\s*name/i);
   495→   495→  if (given) put("firstName", scanRight(ws, given.row, given.col));
   496→   496→  const middle = findRow(ws, /middle\s*name/i);
   497→   497→  if (middle) put("middleName", scanRight(ws, middle.row, middle.col));
   498→   498→  const ext = findRow(ws, /name\s*extension/i);
   499→   499→  if (ext) put("extensionName", scanRight(ws, ext.row, ext.col));
   500→   500→  const dob = findRow(ws, /date\s*of\s*birth/i);
   501→   501→  if (dob) put("birthDate", normDate(scanRight(ws, dob.row, dob.col)), "high");
   502→   502→  const pob = findRow(ws, /place\s*of\s*birth/i);
   503→   503→  if (pob) put("birthPlace", scanRight(ws, pob.row, pob.col));
   504→   504→  const sex = findRow(ws, /\bsex\b|\bgender\b/i); // "5. SEX" (2017) / "5. SEX AT BIRTH" (2026)
   505→   505→  if (sex) {
   506→   506→    const raw = scanRight(ws, sex.row, sex.col);
   507→   507→    const m = raw.match(/male|female/i);
   508→   508→    if (m) put("gender", m[0][0].toUpperCase() + m[0].slice(1).toLowerCase());
   509→   509→  }
   510→   510→  const cs = findRow(ws, /civil\s*status/i);
   511→   511→  if (cs) {
   512→   512→    const raw = scanRight(ws, cs.row, cs.col).toLowerCase();
   513→   513→    const hit = ["single", "married", "widowed", "separated", "divorced"].find((s) => raw.includes(s));
   514→   514→    if (hit) put("civilStatus", hit[0].toUpperCase() + hit.slice(1));
   515→   515→  }
   516→   516→  const cit = findRow(ws, /citizenship/i);
   517→   517→  if (cit) put("citizenship", scanRight(ws, cit.row, cit.col));
   518→   518→
   519→   519→  // Address blocks — caption-anchored: each underline caption ("House/Block/
   520→   520→  // Lot No.", "Subdivision/Village", …) sits directly BENEATH its value cell,
   521→   521→  // and the ZIP caption carries its value to its right. Residential wins over
   522→   522→  // permanent when both are filled.
   523→   523→  const permAnchor = findRow(ws, /permanent\s*address/i);
   524→   524→  const resAnchor = findRow(ws, /residential\s*address/i);
   525→   525→  type AddrVals = Partial<Record<"houseNumber" | "street" | "subdivision" | "barangay" | "city" | "province" | "zipCode", string>>;
   526→   526→  const readAddressBlock = (anchor: { row: number; col: number } | null, hardEnd?: number): AddrVals => {
   527→   527→    const vals: AddrVals = {};
   528→   528→    if (!anchor) return vals;
   529→   529→    const ref = ws["!ref"] ? XLSX.utils.decode_range(ws["!ref"]) : null;
   530→   530→    const endRow = Math.min(hardEnd ?? ref?.e.r ?? anchor.row, anchor.row + 10);
   531→   531→    const cEnd = Math.min(ref?.e.c ?? anchor.col + 8, anchor.col + 8);
   532→   532→    for (let r = anchor.row; r <= endRow; r++) {
   533→   533→      for (let c = anchor.col; c <= cEnd; c++) {
   534→   534→        const t = cellText(ws[XLSX.utils.encode_cell({ r, c })]?.v);
   535→   535→        if (!t) continue;
   536→   536→        const cap = /house\/?block/i.test(t) ? "houseNumber"
   537→   537→          : /^street$/i.test(t) ? "street"
   538→   538→          : /subdivision/i.test(t) ? "subdivision"
   539→   539→          : /^barangay$/i.test(t) ? "barangay"
   540→   540→          : /city\/?municipality/i.test(t) ? "city"
   541→   541→          : /^province$/i.test(t) ? "province"
   542→   542→          : /^zip\s*code/i.test(t) ? "zipCode"
   543→   543→          : null;
   544→   544→        if (!cap || vals[cap]) continue;
   545→   545→        const value = cap === "zipCode" ? scanRight(ws, r, c) : scanAbove(ws, r, c);
   546→   546→        if (value && !isNA(value)) vals[cap] = value;
   547→   547→      }
   548→   548→    }
   549→   549→    return vals;
   550→   550→  };
   551→   551→  const addr: AddrVals = {
   552→   552→    ...readAddressBlock(permAnchor),
   553→   553→    ...readAddressBlock(resAnchor, permAnchor ? permAnchor.row - 1 : undefined),
   554→   554→  };
   555→   555→  for (const [k, v] of Object.entries(addr)) if (v) put(k, v, "medium");
   556→   556→  const presentAddress = (["houseNumber", "street", "subdivision", "barangay", "city", "province"] as const)
   557→   557→    .map((k) => addr[k])
   558→   558→    .filter(Boolean)
   559→   559→    .join(", ");
   560→   560→  if (presentAddress) put("presentAddress", presentAddress, "medium");
   561→   561→
   562→   562→  const tel = findRow(ws, /tele(phone)?\s*\/?\s*mobile|mobile/i);
   563→   563→  if (tel) {
   564→   564→    // "09270035829 / 09922092759" — several numbers may share the cell; the
   565→   565→    // profile keeps one, so take the first valid PH-style number.
   566→   566→    const digits = scanRight(ws, tel.row, tel.col).replace(/[\s()\-]/g, "");
   567→   567→    const m = digits.match(/(?:\+?63|0)\d{9,10}|\d{7,12}/);
   568→   568→    if (m) put("mobileNumber", m[0]);
   569→   569→  }
   570→   570→  const email = findRow(ws, /e-?mail/i);
   571→   571→  if (email) {
   572→   572→    const raw = scanRight(ws, email.row, email.col);
   573→   573→    if (raw.includes("@")) put("emailAddress", raw);
   574→   574→  }
   575→   575→
   576→   576→  const fieldsExtracted = Object.keys(personal).length;
   577→   577→  if (fieldsExtracted === 0 && wb.SheetNames.length <= 1) {
   578→   578→    result.warnings?.push("No PDS labels recognized");
   579→   579→    return result;
   580→   580→  }
   581→   581→  if (fieldsExtracted > 0) result.personalInfo = personal;
   582→   582→
   583→   583→  // ── Section tables — header-resolved, collected across every sheet ────────
   584→   584→  for (const spec of TABLE_SPECS) {
   585→   585→    const entries = dedupeSectionArray(parseTableAcrossSheets(wb, spec), spec.key);
   586→   586→    if (entries.length > 0) result[spec.key] = entries;
   587→   587→  }
   588→   588→
   589→   589→  // ── Legacy/lower-version fallback: simple label:value education rows ──────
   590→   590→  // Older single-sheet PDS exports list educational background as plain
   591→   591→  // "ELEMENTARY: <school>" style rows instead of a headered table.
   592→   592→  if (!result.educations) {
   593→   593→    const legacyEdu: ExtractionSection[] = [];
   594→   594→    for (const name of wb.SheetNames) {
   595→   595→      if (/lookup/i.test(name)) continue;
   596→   596→      const s = wb.Sheets[name];
   597→   597→      const ref = s["!ref"] ? XLSX.utils.decode_range(s["!ref"]) : null;
   598→   598→      if (!ref) continue;
   599→   599→      for (let r = ref.s.r; r <= Math.min(ref.e.r, 120); r++) {
   600→   600→        for (let c = ref.s.c; c <= Math.min(ref.e.c, ref.s.c + 3); c++) {
   601→   601→          const label = probeText(s[XLSX.utils.encode_cell({ r, c })]?.v);
   602→   602→          if (!/^(elementary|secondary|high\s*school|vocational|college|bachelor|graduate|post\s*graduate)/i.test(label)) continue;
   603→   603→          const value = scanRight(s, r, c);
   604→   604→          if (isNA(value) || value.length < 3) continue;
   605→   605→          legacyEdu.push({ educationLevel: cv(EDUCATION_LEVEL_MAP(label), "medium"), schoolName: cv(value, "medium") });
   606→   606→        }
   607→   607→      }
   608→   608→    }
   609→   609→    const deduped = dedupeSectionArray(legacyEdu, "educations");
   610→   610→    if (deduped.length > 0 && deduped.length <= 8) result.educations = deduped;
   611→   611→  }
   612→   612→
   613→   613→  return result;
   614→   614→}

   644→   644→function parseAiJson(raw: string): ExtractionResult | null {

   661→   661→// (they live in form controls), so the xlsx path drops those guesses when the
   662→   662→// deterministic parser confirmed the form is otherwise empty.
   663→   663→const AI_CAPTION_VALUE = /^(surname|last\s*name|first\s*name|given\s*name|middle\s*name|name\s*extension|date\s*of\s*birth|place\s*of\s*birth|civil\s*status|sex(\s*at\s*birth)?|gender|citizenship|residential\s*address|permanent\s*address|zip\s*code|mobile\s*no\.?|telephone\s*no\.?|tele(phone)?\s*no\.?|e-?mail(\s*address)?(\s*if\s*any)?|agency\s*employee\s*no\.?|height.*|weight.*|blood\s*type|gsis\s*id\s*no\.?|umid\s*id\s*no\.?|pag-?ibig\s*id\s*no\.?|philhealth\s*no\.?|sss\s*no\.?|tin\s*no\.?|philsys.*|pls\.?\s*indicate.*|if\s*holder.*|house\/?block.*|subdivision.*|city\/?municipality.*|^street$|^barangay$|^province$|dual\s*citizenship|by\s*birth|by\s*naturalization|n\/?a\.?|none)$/i;
   664→   664→const AI_TABLE_CAPTION = /^(n\/?a\.?|none|name\s*of\s*school|position\s*title|title\s*of\s*(learning|training).*|career\s*service.*ra\s*1080.*|civil\s*service\s*eligib.*|non-?academic\s*distinctions.*|inclusive\s*dates.*|department\s*\/?\s*agency.*|scholarship.*|highest\s*level.*|number\s*of\s*hours.*|type\s*of\s*l\s*&\s*d.*|conducted.*)$/i;
   665→   665→
   666→   666→function sanitizeAiExtraction(
   667→   667→  result: ExtractionResult | null,
   668→   668→  opts: { dropCheckboxGuesses?: boolean } = {}
   669→   669→): ExtractionResult | null {
   670→   670→  if (!result) return null;
   671→   671→  const out: ExtractionResult = { ...result };
   672→   672→  if (out.personalInfo) {
   673→   673→    const clean: ExtractionSection = {};
   674→   674→    for (const [k, v] of Object.entries(out.personalInfo)) {
   675→   675→      if (v?.value == null || v.value === "") continue;
   676→   676→      const s = String(v.value).trim();
   677→   677→      if (AI_CAPTION_VALUE.test(s)) continue;
   678→   678→      if (k === "extensionName" && !/^(jr|sr|iii|iv|v)\.?$/i.test(s)) continue;
   679→   679→      if (opts.dropCheckboxGuesses && (k === "gender" || k === "civilStatus" || k === "citizenship")) continue;
   680→   680→      clean[k] = v;
   681→   681→    }
   682→   682→    out.personalInfo = Object.keys(clean).length > 0 ? clean : undefined;
   683→   683→  }
   684→   684→  const filterRows = (rows?: ExtractionSection[], fields: string[] = []) =>
   685→   685→    rows?.filter((row) =>
   686→   686→      fields.some((f) => {
   687→   687→        const v = row[f]?.value;
   688→   688→        return v != null && String(v).trim() !== "" && !AI_TABLE_CAPTION.test(String(v).trim());
   689→   689→      })
   690→   690→    );
   691→   691→  out.educations = filterRows(out.educations, ["schoolName"]);
   692→   692→  out.workExperiences = filterRows(out.workExperiences, ["positionTitle"]);
   693→   693→  out.trainings = filterRows(out.trainings, ["title"]);
   694→   694→  out.eligibilities = filterRows(out.eligibilities, ["title"]);
   695→   695→  out.awards = filterRows(out.awards, ["details"]);
   696→   696→  return out;
   697→   697→}

   720→   720→export function extractExcelText(filePath: string): string {
   721→   721→  try {
   722→   722→    const wb = XLSX.readFile(filePath, { cellDates: false });
   723→   723→    const parts: string[] = [];
   724→   724→    for (const name of wb.SheetNames) {
   725→   725→      if (/lookup/i.test(name)) continue;
   726→   726→      const ws = wb.Sheets[name];
   727→   727→      parts.push(`## Sheet: ${name}`);
   728→   728→      const rows = XLSX.utils.sheet_to_json<unknown[]>(ws, { header: 1, blankrows: false });
   729→   729→      for (const row of rows.slice(0, 300)) {
   730→   730→        const cells = (row as unknown[]).map((c) => cellText(c)).filter(Boolean);
   731→   731→        if (cells.length > 0) parts.push(cells.join(" | "));
   732→   732→      }
   733→   733→    }
   734→   734→    return parts.join("\n").slice(0, 16000);
   735→   735→  } catch {
   736→   736→    return "";
   737→   737→  }
   738→   738→}

   740→   740→export async function extractPdfText(filePath: string): Promise<string> {
   741→   741→  try {
   742→   742→    const { getDocumentProxy } = await import("unpdf");
   743→   743→    const buffer = fs.readFileSync(filePath);
   744→   744→    const pdf = await getDocumentProxy(new Uint8Array(buffer));
   745→   745→    // Form PDFs (like the PDS) store glyphs with coordinates, so the raw text
   746→   746→    // stream is jumbled and labels get separated from their values. Rebuild
   747→   747→    // VISUAL LINES by grouping glyphs on the same baseline (y) and inserting
   748→   748→    // a "│" cell separator on wide horizontal gaps — the AI then sees the
   749→   749→    // form's rows the way a human does.
   750→   750→    const pages: string[] = [];
   751→   751→    for (let p = 1; p <= pdf.numPages; p++) {
   752→   752→      const page = await pdf.getPage(p);
   753→   753→      const tc = await page.getTextContent();
   754→   754→      const items = tc.items as Array<{ str?: string; transform?: number[]; width?: number }>;
   755→   755→      const lines: Array<{ y: number; parts: Array<{ x: number; str: string; w: number }> }> = [];
   756→   756→      for (const it of items) {
   757→   757→        const s = it.str?.trim();
   758→   758→        if (!s || !it.transform) continue;
   759→   759→        const y = it.transform[5];
   760→   760→        let line = lines.find((l) => Math.abs(l.y - y) <= 3);
   761→   761→        if (!line) {
   762→   762→          line = { y, parts: [] };
   763→   763→          lines.push(line);
   764→   764→        }
   765→   765→        line.parts.push({ x: it.transform[4], str: s, w: it.width ?? 0 });
   766→   766→      }
   767→   767→      lines.sort((a, b) => b.y - a.y);
   768→   768→      const rows: string[] = [];
   769→   769→      for (const l of lines) {
   770→   770→        l.parts.sort((a, b) => a.x - b.x);
   771→   771→        let row = "";
   772→   772→        let prevEnd = -Infinity;
   773→   773→        for (const part of l.parts) {
   774→   774→          if (prevEnd > -Infinity && part.x - prevEnd > 18) row += " │ ";
   775→   775→          else if (row) row += " ";
   776→   776→          row += part.str;
   777→   777→          prevEnd = part.x + part.w;
   778→   778→        }
   779→   779→        if (row.trim()) rows.push(row.trim());
   780→   780→      }
   781→   781→      pages.push(rows.join("\n"));
   782→   782→    }
   783→   783→    return pages.map((t, i) => `=== Page ${i + 1} ===\n${t}`).join("\n").slice(0, 16000);
   784→   784→  } catch {
   785→   785→    // Fallback: plain merged text (no coordinates).
   786→   786→    try {
   787→   787→      const { extractText, getDocumentProxy } = await import("unpdf");
   788→   788→      const buffer = fs.readFileSync(filePath);
   789→   789→      const pdf = await getDocumentProxy(new Uint8Array(buffer));
   790→   790→      const { text } = await extractText(pdf, { mergePages: true });
   791→   791→      return text.slice(0, 16000);
   792→   792→    } catch {
   793→   793→      return "";
   794→   794→    }
   795→   795→  }
   796→   796→}

   810→   810→export function countExtractedFields(result: ExtractionResult | null): number {
   811→   811→  if (!result) return 0;
   812→   812→  let n = 0;
   813→   813→  const countSection = (s?: ExtractionSection) => { if (s) n += Object.values(s).filter((v) => v && v.value !== null && v.value !== "").length; };
   814→   814→  countSection(result.personalInfo);
   815→   815→  for (const arr of [result.educations, result.workExperiences, result.trainings, result.eligibilities, result.awards]) {
   816→   816→    for (const s of arr ?? []) countSection(s);
   817→   817→  }
   818→   818→  return n;

   821→   821→export function mergeExtractions(results: (ExtractionResult | null)[]): ExtractionResult | null {
   822→   822→  const valid = results.filter((r): r is ExtractionResult => !!r);
   823→   823→  if (valid.length === 0) return null;
   824→   824→  const merged: ExtractionResult = { warnings: [] };
   825→   825→  // personal: per-field highest-confidence non-null wins
   826→   826→  const rank: Record<string, number> = { high: 3, medium: 2, low: 1, none: 0 };
   827→   827→  const personal: ExtractionSection = {};
   828→   828→  for (const r of valid) {
   829→   829→    for (const [k, v] of Object.entries(r.personalInfo ?? {})) {
   830→   830→      if (v.value === null || v.value === "") continue;
   831→   831→      const cur = personal[k];
   832→   832→      if (!cur || (rank[v.confidence] ?? 0) > (rank[cur.confidence] ?? 0)) personal[k] = v;
   833→   833→    }
   834→   834→  }
   835→   835→  if (Object.keys(personal).length > 0) merged.personalInfo = personal;
   836→   836→  const mergeArr = (key: keyof ExtractionResult) => {
   837→   837→    const all = valid.flatMap((r) => (r[key] as ExtractionSection[] | undefined) ?? []);
   838→   838→    if (all.length > 0) (merged as Record<string, unknown>)[key as string] = dedupeSectionArray(all, key as string);
   839→   839→  };
   840→   840→  mergeArr("educations");
   841→   841→  mergeArr("workExperiences");
   842→   842→  mergeArr("trainings");
   843→   843→  mergeArr("eligibilities");
   844→   844→  mergeArr("awards");
   845→   845→  merged.warnings = valid.flatMap((r) => r.warnings ?? []);
   846→   846→  return merged;
   847→   847→}
   848→   848→
   849→   849→function dedupeSectionArray(arr: ExtractionSection[], key: string): ExtractionSection[] {
   850→   850→  const seen = new Set<string>();
   851→   851→  const out: ExtractionSection[] = [];
   852→   852→  for (const item of arr) {
   853→   853→    const sig =
   854→   854→      key === "trainings" ? `${item.title?.value ?? ""}|${item.dateFrom?.value ?? ""}`
   855→   855→      : key === "awards" ? `${item.details?.value ?? ""}`
   856→   856→      : key === "workExperiences" ? `${item.positionTitle?.value ?? ""}|${item.employerName?.value ?? ""}|${item.dateFrom?.value ?? ""}`
   857→   857→      : key === "eligibilities" ? `${item.title?.value ?? ""}|${item.examDate?.value ?? ""}`
   858→   858→      : `${item.educationLevel?.value ?? ""}|${item.schoolName?.value ?? ""}`;
   859→   859→    if (seen.has(sig)) continue;
   860→   860→    seen.add(sig);
   861→   861→    out.push(item);
   862→   862→  }
   863→   863→  return out;
   864→   864→}

   866→   866→/**
   867→   867→ * Extract structured data from a stored document (spec §6.4). Deterministic
   868→   868→ * parser first for PDS × Excel; AI fallback otherwise.
   869→   869→ */
   870→   870→export async function extractFromDocument(
   871→   871→  filePath: string,
   872→   872→  category: string,
   873→   873→  mimeType: string
   874→   874→): Promise<{ result: ExtractionResult | null; error?: string }> {
   875→   875→  const ext = path.extname(filePath).toLowerCase();
   876→   876→  if (/\.(xlsx|xls|xlsm)$/.test(ext)) {
   877→   877→    const deterministic = parsePdsWorkbook(filePath);
   878→   878→    if (deterministic && countExtractedFields(deterministic) >= 3) return { result: deterministic };
   879→   879→    const text = extractExcelText(filePath);
   880→   880→    if (text.trim().length > 40) {
   881→   881→      // When the deterministic parser RECOGNIZED the form but found nothing
   882→   882→      // in it, the file is a blank template — checkbox guesses from raw cell
   883→   883→      // text would be pure fabrication, so they are suppressed.
   884→   884→      const ai = sanitizeAiExtraction(await aiExtractText(text, category), {
   885→   885→        dropCheckboxGuesses: !!deterministic && countExtractedFields(deterministic) === 0,
   886→   886→      });
   887→   887→      return { result: mergeExtractions([deterministic, ai]) };
   888→   888→    }
   889→   889→    return { result: deterministic };
   890→   890→  }
   891→   891→  if (ext === ".pdf") {
   892→   892→    const text = await extractPdfText(filePath);
   893→   893→    if (!text.trim()) return { result: null, error: "Could not read text from this PDF. Try uploading the Excel version of the PDS." };
   894→   894→    return { result: sanitizeAiExtraction(await aiExtractText(text, category)) };
   895→   895→  }
   896→   896→  if (ext === ".docx" || ext === ".doc") {
   897→   897→    const text = await extractDocxText(filePath);
   898→   898→    if (!text.trim()) return { result: null, error: "Could not read this document. Try a DOCX or XLSX file." };
   899→   899→    return { result: sanitizeAiExtraction(await aiExtractText(text, category)) };
   900→   900→  }
   901→   901→  if (/\.(png|jpg|jpeg|gif|webp|bmp)$/.test(ext)) {
   902→   902→    return { result: null, error: "Image extraction is not available in this deployment. Please upload the PDS as XLSX, PDF, or DOCX." };
   903→   903→  }
   904→   904→  return { result: null, error: `Unsupported file type for extraction (${mimeType}).` };
   905→   905→}
