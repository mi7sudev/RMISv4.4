   131→export function toneClass(tone: Tone): string {
   132→  const map: Record<Tone, StatusVariant> = {

   134→    warning: "warn",
   135→    danger: "bad",
   136→    info: "info",

   143→/**
   144→ * Canonical stage/status pill. Displays the normalized status label; the
   145→ * variant comes from variantForStatus unless explicitly overridden (used for
   146→ * completion / verdict labels that are not application statuses).
   147→ */
   148→export function StatusPill({
   149→  status,
   150→  variant,
   151→  className = "",
   152→}: {
   153→  status: string | null | undefined;
   154→  variant?: StatusVariant;
   155→  className?: string;
   156→}) {
   157→  const label = getStatusMeta(status).label;
   158→  const v: StatusVariant = variant ?? variantForStatus(status);
   159→  return <span className={cn(pillClass(v), "shrink-0", className)}>{label}</span>;
   160→}

   162→function verdictMeta(verdict: string): { chip: string; variant: StatusVariant; meaning: string } {
   163→  switch (verdict) {
   164→    case "ALL_MET":
   165→      return { chip: "Qualified", variant: variantForVerdict("Qualified"), meaning: "Meets the minimum requirements" };
   166→    case "PARTIAL":
   167→      return { chip: "Partial", variant: variantForVerdict("Partial"), meaning: "Partially meets the minimum requirements" };
   168→    case "NONE_MET":
   169→      return { chip: "Not qualified", variant: variantForVerdict("Not qualified"), meaning: "Does not meet the minimum requirements" };
   170→    case "NEEDS_REVIEW":
   171→      return { chip: "Verify", variant: variantForVerdict("Verify"), meaning: "Needs manual verification" };
   172→    case "NO_REQUIREMENTS":
   173→      return { chip: "No reqs", variant: variantForVerdict("No reqs"), meaning: "No published requirements" };
   174→    default:
   175→      return { chip: verdict, variant: variantForVerdict(verdict), meaning: "" };
   176→  }
   177→}

   193→/** Gradient monogram avatar (wave-3) with initials derived from the name. */
   194→function MonogramAvatar({ name, size = 36, warm = false }: { name: string; size?: number; warm?: boolean }) {
   195→  const initials = name
   196→    .split(/\s+/)
   197→    .filter(Boolean)
   198→    .slice(0, 2)
   199→    .map((w) => w[0]?.toUpperCase() ?? "")
   200→    .join("");
   201→  return (
   202→    <span aria-hidden="true">
   203→      <Monogram size={size} warm={warm} className="rounded-full">{initials || "?"}</Monogram>
   204→    </span>
   205→  );
   206→}

   226→  return s;

   757→function parseAttachments(json: string | null): { name: string; bytes: number }[] {
   758→  if (!json) return [];
   759→  try {
   760→    const parsed: unknown = JSON.parse(json);
   761→    return Array.isArray(parsed) ? (parsed as { name: string; bytes: number }[]) : [];
   762→  } catch {
   763→    return [];
   764→  }

   771→/** Composer dialog: subject + content, recipient auto-retrieved from the
   772→    applicant profile (the server re-resolves it on send). Send is guarded by
   773→    a confirmation dialog. */
   774→function DirectEmailDialog({
   775→  applicationId,
   776→  to,
   777→  onClose,
   778→  onSent,
   779→}: {
   780→  applicationId: number;
   781→  to: string;
   782→  onClose: () => void;
   783→  onSent: () => void;
   784→}) {
   785→  const [subject, setSubject] = useState("");
   786→  const [message, setMessage] = useState("");
   787→  const [files, setFiles] = useState<File[]>([]);
   788→  const [busy, setBusy] = useState(false);
   789→  const [confirmOpen, setConfirmOpen] = useState(false);
   790→  const fileRef = useRef<HTMLInputElement>(null);
   791→
   792→  const pickFiles = (list: FileList | null) => {
   793→    if (!list) return;
   794→    const next = Array.from(list);
   795→    if (next.length + files.length > 3) {
   796→      toast.error("A maximum of 3 attachments is allowed");
   797→      return;
   798→    }
   799→    if (next.some((f) => f.size > 5 * 1024 * 1024)) {
   800→      toast.error("Each attachment must be 5 MB or smaller");
   801→      return;
   802→    }
   803→    setFiles((prev) => [...prev, ...next].slice(0, 3));
   804→    if (fileRef.current) fileRef.current.value = "";
   805→  };
   806→
   807→  const send = async () => {
   808→    if (!message.trim()) return;
   809→    setBusy(true);
   810→    try {
   811→      const fd = new FormData();
   812→      fd.set("subject", subject.trim());
   813→      fd.set("message", message.trim());
   814→      for (const f of files) fd.append("attachments", f);
   815→      const res = await apiFetch<{ email: { status: string; error?: string | null }; attachments: number }>(
   816→        `/api/evaluator/applications/${applicationId}/email`,
   817→        { method: "POST", formData: fd }
   818→      );
   819→      if (res.email?.status === "failed") {
   820→        toast.error("Email delivery failed", { description: res.email.error ?? undefined });
   821→      } else {
   822→        toast.success(`Direct email ${res.email?.status}`, {
   823→          description: res.attachments ? `${res.attachments} attachment(s) included.` : undefined,
   824→        });
   825→      }
   826→      setConfirmOpen(false);
   827→      onClose();
   828→      onSent();
   829→    } catch (e) {
   830→      setConfirmOpen(false);
   831→      toast.error(e instanceof Error ? e.message : "Failed to send the email");
   832→    } finally {
   833→      setBusy(false);
   834→    }
   835→  };
   836→
   837→  return (
   838→    <>
   839→      <Dialog open onOpenChange={(v) => !v && onClose()}>
   840→        <DialogContent className="max-h-[92dvh] overflow-y-auto scroll-thin sm:max-w-lg">

   842→            <DialogTitle className="text-2xl font-semibold leading-[1.1] tracking-[-0.01em]">Send a direct email</DialogTitle>

   844→              The recipient is retrieved automatically from the email on the applicant's profile.

   846→          </DialogHeader>

   848→            <div className="space-y-1.5">
   849→              <Label htmlFor="direct-email-to" className="text-xs text-stone">
   850→                To
   851→              </Label>
   852→              <Input id="direct-email-to" className="dlg-input" value={to} readOnly aria-label="Recipient email" />
   853→            </div>
   854→            <div className="space-y-1.5">
   855→              <Label htmlFor="direct-email-subject" className="text-xs text-stone">
   856→                Subject
   857→              </Label>
   858→              <Input
   859→                id="direct-email-subject"
   860→                className="dlg-input"
   861→                placeholder="Subject"
   862→                value={subject}
   863→                maxLength={200}
   864→                onChange={(e) => setSubject(e.target.value)}
   865→              />
   866→            </div>
   867→            <div className="space-y-1.5">
   868→              <Label htmlFor="direct-email-message" className="text-xs text-stone">

   894→                <ul className="mt-2 space-y-1">
