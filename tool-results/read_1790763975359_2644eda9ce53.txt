    67→function hasText(v: string | null | undefined): boolean {
    68→  return typeof v === "string" && v.trim() !== "";

    71→/** Circular completion ring — square line caps, sanctioned ember gradient. */
    72→function CompletionRing({ pct }: { pct: number }) {
    73→  const R = 30;
    74→  const C = 2 * Math.PI * R;
    75→  return (
    76→    <div className="relative h-[72px] w-[72px] shrink-0" role="img" aria-label={`${pct}% profile complete`}>
    77→      <svg viewBox="0 0 72 72" className="h-full w-full -rotate-90">
    78→        <defs>
    79→          <linearGradient id="profileRingGrad" x1="0%" y1="0%" x2="100%" y2="100%">
    80→            <stop offset="0%" stopColor="#1591dc" />
    81→            <stop offset="100%" stopColor="#0e7abf" />
    82→          </linearGradient>
    83→        </defs>
    84→        <circle cx="36" cy="36" r={R} fill="none" className="stroke-fog" strokeWidth="7" />
    85→        <circle
    86→          cx="36"
    87→          cy="36"
    88→          r={R}
    89→          fill="none"
    90→          stroke="url(#profileRingGrad)"
    91→          strokeWidth="7"
    92→          strokeDasharray={`${(pct / 100) * C} ${C}`}
    93→          strokeLinecap="butt"
    94→          className="motion-safe:transition-[stroke-dasharray] duration-700 ease-out"
    95→        />
    96→      </svg>
    97→      <span className="absolute inset-0 grid place-items-center">
    98→        <span className="num font-display text-lg leading-none text-ink">
    99→          {pct}
   100→          <span className="text-[10px]">%</span>
   101→        </span>
   102→      </span>
   103→    </div>
   104→  );
   105→}

   107→export default function ProfileView() {
   108→  const { user, refresh } = useSession();
   109→  const sectionsRef = useRef<HTMLDivElement>(null);
   110→  const [profile, setProfile] = useState<ApplicantProfile | null>(null);
   111→  const [reference, setReference] = useState<ReferenceData | null>(null);
   112→  const [docs, setDocs] = useState<DocumentWire[]>([]);
   113→  const [loading, setLoading] = useState(true);
   114→  const [error, setError] = useState<string | null>(null);
   115→  const [active, setActive] = useState(1);
   116→  const [seed, setSeed] = useState(0);
   117→  const [markOpen, setMarkOpen] = useState(false);
   118→  const [marking, setMarking] = useState(false);
   119→
   120→  const loadAll = useCallback(async (silent: boolean) => {
   121→    if (!silent) setLoading(true);
   122→    try {
   123→      const [profileRes, docsRes] = await Promise.all([
   124→        apiFetch<ApplicantProfile>("/api/applicant/profile"),
   125→        apiFetch<DocumentWire[]>("/api/applicant/documents"),
   126→      ]);
   127→      setProfile(profileRes);
   128→      setDocs(docsRes);
   129→      setError(null);
   130→    } catch (e) {
   131→      if (!silent) setError(e instanceof Error ? e.message : "Unable to load your profile");
   132→    } finally {
   133→      if (!silent) setLoading(false);
   134→    }
   135→  }, []);
   136→
   137→  useEffect(() => {
   138→    void (async () => {
   139→      try {
   140→        const ref = await apiFetch<ReferenceData>("/api/reference");
   141→        setReference(ref);
   142→      } catch {
   143→        // Reference data is optional (datalist/eligibility suggestions).
   144→      }
   145→      await loadAll(false);
   146→    })();
   147→  }, [loadAll]);
   148→
   149→  const reloadSilent = useCallback(async () => {
   150→    await loadAll(true);
   151→  }, [loadAll]);
   152→
   153→  /** After PDS auto-apply / clear: reload AND re-seed the personal form. */
   154→  const onProfileReplaced = useCallback(async () => {
   155→    await loadAll(true);
   156→    setSeed((s) => s + 1);
   157→  }, [loadAll]);
   158→
   159→  // ── Completeness (§7.4 header card) ───────────────────────────────────────
   160→  const personalDone = profile ? hasText(profile.firstName) && hasText(profile.lastName) && hasText(profile.emailAddress) : false;
   161→  const educationDone = (profile?.educations?.length ?? 0) > 0;
   162→  const workDone = (profile?.workExperiences?.length ?? 0) > 0;
   163→  const trainingDone = (profile?.trainings?.length ?? 0) > 0;
   164→  const eligibilityDone = (profile?.eligibilities?.length ?? 0) > 0;
   165→  const awardsDone = (profile?.awards?.length ?? 0) > 0;
   166→  const documentsDone = docs.length > 0;
   167→  const sectionDone = [personalDone, educationDone, workDone, trainingDone, eligibilityDone, awardsDone, documentsDone];
   168→  const completedCount = sectionDone.filter(Boolean).length;
   169→  const isComplete = profile?.isProfileComplete === true;
   170→  const canMarkComplete = personalDone && educationDone && workDone && !isComplete;
   171→  const pct = Math.round((completedCount / 7) * 100);
   172→  const remaining = 7 - completedCount;
   173→
   174→  const markComplete = async () => {
   175→    setMarking(true);
   176→    try {
   177→      await apiFetch("/api/applicant/profile", { method: "PUT", body: { isProfileComplete: true } });
   178→      toast.success("Profile marked as complete", {
   179→        description: "You can now apply to open positions. HR verifies your credentials at the next stage.",
   180→      });
   181→      setMarkOpen(false);
   182→      await refresh();
   183→      await reloadSilent();
   184→    } catch (e) {
   185→      toast.error(e instanceof Error ? e.message : "Unable to mark your profile complete");
   186→    } finally {
   187→      setMarking(false);
   188→    }
   189→  };
   190→
   191→  const pictureDoc = docs.find((d) => d.category === "PROFILE_PICTURE");
   192→  const initials = initialsOf(user?.firstName ?? profile?.firstName, user?.lastName ?? profile?.lastName);
   193→  const applicantName =
   194→    fullName({
   195→      firstName: user?.firstName ?? profile?.firstName,
   196→      lastName: user?.lastName ?? profile?.lastName,
   197→    }) || "Applicant";
   198→
   199→  // ── Mobile stepper: switch section AND bring the rail back into view —
   200→  // tapping "04" while stranded deep inside section 01's long form must not
   201→  // leave the viewport mid-scroll in the old section. Desktop (lg+) skips
   202→  // this: the nav band is sticky above the content, no jump needed.
   203→  const mobileNavRef = useRef<HTMLElement>(null);
   204→
   205→  function handleMobileSectionChange(n: number) {
   206→    if (n === active) return;
   207→    setActive(n);
   208→    if (typeof window !== "undefined" && window.innerWidth < 1024) {
   209→      // The strip lives ABOVE the swapped section content, so its document
   210→      // position is layout-stable across the swap — unlike the wrapper's.
   211→      // Double rAF lets React commit the new section first, then we jump
   212→      // INSTANTLY (smooth is canceled by Chrome's scroll-anchoring yank,
   213→      // which fires while the tall section replaces the short one and
   214→      // re-anchors the viewport to keep the old bottom edge in view). The
   215→      // instant jump lands the pinned strip right under the top bar; the
   216→      // 200ms content fade masks the transition.
   217→      requestAnimationFrame(() => {
   218→        requestAnimationFrame(() => {
   219→          const el = mobileNavRef.current;
   220→          if (!el) return;
   221→          const top = el.getBoundingClientRect().top + window.scrollY - 72;
   222→          window.scrollTo({ top: Math.max(0, top), behavior: "instant" });
   223→        });
   224→      });
   225→    }
   226→  }
   227→
   228→  // Index of the active section — drives the mobile caption + prev/next.
   229→  const activeIdx = Math.max(
   230→    0,
   231→    SECTIONS.findIndex((s) => s.n === active)
   232→  );
   233→
   234→  // The section content column is a bounded scroll container (min/max height
   235→  // with an internal scrollbar), so every freshly-swapped section must land
   236→  // at its own top — otherwise switching from a short into a tall section
   237→  // inherits the previous section's inner scroll offset.
   238→  useEffect(() => {
   239→    sectionsRef.current?.scrollTo({ top: 0 });
   240→  }, [active]);
   241→
   242→  // ── Loading / error states ────────────────────────────────────────────────
   243→  if (loading) {
   244→    return (
   245→      <div>
   246→        {/* Record card skeleton */}
   247→        <div className="dlg-card p-4 sm:p-6">
   248→          <div className="flex items-center gap-4">
   249→            <div className="skel h-14 w-14 shrink-0" />
   250→            <div className="min-w-0 flex-1 space-y-2">
   251→              <div className="skel h-5 w-48" />
   252→              <div className="skel h-3 w-36" />
   253→            </div>
   254→            <div className="skel h-[72px] w-[72px] shrink-0 lg:hidden" />
   255→          </div>
   256→          <div className="skel mt-5 h-16 w-full" />
   257→        </div>
   258→        {/* Section nav skeleton — desktop band / mobile strip */}
   259→        <div className="glass mt-6 hidden border-b border-divider px-8 py-4 lg:block">
   260→          <div className="flex gap-6 overflow-hidden">
   261→            {SECTIONS.map((s) => (
   262→              <div key={s.n} className="skel h-9 w-24 shrink-0" />
   263→            ))}
   264→          </div>
   265→        </div>
   266→        <div className="dlg-card mt-6 p-2 lg:hidden">
   267→          <div className="flex gap-2 overflow-hidden">
   268→            {SECTIONS.map((s) => (
   269→              <div key={s.n} className="skel h-11 w-12 shrink-0" />
   270→            ))}
   271→          </div>
   272→        </div>
   273→        <div className="mt-6">
   274→          <SkeletonRows rows={4} rowClassName="h-20" />
   275→        </div>
   276→      </div>
   277→    );
   278→  }
   279→
   280→  if (error && !profile) {
   281→    return (
   282→      <div>
   283→        <div className="dlg-card flex flex-col items-center gap-3 p-10 text-center">
   284→          <AlertCircle className="h-6 w-6 text-[var(--bad)]" />
   285→          <p className="text-sm text-stone">{error}</p>
   286→          <button type="button" onClick={() => void loadAll(false)} className="dlg-ghost min-h-[44px] px-6 py-2.5 text-sm">
   287→            <RefreshCw className="mr-2 inline h-3.5 w-3.5" /> Retry
   288→          </button>
   289→        </div>
   290→      </div>
   291→    );
   292→  }
   293→
   294→  return (
   295→    <div>
   296→      {/* ── A–C. Employee record — ONE cohesive card: identity + completion
   297→              indicator, the quiet completion note, and the PDS import
   298→              workspace. Flat sharp surfaces, no floating card stack. ── */}
   299→      <section aria-label="Profile record" className="dlg-card p-4 sm:p-6">
   300→        <div className="flex flex-col gap-4 sm:gap-5 lg:flex-row lg:items-center lg:justify-between lg:gap-6">
   301→          {/* Identity — avatar + name/status/documents. The completion ring
   302→              is pinned to the RIGHT of this row on phones (shrink-0, and
   303→              the name column is min-w-0 + truncate, so a long applicant
   304→              name only ever ellipsizes — it can never push the circle out
   305→              of view); on lg the ring lives in the right-hand lockup. */}
   306→          <div className="flex min-w-0 items-center gap-3 sm:gap-4">
   307→            <div className="h-14 w-14 shrink-0">
   308→              {pictureDoc ? (
   309→                <img
   310→                  src={`/api/files/${pictureDoc.filePath}`}
   311→                  alt="Profile picture"
   312→                  className="h-14 w-14 rounded-none object-cover shadow-e1"
   313→                />
   314→              ) : (
   315→                <Monogram size={56}>{initials}</Monogram>
   316→              )}
   317→            </div>
   318→            <div className="min-w-0 flex-1">
   319→              <h1 className="truncate font-display text-xl text-ink sm:text-2xl" title={applicantName}>
   320→                {applicantName}
   321→              </h1>
   322→              {/* Status BELOW the name — never squeezed beside it. */}
   323→              <div className="mt-1.5 flex flex-wrap items-center gap-x-2 gap-y-1">
   324→                {isComplete ? (
   325→                  <span className="status-pill status-ok">Complete</span>
   326→                ) : (
   327→                  <span className="status-pill status-neutral">Incomplete</span>
   328→                )}
   329→              </div>
   330→              <p className="mt-1.5 flex items-center gap-1.5 text-[13px] text-stone">
   331→                <FileText className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
   332→                <span className="num">{docs.length}</span> document{docs.length === 1 ? "" : "s"}
   333→              </p>
   334→            </div>
   335→            {/* Phones: compact ring at the right edge of the identity row.
   336→                lg hides it — the lockup below-right owns the ring there. */}
   337→            <div className="shrink-0 lg:hidden" title={`Profile ${pct}% complete`}>
   338→              <CompletionRing pct={pct} />
   339→            </div>
   340→          </div>
   341→
   342→          {/* Completion lockup — lg: ring + readout + finalize CTA on the
   343→              right (hairline-separated). Phones: the ring is already in
   344→              the identity row, so this row carries the readout + CTA only
   345→              under a hairline. Exactly one ring is visible per viewport. */}
   346→          <div className="flex items-center justify-between gap-4 border-t border-divider pt-4 lg:shrink-0 lg:border-l lg:border-t-0 lg:pl-8 lg:pt-0">
   347→            <div className="flex items-center gap-4">
   348→              <div className="hidden lg:block">
   349→                <CompletionRing pct={pct} />
   350→              </div>
   351→              <div>
   352→                <p className="text-sm font-semibold leading-tight text-ink">
   353→                  <span className="num">{completedCount}</span> of <span className="num">7</span> sections
   354→                </p>
   355→                <p className="mt-0.5 text-[13px] leading-tight text-stone">
   356→                  {isComplete ? "All sections complete" : `${remaining} to go`}
   357→                </p>
   358→              </div>
   359→            </div>
   360→            {!isComplete && canMarkComplete && (
   361→              <button
   362→                type="button"
   363→                onClick={() => setMarkOpen(true)}
   364→                disabled={marking}
   365→                className="dlg-cta focus-ring inline-flex min-h-[44px] shrink-0 items-center gap-2 px-5 py-2.5 text-sm disabled:opacity-50"
   366→              >
   367→                <ShieldCheck className="h-4 w-4" aria-hidden="true" /> Mark Complete
   368→              </button>
   369→            )}
   370→          </div>
   371→        </div>
   372→
   373→        {/* ── B. Completion note — quiet informational banner (never an alert). ── */}
   374→        {!isComplete && (
   375→          <div className="mt-5 flex items-start gap-3 bg-fog px-4 py-3">
   376→            <Info className="h-4 w-4 shrink-0 text-stone" aria-hidden="true" />
   377→            <p className="min-w-0 flex-1 text-sm leading-relaxed text-stone">
   378→              Profile completion requires your Personal Information (first name, last name, and email) and at least one
   379→              entry each in Education and Work Experiences.
   380→            </p>
   381→          </div>
   382→        )}
   383→
   384→        {/* ── C. Import workspace — PDS Upload · AI Auto-Fill ── */}
   385→        <div className="mt-4 sm:mt-5">
   386→          <PdsUploadCard docs={docs} onProfileReplaced={onProfileReplaced} onDocsChanged={reloadSilent} />
   387→        </div>
   388→      </section>
   389→
   390→      {/* ── D. Section navigation (lg+) — sticky numbered workflow band.
   391→              Desktop content column has no sticky header above it, so the
   392→              band pins to the very top of the viewport. NOT scrollable by
   393→              design: the row always fits its container. Below xl the items
   394→              compact (28px chips · 11.5px labels · "Approx" line hidden);
   395→              from xl up the full register renders (32px chips · 13.5px
   396→              labels · "Approx N Min" second line). Hairline connectors are
   397→              flex-1 so they absorb any leftover width. ── */}
   398→      <nav
   399→        aria-label="Profile section steps"
   400→        className="glass sticky top-0 z-20 -mx-4 mt-6 hidden border-b border-divider px-4 py-3 sm:-mx-6 sm:px-6 lg:block lg:-mx-8 lg:px-8"
   401→      >
   402→        <ol className="flex items-center">
   403→          {SECTIONS.map((s, i) => {
   404→            const isActive = active === s.n;
   405→            const done = sectionDone[s.n - 1];
   406→            const reached = done || isActive;
   407→            const approx = s.minutes != null ? `Approx ${s.minutes} Min` : null;
   408→            return (
   409→              <Fragment key={s.n}>
   410→                {i > 0 && <li aria-hidden="true" className="h-px min-w-1 flex-1 bg-divider xl:min-w-4" />}
   411→                <li className="shrink-0">
   412→                  <button
   413→                    type="button"
   414→                    onClick={() => setActive(s.n)}
   415→                    aria-current={isActive ? "step" : undefined}
   416→                    aria-label={`Section ${i + 1}: ${s.label}`}
   417→                    className="focus-ring group flex items-center gap-2 py-1 pl-0.5 pr-1 transition-colors xl:gap-2.5"
   418→                  >
   419→                    <span
   420→                      aria-hidden="true"
   421→                      className={cn(
   422→                        "grid h-7 w-7 shrink-0 place-items-center border text-[11px] font-semibold tabular-nums transition-colors duration-200 xl:h-8 xl:w-8 xl:text-[12px]",
   423→                        reached
   424→                          ? "border-transparent bg-gradient-to-b from-[#1591dc] to-[#0e7abf] text-white shadow-e1"
   425→                          : "border-divider bg-white text-stone group-hover:border-[#1591dc]/40 group-hover:text-ink"
   426→                      )}
   427→                    >
   428→                      {done ? <Check className="h-3.5 w-3.5" strokeWidth={3} /> : s.num}
   429→                    </span>
   430→                    <span className="flex min-w-0 flex-col leading-tight">
   431→                      <span
   432→                        className={cn(
   433→                          "whitespace-nowrap text-[11.5px] font-semibold transition-colors xl:text-[13.5px]",
   434→                          reached ? "text-ink" : "text-stone group-hover:text-ink"
   435→                        )}
   436→                      >
   437→                        {s.short}
   438→                      </span>
   439→                      {approx && (
   440→                        <span className="mt-0.5 hidden whitespace-nowrap text-[11px] leading-none text-pebble xl:block">
   441→                          {approx}
   442→                        </span>
   443→                      )}
   444→                    </span>
   445→                  </button>
   446→                </li>
   447→              </Fragment>
   448→            );
   449→          })}
   450→        </ol>
   451→      </nav>
   452→
   453→      {/* MOBILE (< lg) — sticky segmented strip + caption + prev/next.
   454→          The active cell is a sliding ember block (motion layoutId) —
   455→          one saturated element, no double borders. The strip is a DIRECT
   456→          child of the tall view root, NOT a grid item: a sticky element
   457→          can never travel past its containing block, and a grid item's
   458→          containing block is its own grid area (exactly the strip's own
   459→          height), which left it zero stick range. As a root sibling it
   460→          pins under the mobile top bar (top-16) for the whole scroll. */}
   461→      <nav ref={mobileNavRef} aria-label="Profile sections" className="dlg-card-plain sticky top-16 z-30 mt-4 p-1.5 shadow-e2 lg:hidden">
   462→        <div className="grid grid-cols-7 gap-0.5">
   463→          {SECTIONS.map((s, i) => {
   464→            const isActive = active === s.n;
   465→            const filled = sectionDone[s.n - 1];
   466→            return (
   467→              <button
   468→                key={s.n}
   469→                type="button"
   470→                onClick={() => handleMobileSectionChange(s.n)}
   471→                aria-label={`Section ${i + 1}: ${s.label}`}
   472→                aria-current={isActive ? "page" : undefined}
   473→                className={cn(
   474→                  "relative isolate flex min-h-11 min-w-0 flex-col items-center justify-center gap-0.5 px-0.5 transition-colors",
   475→                  isActive ? "text-white" : "text-ink hover:bg-fog"
   476→                )}
   477→              >
   478→                {/* Sliding block — rendered INSIDE the active cell so
   479→                    inset-0 matches the cell bounds; layoutId slides it
   480→                    between cells on section change. */}
   481→                {isActive && (
   482→                  <motion.span
   483→                    aria-hidden="true"
