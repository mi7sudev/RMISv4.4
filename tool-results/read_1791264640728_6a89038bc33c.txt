    69→function hasText(v: string | null | undefined): boolean {
    70→  return typeof v === "string" && v.trim() !== "";

    73→/** Circular completion ring — Metric Blue progress on a cloud track. */
    74→function CompletionRing({ pct }: { pct: number }) {
    75→  const R = 30;
    76→  const C = 2 * Math.PI * R;
    77→  return (
    78→    <div className="relative h-[72px] w-[72px] shrink-0" role="img" aria-label={`${pct}% profile complete`}>
    79→      <svg viewBox="0 0 72 72" className="h-full w-full -rotate-90">
    80→        <circle cx="36" cy="36" r={R} fill="none" className="stroke-cloud" strokeWidth="7" />
    81→        <circle
    82→          cx="36"
    83→          cy="36"
    84→          r={R}
    85→          fill="none"
    86→          className="stroke-metric-blue motion-safe:transition-[stroke-dasharray] duration-700 ease-out"
    87→          strokeWidth="7"
    88→          strokeDasharray={`${(pct / 100) * C} ${C}`}
    89→          strokeLinecap="butt"
    90→        />
    91→      </svg>
    92→      <span className="absolute inset-0 grid place-items-center">
    93→        <span className="num text-lg font-semibold leading-none text-ink">
    94→          {pct}
    95→          <span className="text-[10px]">%</span>
    96→        </span>
    97→      </span>
    98→    </div>
    99→  );
   100→}

   105→  const [profile, setProfile] = useState<ApplicantProfile | null>(null);

   111→  const [seed, setSeed] = useState(0);
   112→  const [markOpen, setMarkOpen] = useState(false);

   130→  }, []);

   132→  useEffect(() => {

   134→      try {

   137→      } catch {

   140→      await loadAll(false);

   143→
   144→  const reloadSilent = useCallback(async () => {
   145→    await loadAll(true);
   146→  }, [loadAll]);
   147→

   152→  }, [loadAll]);
   153→

   200→  function handleMobileSectionChange(n: number) {
   201→    if (n === active) return;

   203→    if (typeof window !== "undefined" && window.innerWidth < 1024) {
   204→      // The strip lives ABOVE the swapped section content, so its document
   205→      // position is layout-stable across the swap — unlike the wrapper's.
   206→      // Double rAF lets React commit the new section first, then we jump
   207→      // INSTANTLY (smooth is canceled by Chrome's scroll-anchoring yank,
   208→      // which fires while the tall section replaces the short one and
   209→      // re-anchors the viewport to keep the old bottom edge in view). The
   210→      // instant jump lands the pinned strip right under the top bar; the
   211→      // 200ms content fade masks the transition.
   212→      requestAnimationFrame(() => {
   213→        requestAnimationFrame(() => {
   214→          const el = mobileNavRef.current;
   215→          if (!el) return;
   216→          const top = el.getBoundingClientRect().top + window.scrollY - 72;

   219→      });

   221→  }

   225→    0,

   267→            ))}
   268→          </div>
   269→        </div>

   284→            <RefreshCw className="mr-2 inline h-3.5 w-3.5" /> Retry

   293→      {/* ── A–C. Employee record — ONE cohesive panel: identity + completion
   294→              indicator, the quiet completion note, and the PDS import

   296→      <section aria-label="Profile record" className="dlg-card p-5 sm:p-6">
   297→        {/* Cloud-panel layering (mirrors the job description surface): the

   301→        <div className="rounded-xl bg-white p-4 sm:p-5">
   302→          <div className="flex flex-col gap-4 sm:gap-5 lg:flex-row lg:items-center lg:justify-between lg:gap-6">

   309→            <div className="h-14 w-14 shrink-0">
   310→              {pictureDoc ? (

   312→                  src={`/api/files/${pictureDoc.filePath}`}

   327→                  <span className="status-pill status-ok">Complete</span>

   329→                  <span className="status-pill status-neutral">Incomplete</span>

   331→              </div>
   332→              <p className="mt-1.5 flex items-center gap-1.5 text-[13px] text-body-gray">
   333→                <FileText className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />

   339→            <div className="shrink-0 lg:hidden" title={`Profile ${pct}% complete`}>
   340→              <CompletionRing pct={pct} />
   341→            </div>

   343→
   344→          {/* Completion lockup — lg: ring + readout + finalize CTA on the

   350→              <div className="hidden lg:block">
   351→                <CompletionRing pct={pct} />
   352→              </div>

   354→                <p className="text-sm font-semibold leading-tight text-ink">
   355→                  <span className="num">{completedCount}</span> of <span className="num">7</span> sections

   369→                <ShieldCheck className="h-4 w-4" aria-hidden="true" /> Mark Complete

   373→          </div>

   375→
   376→        {/* ── B. Completion note — quiet informational banner (never an alert). ── */}

   378→          <div className="mt-5 flex items-start gap-3 rounded-xl bg-white px-4 py-3">
   379→            <Info className="h-4 w-4 shrink-0 text-stone" aria-hidden="true" />
   380→            <p className="min-w-0 flex-1 text-sm leading-relaxed text-stone">
   381→              Profile completion requires your Personal Information (first name, last name, and email) and at least one
   382→              entry each in Education and Work Experiences.
   383→            </p>
   384→          </div>

   386→
   387→        {/* ── C. Import workspace — PDS Upload · AI Auto-Fill ── */}

   389→          <PdsUploadCard docs={docs} onProfileReplaced={onProfileReplaced} onDocsChanged={reloadSilent} />

   392→

   400→              flex-1 so they absorb any leftover width. ── */}
   401→      <nav
   402→        aria-label="Profile section steps"
   403→        className="glass sticky top-0 z-20 -mx-4 hidden border-b border-border px-4 py-3 sm:-mx-6 sm:px-6 lg:block lg:-mx-8 lg:px-8"

   411→            return (

   418→                    aria-current={isActive ? "step" : undefined}
   419→                    aria-label={`Section ${i + 1}: ${s.label}`}
   420→                    className="focus-ring group flex items-center gap-2 py-1 pl-0.5 pr-1 transition-colors xl:gap-2.5"

   423→                      aria-hidden="true"

   439→                      >
   440→                        {s.short}

   445→                        </span>

   456→      {/* MOBILE (< lg) — sticky segmented strip + caption + prev/next.

   463→          pins under the mobile top bar (top-16) for the whole scroll. */}
   464→      <nav ref={mobileNavRef} aria-label="Profile sections" className="dlg-card-plain sticky top-16 z-30 p-1.5 shadow-e2 lg:hidden">
   465→        <div className="grid grid-cols-7 gap-0.5">
   466→          {SECTIONS.map((s, i) => {
   467→            const isActive = active === s.n;
   468→            const filled = sectionDone[s.n - 1];
   469→            return (
   470→              <button
   471→                key={s.n}
   472→                type="button"
   473→                onClick={() => handleMobileSectionChange(s.n)}
   474→                aria-label={`Section ${i + 1}: ${s.label}`}
   475→                aria-current={isActive ? "page" : undefined}
   476→                className={cn(
   477→                  "relative isolate flex min-h-11 min-w-0 flex-col items-center justify-center gap-0.5 px-0.5 transition-colors",
   478→                  isActive ? "text-primary-foreground" : "text-ink hover:bg-fog"
   479→                )}
   480→              >
   481→                {/* Sliding block — rendered INSIDE the active cell so
   482→                    inset-0 matches the cell bounds; layoutId slides it
   483→                    between cells on section change. */}
   484→                {isActive && (
   485→                  <motion.span
   486→                    aria-hidden="true"

   492→                <span
   493→                  className={cn(
   494→                    "num text-[10px] font-semibold leading-none",

   517→            );
   518→          })}
   519→        </div>
   520→        {/* Caption + sequential stepping — hairline footer bar with quiet

   522→        <div className="mt-1.5 flex items-stretch border-t border-border">
   523→          <button
   524→            type="button"
   525→            onClick={() => activeIdx > 0 && handleMobileSectionChange(SECTIONS[activeIdx - 1].n)}
   526→            disabled={activeIdx === 0}
   527→            aria-label="Previous section"
   528→            className="focus-ring grid w-12 shrink-0 place-items-center text-stone transition-colors hover:bg-fog disabled:opacity-40"

   538→          <button
   539→            type="button"
   540→            onClick={() => activeIdx < SECTIONS.length - 1 && handleMobileSectionChange(SECTIONS[activeIdx + 1].n)}
   541→            disabled={activeIdx === SECTIONS.length - 1}
   542→            aria-label="Next section"

   560→      <div
   561→        ref={sectionsRef}
   562→        className={cn(
   563→          "scroll-thin min-h-[360px] lg:min-h-[420px]",
   564→          active === 1 ? "overflow-visible" : "max-h-[65dvh] overflow-y-auto lg:max-h-[72dvh]"

   566→        style={{ overflowAnchor: "none" }}
   567→      >
   568→        {/* SECTION CONTENT — sections swap with a quiet fade/rise (200ms):
   569→            perceptible placement without choreography. */}
   570→        <div className="min-w-0">
   571→          <div key={active} className="animate-in fade-in slide-in-from-bottom-2 duration-200">

   579→            )}

   586→                onChanged={reloadSilent}

   599→      <div className="flex flex-wrap items-center justify-between gap-3 border-t border-border pt-4">
   600→        {activeIdx > 0 ? (
   601→          <button
   602→            type="button"
   603→            onClick={() => handleMobileSectionChange(SECTIONS[activeIdx - 1].n)}
   604→            className="dlg-ghost min-h-[44px] px-6 text-sm"
   605→          >
   606→            <ChevronLeft className="mr-1.5 inline h-4 w-4" aria-hidden="true" />

   610→          <span aria-hidden="true" />
   611→        )}
   612→
   613→        {activeIdx < SECTIONS.length - 1 ? (
   614→          <button
   615→            type="button"
   616→            onClick={() => handleMobileSectionChange(SECTIONS[activeIdx + 1].n)}
   617→            className="dlg-cta min-h-[44px] px-7 text-sm font-medium"
   618→          >
   619→            Continue to {SECTIONS[activeIdx + 1]?.label ?? "next section"}
   620→            <ChevronRight className="ml-1.5 inline h-4 w-4" aria-hidden="true" />
   621→          </button>
   622→        ) : (
   623→          <button
   624→            type="button"
   625→            onClick={() => navigate("home")}
   626→            className="dlg-cta min-h-[44px] px-7 text-sm font-medium"
   627→          >
   628→            Return to Portal
   629→            <ChevronRight className="ml-1.5 inline h-4 w-4" aria-hidden="true" />
   630→          </button>
   631→        )}
   632→      </div>
   633→
   634→      {/* Mark Complete confirm (§7.4) */}
   635→      <AlertDialog open={markOpen} onOpenChange={setMarkOpen}>
   636→        <AlertDialogContent className="dlg-card-plain">
   637→          <AlertDialogHeader>
   638→            <AlertDialogTitle className="text-2xl font-semibold leading-[1.1] tracking-[-0.01em] text-ink">Mark your profile as complete?</AlertDialogTitle>
   639→            <AlertDialogDescription className="text-sm leading-relaxed text-stone">
   640→              Your profile will be ready for application and subject to verification by HR. All statements become part
   641→              of your civil-service application record.
   642→            </AlertDialogDescription>
   643→          </AlertDialogHeader>
   644→          <AlertDialogFooter>
   645→            <AlertDialogCancel className="dlg-ghost min-h-[44px] px-5 py-2.5 text-sm">Not yet</AlertDialogCancel>
   646→            <AlertDialogAction
   647→              onClick={(e) => {
   648→                e.preventDefault();

   650→              }}
   651→              disabled={marking}
   652→              className="dlg-cta min-h-[44px] px-5 py-2.5 text-sm disabled:opacity-50"
   653→            >
   654→              {marking ? "Marking…" : "Mark Complete"}
   655→            </AlertDialogAction>
   656→          </AlertDialogFooter>
   657→        </AlertDialogContent>
   658→      </AlertDialog>
   659→    </div>
   660→  );
