   127→function CandidateModal({ id, onClose }: { id: number | null; onClose: () => void }) {
   128→  const [detail, setDetail] = useState<ApplicantDetail | null>(null);
   129→  const [error, setError] = useState<string | null>(null);
   130→
   131→  useEffect(() => {
   132→    if (id === null) {
   133→      setDetail(null);
   134→      setError(null);
   135→      return;
   136→    }
   137→    let alive = true;
   138→    apiFetch<ApplicantDetail>(`/api/admin/applicants/${id}`)
   139→      .then((d) => {
   140→        if (alive) setDetail(d);

   146→      alive = false;

   152→  const anyRejected = apps.some((a) => isRejectedStatus(a.status));
   153→
   154→  return (
   155→    <Dialog open={id !== null} onOpenChange={(v) => !v && onClose()}>
   156→      <DialogContent className="sm:max-w-lg max-h-[92dvh] overflow-y-auto scroll-thin">
   157→        <DialogHeader>
   158→          <DialogTitle className="text-2xl font-semibold leading-[1.1] tracking-[-0.01em]">Candidate</DialogTitle>
   159→          <DialogDescription>Quick identity and pipeline snapshot.</DialogDescription>
   160→        </DialogHeader>
   161→
   162→        {error ? (
   163→          <p className="py-4 text-sm text-[var(--bad)]">{error}</p>
   164→        ) : !detail ? (
   165→          <div className="space-y-3">
   166→            <Skeleton className="h-10 w-2/3" />
   167→            <Skeleton className="h-20 w-full rounded-lg" />
   168→          </div>
   169→        ) : (
   170→          <div className="space-y-4">
   171→            <div className="flex items-center gap-3">
   172→              <KanbanAvatar name={fullName(detail)} size={40} />
   173→              <div className="min-w-0">
   174→                <p className="truncate text-sm font-medium text-ink">{fullName(detail)}</p>
   175→                <p className="num text-xs text-pebble">#{detail.id}</p>
   176→              </div>
   177→              <span className="ml-auto shrink-0">
   178→                <StatusPill
   179→                  status={detail.isProfileComplete ? "Complete" : "Incomplete"}
   180→                  variant={variantForCompletion(detail.isProfileComplete)}
   181→                />
   182→              </span>
   183→            </div>
   184→
   185→            <div className="flex flex-wrap gap-2">
   186→              {detail.emailAddress && (
   187→                <a
   188→                  href={`mailto:${detail.emailAddress}`}
   189→                  className="dlg-pill focus-ring inline-flex min-h-[36px] items-center gap-1.5 bg-fog px-3 text-xs text-ink transition-colors hover:bg-cloud"
   190→                >
   191→                  <Mail className="h-3.5 w-3.5" aria-hidden />
   192→                  {detail.emailAddress}
   193→                </a>
   194→              )}
   195→              {detail.mobileNumber && (
   196→                <a
   197→                  href={`tel:${detail.mobileNumber}`}
   198→                  className="dlg-pill focus-ring inline-flex min-h-[36px] items-center gap-1.5 bg-fog px-3 text-xs text-ink transition-colors hover:bg-cloud"
   199→                >
   200→                  <Phone className="h-3.5 w-3.5" aria-hidden />
   201→                  {detail.mobileNumber}
   202→                </a>
   203→              )}
   204→            </div>
   205→
   206→            {/* Mini pipeline dots: Submitted → Review → Shortlisted */}
   207→            <div className="rounded-lg bg-cloud/60 p-4">
   208→              <div className="flex flex-wrap items-center justify-between gap-2">
   209→                {(["Applied", "Under Review", "Shortlisted"] as StageKey[]).map((stage, i) => (
   210→                  <div key={stage} className="flex flex-1 items-center gap-2">
   211→                    <div className="flex items-center gap-1.5">
   212→                      <StatusDot variant={countBy(stage) > 0 ? STAGE_VARIANT[stage] : "neutral"} />
   213→                      <span className="text-xs text-stone">{["Submitted", "Review", "Shortlisted"][i]}</span>
   214→                      <span className="num text-xs text-ink">{countBy(stage)}</span>
   215→                    </div>
   216→                  </div>
   217→                ))}
   218→                {anyRejected && <span className="status-pill status-bad shrink-0">Not Selected</span>}
   219→              </div>
   220→            </div>
   221→
   222→            <div>
   223→              <p className="text-xs font-medium text-stone">Education</p>
   224→              {detail.educations.length === 0 ? (
   225→                <p className="mt-1 text-xs text-pebble">No education entries.</p>
   226→              ) : (
   227→                <ul className="mt-1 space-y-1">
   228→                  {detail.educations.slice(0, 2).map((e, i) => (
   229→                    <li key={i} className="text-sm text-ink">
   230→                      {[s(e.degree) || s(e.course), s(e.schoolName)].filter(Boolean).join(" · ") || "-"}
   231→                    </li>
   232→                  ))}
   233→                </ul>
   234→              )}
   235→            </div>
   236→
   237→            <div className="flex flex-wrap gap-1.5">
   238→              <span className="status-pill status-neutral">
   239→                Documents: <span className="num">{detail.documents.length}</span>
   240→              </span>
   241→              <span className="status-pill status-neutral">
   242→                Applications: <span className="num">{detail.applicationCount ?? apps.length}</span>
   243→              </span>
   244→            </div>
   245→          </div>
   246→        )}
   247→
   248→        <DialogFooter>
   249→          <button type="button" className={ghostBtn} onClick={onClose}>

   253→            <button type="button" className={ctaBtn} onClick={() => navigate("candidate", { id: String(detail.id) })}>
   254→              View full profile
   255→            </button>
   256→          )}
   257→        </DialogFooter>
   258→      </DialogContent>
   259→    </Dialog>
   260→  );
   261→}

   566→                  </tbody>
   567→                </table>
   568→              </div>
   569→            </div>
   570→          )}
   571→
   572→          {/* Pagination */}
   573→          {total > PAGE_SIZE && (

   586→                  type="button"
   587→                  className={ghostBtn + " min-h-[44px] px-3 text-xs"}

   592→                </button>

   594→            </div>
   595→          )}
   596→        </div>
   597→      ) : (
   598→        /* Kanban — reference board: bordered columns with dot + label + count headers */
   599→        <div className="flex snap-x snap-proximity items-stretch gap-4 overflow-x-auto scroll-thin pb-2 md:grid md:grid-cols-2 xl:grid-cols-4 md:overflow-visible md:pb-0">

   615→                      >

   641→                          </div>

   643→                      </button>
   644→                    );
   645→                  })}
   646→              </KanbanColumn>
   647→            );
   648→          })}
   649→        </div>
   650→      )}
   651→
   652→      <CandidateModal id={modalId} onClose={() => setModalId(null)} />
   653→    </div>
