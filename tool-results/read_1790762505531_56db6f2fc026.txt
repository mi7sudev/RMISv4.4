    69→    69→/** Client-persisted bookmark list for the frontpage explorer. */
    70→    70→function useSavedJobs() {
    71→    71→  const [saved, setSaved] = useState<number[]>(() => {
    72→    72→    if (typeof window === "undefined") return [];
    73→    73→    try {
    74→    74→      const raw = window.localStorage.getItem(SAVED_KEY);
    75→    75→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
    76→    76→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
    77→    77→    } catch {
    78→    78→      return [];
    79→    79→    }
    80→    80→  });
    81→    81→
    82→    82→  const toggle = useCallback((id: number) => {
    83→    83→    setSaved((prev) => {
    84→    84→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
    85→    85→      try {
    86→    86→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
    87→    87→      } catch {
    88→    88→        /* storage unavailable — in-session state still works */
    89→    89→      }
    90→    90→      return next;
    91→    91→    });
    92→    92→  }, []);
    93→    93→
    94→    94→  return { saved, toggle };
    95→    95→}

   281→   281→/** Plain-text posting section rendered as a disc bullet list. */
   282→   282→function TextBullets({ text }: { text: string }) {
   283→   283→  const items = text
   284→   284→    .split(/\r?\n/)
   285→   285→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   286→   286→    .filter(Boolean);
   287→   287→  if (items.length === 0) return null;
   288→   288→  return (
   289→   289→    <ul className="list-disc space-y-2 pl-5 text-[15px] leading-relaxed text-stone">
   290→   290→      {items.map((item, i) => (
   291→   291→        <li key={i}>{item}</li>
   292→   292→      ))}
   293→   293→    </ul>
   294→   294→  );
   295→   295→}

   447→   447→export default function PublicLanding() {
   448→   448→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   449→   449→  const [error, setError] = useState<string | null>(null);
   450→   450→  const [selectedId, setSelectedId] = useState<number | null>(null);
   451→   451→  const [expandedId, setExpandedId] = useState<number | null>(null);
   452→   452→  const [isDesktop, setIsDesktop] = useState(false);
   453→   453→  const lastJson = useRef("");
   454→   454→  const paneRef = useRef<HTMLDivElement>(null);
   455→   455→  const { saved, toggle: toggleSave } = useSavedJobs();
   456→   456→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   457→   457→  // applicants get the confirm flow (same as the board), incomplete profiles
   458→   458→  // are sent to finish their profile first (the fast-track was retired).
   459→   459→  const { requestApply, dialogs } = useApplyFlow();
   460→   460→
   461→   461→  const load = useCallback(async (silent: boolean) => {
   462→   462→    try {
   463→   463→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   464→   464→      const json = JSON.stringify(data);
   465→   465→      if (!silent || json !== lastJson.current) {
   466→   466→        lastJson.current = json;
   467→   467→        setJobs(data);
   468→   468→      }
   469→   469→      setError(null);
   470→   470→    } catch (e) {
   471→   471→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   472→   472→    }
   473→   473→  }, []);
   474→   474→
   475→   475→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   476→   476→  useEffect(() => {
   477→   477→    void load(false);
   478→   478→    const timer = setInterval(() => {
   479→   479→      if (!document.hidden) void load(true);
   480→   480→    }, 30_000);
   481→   481→    const onFocus = () => void load(true);
   482→   482→    window.addEventListener("focus", onFocus);
   483→   483→    return () => {
   484→   484→      clearInterval(timer);
   485→   485→      window.removeEventListener("focus", onFocus);
   486→   486→    };
   487→   487→  }, [load]);
   488→   488→
   489→   489→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   490→   490→  useEffect(() => {
   491→   491→    const mq = window.matchMedia("(min-width: 1024px)");
   492→   492→    const sync = () => setIsDesktop(mq.matches);
   493→   493→    sync();
   494→   494→    mq.addEventListener("change", sync);
   495→   495→    return () => mq.removeEventListener("change", sync);
   496→   496→  }, []);
   497→   497→
   498→   498→  const openJobs = useMemo(
   499→   499→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   500→   500→    [jobs]
   501→   501→  );
   502→   502→
   503→   503→  const selectedJob = useMemo(
   504→   504→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   505→   505→    [openJobs, selectedId]
   506→   506→  );
   507→   507→
   508→   508→  // Desktop opens with the first posting pre-selected (reference-board
   509→   509→  // behavior); a stale selection after a poll snaps back to the top card.
   510→   510→  useEffect(() => {
   511→   511→    if (isDesktop && selectedId === null && openJobs.length > 0) {
   512→   512→      setSelectedId(openJobs[0].id);
   513→   513→    }
   514→   514→  }, [isDesktop, openJobs, selectedId]);
   515→   515→
   516→   516→  useEffect(() => {
   517→   517→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   518→   518→      setSelectedId(openJobs[0]?.id ?? null);
   519→   519→    }
   520→   520→  }, [openJobs, selectedId]);
   521→   521→
   522→   522→  // Reference-board behavior — a freshly picked posting opens from the top,
   523→   523→  // even when the previous posting was scrolled deep into its body.
   524→   524→  useEffect(() => {
   525→   525→    if (paneRef.current) paneRef.current.scrollTop = 0;
   526→   526→  }, [selectedId]);
   527→   527→
   528→   528→  const select = useCallback(
   529→   529→    (id: number) => {
   530→   530→      setSelectedId(id);
   531→   531→      if (!window.matchMedia("(min-width: 1024px)").matches) {
   532→   532→        window.scrollTo({ top: 0, behavior: "smooth" });
   533→   533→      }
   534→   534→    },
   535→   535→    []
   536→   536→  );
   537→   537→
   538→   538→  const saveAndToast = useCallback(
   539→   539→    (job: JobWire) => {
   540→   540→      const wasSaved = saved.includes(job.id);
   541→   541→      toggleSave(job.id);
   542→   542→      toast.success(
   543→   543→        wasSaved
   544→   544→          ? "Removed from your saved jobs"
   545→   545→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   546→   546→      );
   547→   547→    },
   548→   548→    [saved, toggleSave]
   549→   549→  );
   550→   550→
   551→   551→  const showList = isDesktop || selectedId === null;
   552→   552→  const showDetail = isDesktop || selectedId !== null;
   553→   553→
   554→   554→  return (
   555→   555→    <div className="flex min-h-screen flex-col lg:h-dvh">
   556→   556→      <SiteHeader />
   557→   557→
   558→   558→      <main className="mx-auto w-full max-w-[1600px] flex-1 px-4 pb-10 pt-6 sm:px-6 lg:flex lg:min-h-0 lg:flex-col lg:px-8">
   559→   559→        {/* Board toolbar */}
   560→   560→        <div className="flex items-center justify-between gap-3">
   561→   561→          <h1 className="font-display text-lg text-ink">Open Positions</h1>
   562→   562→          {openJobs.length > 0 && (
   563→   563→            <span className="dlg-pill num bg-fog px-2.5 py-1 text-[11px] font-medium text-stone">
   564→   564→              {openJobs.length} open
   565→   565→            </span>
   566→   566→          )}
   567→   567→        </div>
   568→   568→
   569→   569→        {error !== null ? (
   570→   570→          <div className="mt-6 rounded-[12px] border border-dusty-rose/30 bg-dusty-rose/10 p-4">
   571→   571→            <p className="text-sm text-ink">{error}</p>
   572→   572→            <button
   573→   573→              onClick={() => void load(false)}
   574→   574→              className="mt-2 text-sm text-stone underline underline-offset-4"
   575→   575→            >
   576→   576→              Try again
   577→   577→            </button>
   578→   578→          </div>
   579→   579→        ) : jobs === null ? (
   580→   580→          <div className="mt-6 space-y-3.5" aria-hidden="true">
   581→   581→            {[0, 1, 2].map((i) => (
   582→   582→              <div key={i} className="h-28 animate-pulse rounded-[12px] bg-fog" />
   583→   583→            ))}
   584→   584→          </div>
   585→   585→        ) : openJobs.length === 0 ? (
   586→   586→          <div className="mt-6 rounded-[12px] border border-black/[0.08] bg-white">
   587→   587→            <EmptyState
   588→   588→              icon={Inbox}
   589→   589→              title="No open positions right now"
   590→   590→              description="Check back soon or explore the full board for upcoming announcements."
   591→   591→            />
   592→   592→            <div className="pb-8 text-center">
   593→   593→              <button
   594→   594→                onClick={() => navigate("jobs")}
   595→   595→                className="inline-flex min-h-[44px] items-center gap-2 text-sm text-stone underline underline-offset-4 hover:text-ink"
   596→   596→              >
   597→   597→                Go to the full positions board
   598→   598→                <ArrowRight className="h-4 w-4" aria-hidden="true" />
   599→   599→              </button>
   600→   600→            </div>
   601→   601→          </div>
   602→   602→        ) : (

   605→   605→          // footer (main is a min-h-0 flex column on lg), so the page itself
   606→   606→          // never scrolls and the open posting stays pinned in view — the

   608→   608→          <div className="mt-5 grid gap-8 lg:min-h-0 lg:flex-1 lg:grid-cols-[minmax(340px,430px)_minmax(0,1fr)]">
   609→   609→            {/* Left · the register — 8px of ring room on every side (pl/py)
   610→   610→                with matching negative margins so the cards, the column

   612→   612→                without it the overflow-y padding box clips the selected
   613→   613→                card's outside ring on the left/top/bottom. */}
   614→   614→            {showList && (
   615→   615→              <div
   616→   616→                aria-label="Open positions list"
   617→   617→                className="scroll-thin min-h-0 space-y-3.5 lg:-ml-2 lg:-my-2 lg:overflow-y-auto lg:overscroll-contain lg:py-2 lg:pl-2 lg:pr-1.5"
   618→   618→              >
   619→   619→                {openJobs.map((job) => (

   626→   626→                    onSelect={() => select(job.id)}

   628→   628→                    onSave={() => saveAndToast(job)}
   629→   629→                  />
   630→   630→                ))}
   631→   631→              </div>
   632→   632→            )}
   633→   633→
   634→   634→            {/* Right · the posting */}
   635→   635→            {showDetail && (
   636→   636→              <div ref={paneRef} className="scroll-thin min-h-0 lg:overscroll-contain lg:overflow-y-auto lg:pr-2">
   637→   637→                {selectedJob ? (
   638→   638→                  <JobDetail
   639→   639→                    job={selectedJob}
   640→   640→                    saved={saved.includes(selectedJob.id)}
   641→   641→                    onApply={() => requestApply(selectedJob)}
   642→   642→                    onSave={() => saveAndToast(selectedJob)}
   643→   643→                    onBack={() => setSelectedId(null)}
   644→   644→                  />
   645→   645→                ) : (
   646→   646→                  <div className="rounded-[12px] border border-black/[0.08] bg-white">
   647→   647→                    <EmptyState
   648→   648→                      icon={Briefcase}
   649→   649→                      title="Select a position"
   650→   650→                      description="Choose a role from the list to read the full posting."
   651→   651→                    />
   652→   652→                  </div>
   653→   653→                )}
   654→   654→              </div>
   655→   655→            )}
   656→   656→          </div>
   657→   657→        )}
   658→   658→      </main>
   659→   659→
   660→   660→      <Footer />
   661→   661→      {/* Shared apply dialog (confirm / MQR / done), mounted once. */}
   662→   662→      {dialogs}
   663→   663→    </div>
   664→   664→  );
   665→   665→}
