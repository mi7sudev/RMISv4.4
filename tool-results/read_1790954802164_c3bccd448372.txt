    68→const SAVED_KEY = "rmis-frontpage-saved-jobs";

    70→/** Client-persisted bookmark list for the frontpage explorer. */
    71→function useSavedJobs() {
    72→  const [saved, setSaved] = useState<number[]>(() => {
    73→    if (typeof window === "undefined") return [];
    74→    try {
    75→      const raw = window.localStorage.getItem(SAVED_KEY);
    76→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
    77→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
    78→    } catch {
    79→      return [];
    80→    }
    81→  });
    82→
    83→  const toggle = useCallback((id: number) => {
    84→    setSaved((prev) => {
    85→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
    86→      try {
    87→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
    88→      } catch {
    89→        /* storage unavailable — in-session state still works */
    90→      }
    91→      return next;
    92→    });
    93→  }, []);
    94→
    95→  return { saved, toggle };
    96→}

   277→/** Plain-text posting section rendered as a disc bullet list. */
   278→function TextBullets({ text }: { text: string }) {
   279→  const items = text
   280→    .split(/\r?\n/)
   281→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   282→    .filter(Boolean);
   283→  if (items.length === 0) return null;
   284→  return (
   285→    <ul className="list-disc space-y-2 pl-5 text-[15px] leading-relaxed text-body-gray">
   286→      {items.map((item, i) => (
   287→        <li key={i}>{item}</li>
   288→      ))}
   289→    </ul>
   290→  );
   291→}
   292→
   293→/** Posting section: sanitized rich HTML when present, else text bullets. */
   294→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {
   295→  if (html && html.trim()) {
   296→    return (
   297→      <section className="space-y-3">
   298→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   299→        <div className="rich-text text-[15px] leading-relaxed text-body-gray" dangerouslySetInnerHTML={{ __html: html }} />
   300→      </section>
   301→    );
   302→  }
   303→  if (text && text.trim()) {
   304→    return (
   305→      <section className="space-y-3">
   306→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   307→        <TextBullets text={text} />
   308→      </section>
   309→    );
   310→  }
   311→  return null;
   312→}

   352→    <div>

   443→export default function PublicLanding() {
   444→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   445→  const [error, setError] = useState<string | null>(null);
   446→  const [selectedId, setSelectedId] = useState<number | null>(null);
   447→  const [expandedId, setExpandedId] = useState<number | null>(null);
   448→  const [isDesktop, setIsDesktop] = useState(false);
   449→  const lastJson = useRef("");
   450→  const paneRef = useRef<HTMLDivElement>(null);
   451→  const { saved, toggle: toggleSave } = useSavedJobs();
   452→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   453→  // applicants get the confirm flow (same as the board), incomplete profiles
   454→  // are sent to finish their profile first (the fast-track was retired).
   455→  const { requestApply, dialogs } = useApplyFlow();
   456→
   457→  const load = useCallback(async (silent: boolean) => {
   458→    try {
   459→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   460→      const json = JSON.stringify(data);
   461→      if (!silent || json !== lastJson.current) {
   462→        lastJson.current = json;
   463→        setJobs(data);
   464→      }
   465→      setError(null);
   466→    } catch (e) {
   467→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   468→    }
   469→  }, []);
   470→
   471→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   472→  useEffect(() => {
   473→    void load(false);
   474→    const timer = setInterval(() => {
   475→      if (!document.hidden) void load(true);
   476→    }, 30_000);
   477→    const onFocus = () => void load(true);
   478→    window.addEventListener("focus", onFocus);
   479→    return () => {
   480→      clearInterval(timer);
   481→      window.removeEventListener("focus", onFocus);
   482→    };
   483→  }, [load]);
   484→
   485→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   486→  useEffect(() => {
   487→    const mq = window.matchMedia("(min-width: 1024px)");
   488→    const sync = () => setIsDesktop(mq.matches);
   489→    sync();
   490→    mq.addEventListener("change", sync);
   491→    return () => mq.removeEventListener("change", sync);
   492→  }, []);
   493→
   494→  const openJobs = useMemo(
   495→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   496→    [jobs]
   497→  );
   498→
   499→  const selectedJob = useMemo(
   500→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   501→    [openJobs, selectedId]
   502→  );
   503→
   504→  // Desktop opens with the first posting pre-selected (reference-board
   505→  // behavior); a stale selection after a poll snaps back to the top card.
   506→  useEffect(() => {
   507→    if (isDesktop && selectedId === null && openJobs.length > 0) {
   508→      setSelectedId(openJobs[0].id);
   509→    }
   510→  }, [isDesktop, openJobs, selectedId]);
   511→
   512→  useEffect(() => {
   513→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   514→      setSelectedId(openJobs[0]?.id ?? null);
   515→    }
   516→  }, [openJobs, selectedId]);
   517→
   518→  // Reference-board behavior — a freshly picked posting opens from the top,
   519→  // even when the previous posting was scrolled deep into its body.
   520→  useEffect(() => {
   521→    if (paneRef.current) paneRef.current.scrollTop = 0;
   522→  }, [selectedId]);
   523→
   524→  const select = useCallback(
   525→    (id: number) => {
   526→      setSelectedId(id);
   527→      if (!window.matchMedia("(min-width: 1024px)").matches) {

   531→    []

   533→
   534→  const saveAndToast = useCallback(

   537→      toggleSave(job.id);
   538→      toast.success(

   540→          ? "Removed from your saved jobs"
   541→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   542→      );
   543→    },
   544→    [saved, toggleSave]
   545→  );
   546→
   547→  const showList = isDesktop || selectedId === null;
   548→  const showDetail = isDesktop || selectedId !== null;
   549→

   553→
   554→      <main className="mx-auto w-full max-w-[1600px] flex-1 px-4 pb-10 pt-6 sm:px-6 lg:flex lg:min-h-0 lg:flex-col lg:px-8">

   583→        {error !== null ? (
