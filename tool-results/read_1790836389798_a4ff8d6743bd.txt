   478→  const [sortBy, setSortBy] = useState<"recent" | "deadline" | "applications">("recent");
   479→  const [page, setPage] = useState(1);
   480→  const [formOpen, setFormOpen] = useState(false);
   481→  const [editJob, setEditJob] = useState<JobWire | null>(null);
   482→  const [spin, setSpin] = useState(false);
   483→
   484→  const openCreate = () => {
   485→    setEditJob(null);
   486→    setFormOpen(true);
   487→  };
   488→  const openEdit = (j: JobWire) => {
   489→    setEditJob(j);
   490→    setFormOpen(true);
   491→  };
   492→
   493→  const load = useCallback(async (silent = false) => {
   494→    if (!silent) setError(null);
   495→    try {
   496→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   497→      setJobs(data);
   498→    } catch (e) {
   499→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load job postings");
   500→    }
   501→  }, []);
   502→
   503→  useEffect(() => {
   504→    void load();
   505→  }, [load]);
   506→
   507→  // 30 s silent poll + refetch-on-focus.
   508→  useEffect(() => {
   509→    const id = setInterval(() => {
   510→      if (!document.hidden) void load(true);
   511→    }, 30_000);
   512→    return () => clearInterval(id);
   513→  }, [load]);
   514→  useEffect(() => {
   515→    const onFocus = () => void load(true);
   516→    window.addEventListener("focus", onFocus);
   517→    document.addEventListener("visibilitychange", onFocus);
   518→    return () => {
   519→      window.removeEventListener("focus", onFocus);
   520→      document.removeEventListener("visibilitychange", onFocus);
   521→    };
   522→  }, [load]);
   523→
   524→  // Debounced search (350 ms).
   525→  useEffect(() => {
   526→    const t = setTimeout(() => {
   527→      setDebouncedSearch(search.trim().toLowerCase());
   528→      setPage(1);
   529→    }, 350);
   530→    return () => clearTimeout(t);
   531→  }, [search]);
   532→
   533→  const filtered = useMemo(() => {
   534→    let rows = jobs ?? [];
   535→    if (debouncedSearch) {
   536→      rows = rows.filter((j) => {
   537→        const hay = [
   538→          j.title,
   539→          j.position?.itemNumber,
   540→          j.position?.placeOfAssignment,
   541→          j.positionType,
   542→        ]
   543→          .filter(Boolean)
   544→          .join(" ")
   545→          .toLowerCase();
   546→        return hay.includes(debouncedSearch);
   547→      });
   548→    }
   549→    if (statusFilter === "open") rows = rows.filter(isOpenJob);
   550→    else if (statusFilter === "closed") rows = rows.filter((j) => !isOpenJob(j));
   551→    const sorted = [...rows];
   552→    if (sortBy === "recent") {
   553→      sorted.sort((a, b) => new Date(b.publishedAt).getTime() - new Date(a.publishedAt).getTime());
   554→    } else if (sortBy === "deadline") {
   555→      sorted.sort((a, b) => {
   556→        const av = a.deadlineDate ? new Date(a.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   557→        const bv = b.deadlineDate ? new Date(b.deadlineDate).getTime() : Number.POSITIVE_INFINITY;
   558→        return av - bv;
   559→      });
   560→    } else {
   561→      sorted.sort((a, b) => b.applicationCount - a.applicationCount);
   562→    }
   563→    return sorted;
   564→  }, [jobs, debouncedSearch, statusFilter, sortBy]);
   565→
   566→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   567→  const safePage = Math.min(page, totalPages);
   568→  const pageRows = filtered.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE);
   569→
   570→  const manualRefresh = useCallback(() => {
   571→    setSpin(true);
   572→    void load(true).finally(() => setTimeout(() => setSpin(false), 500));
   573→  }, [load]);
   574→
   575→  return (
   576→    <div className="space-y-6">
   577→      {/* Enterprise page header (PageHeader) — shared anatomy across all views */}
   578→      <PageHeader
   579→        title="Recruitment"
   580→        description={
   581→          <span className="num">
   582→            {filtered.length} posting{filtered.length === 1 ? "" : "s"}
   583→          </span>
   584→        }
   585→        actions={
   586→          <>
   587→            <button type="button" className={ghostBtn} onClick={manualRefresh}>
   588→              <RefreshCw className={`h-4 w-4 ${spin ? "animate-spin" : ""}`} aria-hidden />
   589→              Refresh
   590→            </button>
   591→            <button type="button" className={ctaBtn} onClick={openCreate}>
   592→              <Plus className="h-4 w-4" aria-hidden />
   593→              Create Job
   594→            </button>
   595→          </>
   596→        }
   597→      />
   598→
   599→      {/* Filter toolbar — white card row: search grows, capsule selects fixed */}
   600→      <div className="dlg-card-plain flex flex-col gap-3 p-3 sm:flex-row sm:items-center sm:p-4">
   601→        <div className="flex min-w-0 flex-1 items-center gap-3">
   602→          <IconChip icon={Search} tone="slate" size={40} className="shrink-0 max-sm:hidden" />
   603→          <Input
   604→            className="dlg-input min-h-[44px] flex-1 pl-4"
   605→            placeholder="Search title, item no., or place…"
   606→            value={search}
   607→            onChange={(e) => setSearch(e.target.value)}
   608→            aria-label="Search job postings"
   609→          />
   610→        </div>
   611→        <Select
   612→          value={statusFilter}
   613→          onValueChange={(v) => {
   614→            setStatusFilter(v as typeof statusFilter);
   615→            setPage(1);
   616→          }}
   617→        >
   618→          <SelectTrigger className="dlg-input min-h-[44px] w-full sm:w-44" aria-label="Status filter">
   619→            <SelectValue />
   620→          </SelectTrigger>
   621→          <SelectContent>
   622→            <SelectItem value="all">All statuses</SelectItem>
   623→            <SelectItem value="open">Open</SelectItem>
   624→            <SelectItem value="closed">Closed</SelectItem>
   625→          </SelectContent>
   626→        </Select>
   627→        <Select
   628→          value={sortBy}
   629→          onValueChange={(v) => {
   630→            setSortBy(v as typeof sortBy);
   631→            setPage(1);
   632→          }}
   633→        >
   634→          <SelectTrigger className="dlg-input min-h-[44px] w-full sm:w-44" aria-label="Sort">
   635→            <SelectValue />
   636→          </SelectTrigger>
   637→          <SelectContent>
   638→            <SelectItem value="recent">Recently posted</SelectItem>
   639→            <SelectItem value="deadline">Deadline</SelectItem>
   640→            <SelectItem value="applications">Applications</SelectItem>
   641→          </SelectContent>
   642→        </Select>
   643→      </div>
   644→
   645→      {error ? (
   646→        <div className="dlg-card p-8 text-center space-y-4">
   647→          <AlertTriangle className="h-8 w-8 text-dusty-rose mx-auto" aria-hidden />
   648→          <p className="text-sm text-stone">{error}</p>
   649→          <button type="button" className={ghostBtn} onClick={() => void load()}>
   650→            Retry
   651→          </button>
   652→        </div>
   653→      ) : jobs === null ? (
   654→        <div className="dlg-card p-6">
   655→          <SkeletonRows rows={6} rowClassName="h-14" />
   656→        </div>
   657→      ) : pageRows.length === 0 ? (
   658→        <div className="dlg-card p-6">
   659→          <EmptyState
   660→            icon={Briefcase}
   661→            tone="amber"
   662→            title="No postings found"
   663→            description="No job postings match the current filters."
   664→            action={
   665→              <button type="button" className={ctaBtn} onClick={openCreate}>
   666→                <Plus className="h-4 w-4" aria-hidden />
   667→                Create Job
   668→              </button>
   669→            }
   670→          />
   671→        </div>
   672→      ) : (
   673→        <div className="dlg-card-plain table-card overflow-hidden">
   674→          <div className="flex items-center gap-3 border-b border-[var(--border)] px-5 py-4">
   675→            <IconChip icon={Briefcase} tone="amber" size={36} iconSize={16} />
   676→            <h2 className="text-base font-semibold tracking-[-0.01em] text-ink">Job postings</h2>
   677→            <span className="dlg-pill num ml-auto bg-fog px-2.5 py-1 text-xs font-medium text-stone">
   678→              {filtered.length} total
   679→            </span>
   680→          </div>
   681→          <div className="overflow-x-auto scroll-thin">
   682→            <table className="w-full min-w-[640px] text-sm">
   683→              <thead>
   684→                <tr className="border-b border-[var(--border)] text-left text-xs font-medium uppercase tracking-[0.08em] text-stone">
   685→                  <th className="px-5 py-3.5">Position</th>
   686→                  <th className="px-4 py-3.5">Vacancies</th>
   687→                  <th className="px-4 py-3.5">Monthly salary</th>
   688→                  <th className="px-4 py-3.5">Applications</th>
   689→                  <th className="px-4 py-3.5">Deadline</th>
   690→                  <th className="px-4 py-3.5">Status</th>
   691→                  <th className="px-5 py-3.5 text-right">
   692→                    <span className="sr-only">Quick actions</span>
   693→                  </th>
   694→                </tr>
   695→              </thead>
   696→              <tbody>
   697→                {pageRows.map((j) => {
   698→                  const dl = deadlineState(j.deadlineDate);
   699→                  const urgent = isOpenJob(j) && dl.closingSoon;
   700→                  return (
   701→                    <tr
   702→                      key={j.id}
   703→                      className="group/row cursor-pointer border-b border-[var(--border)] transition-colors last:border-0 hover:bg-fog/60"

   705→                    >
   706→                      <td className="px-5 py-3.5">
   707→                        <div className="flex items-center gap-3">
   708→                          <span aria-hidden="true" className="contents">
   709→                            <BrandLogo logoSize={20} />
   710→                          </span>
   711→                          <div className="min-w-0">
   712→                            <p className="truncate text-sm font-medium text-ink">{j.title}</p>
   713→                            <p className="mt-0.5 text-xs text-stone">
   714→                              {[j.position?.itemNumber, j.position?.placeOfAssignment, j.position?.division].filter(Boolean).join(" · ") || "—"}
   715→                            </p>
   716→                          </div>
   717→                        </div>
   718→                      </td>
   719→                      <td className="num px-4 py-3.5 font-medium text-stone">{j.numberOfVacancy ?? "—"}</td>
   720→                      <td className="num px-4 py-3.5 font-medium text-stone">
   721→                        {j.position?.salaryAmount != null ? formatCurrency(j.position.salaryAmount) : "—"}
   722→                      </td>
   723→                      <td className="num px-4 py-3.5 text-stone">{j.applicationCount}</td>
   724→                      <td className={`num px-4 py-3.5 ${urgent ? "text-[var(--bad)]" : "text-stone"}`}>
   725→                        <span className="inline-flex items-center gap-1.5">
   726→                          {urgent ? <Clock className="h-3 w-3" aria-hidden /> : null}
   727→                          {formatDate(j.deadlineDate)}
   728→                        </span>
   729→                      </td>
   730→                      <td className="px-4 py-3.5">
   731→                        <JobStatusPill job={j} />
   732→                      </td>
   733→                      <td className="px-5 py-3.5">
   734→                        <div className="flex justify-end gap-1.5 opacity-0 transition-opacity focus-within:opacity-100 group-hover/row:opacity-100 max-lg:opacity-100">
   735→                          <button
   736→                            type="button"

   738→                            aria-label={`View ${j.title}`}
   739→                            title="View"
   740→                            onClick={(e) => {
   741→                              e.stopPropagation();
   742→                              navigate("job", { id: String(j.id) });
   743→                            }}
   744→                          >
   745→                            <Eye className="h-4 w-4" aria-hidden />
   746→                          </button>
   747→                          <button
   748→                            type="button"
   749→                            className={iconBtn}
   750→                            aria-label={`Edit ${j.title}`}
   751→                            title="Edit"
   752→                            onClick={(e) => {
   753→                              e.stopPropagation();
   754→                              openEdit(j);
   755→                            }}
   756→                          >
   757→                            <Pencil className="h-4 w-4" aria-hidden />
   758→                          </button>
   759→                        </div>
   760→                      </td>
   761→                    </tr>
   762→                  );
   763→                })}
   764→              </tbody>
   765→            </table>
   766→          </div>
   767→        </div>
   768→      )}
   769→
   770→      {/* Pagination */}
   771→      {jobs !== null && filtered.length > 0 && (
   772→        <div className="flex items-center justify-between gap-3">
   773→          <p className="num text-xs text-body-gray">
   774→            {(safePage - 1) * PAGE_SIZE + 1}–{Math.min(safePage * PAGE_SIZE, filtered.length)} of {filtered.length}
   775→          </p>
   776→          <div className="flex items-center gap-2">
   777→            <button type="button" className={ghostBtn + " min-h-[36px] px-3 text-xs"} disabled={safePage <= 1} onClick={() => setPage(safePage - 1)}>
   778→              Previous
   779→            </button>
   780→            <span className="num text-xs text-stone">
   781→              Page {safePage} of {totalPages}
   782→            </span>
   783→            <button type="button" className={ghostBtn + " min-h-[36px] px-3 text-xs"} disabled={safePage >= totalPages} onClick={() => setPage(safePage + 1)}>
   784→              Next
   785→            </button>
   786→          </div>
   787→        </div>
   788→      )}
   789→
   790→      <JobFormDialog open={formOpen} onOpenChange={setFormOpen} job={editJob} onSaved={() => void load(true)} />
   791→    </div>
   792→  );
