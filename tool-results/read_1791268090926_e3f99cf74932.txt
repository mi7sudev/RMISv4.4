    83→/** Client-persisted bookmark list for the frontpage explorer. */
    84→function useSavedJobs() {
    85→  const [saved, setSaved] = useState<number[]>(() => {
    86→    if (typeof window === "undefined") return [];
    87→    try {
    88→      const raw = window.localStorage.getItem(SAVED_KEY);
    89→      const parsed = raw ? (JSON.parse(raw) as unknown) : [];
    90→      return Array.isArray(parsed) ? parsed.filter((x): x is number => typeof x === "number") : [];
    91→    } catch {
    92→      return [];
    93→    }
    94→  });
    95→
    96→  const toggle = useCallback((id: number) => {
    97→    setSaved((prev) => {
    98→      const next = prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id];
    99→      try {
   100→        window.localStorage.setItem(SAVED_KEY, JSON.stringify(next));
   101→      } catch {
   102→        /* storage unavailable — in-session state still works */
   103→      }
   104→      return next;
   105→    });
   106→  }, []);
   107→
   108→  return { saved, toggle };
   109→}

   290→/** Plain-text posting section rendered as a disc bullet list. */
   291→function TextBullets({ text }: { text: string }) {
   292→  const items = text
   293→    .split(/\r?\n/)
   294→    .map((l) => l.replace(/^[-•*]\s*/, "").trim())
   295→    .filter(Boolean);
   296→  if (items.length === 0) return null;
   297→  return (
   298→    <ul className="list-disc space-y-2 pl-5 text-base leading-relaxed text-stone">
   299→      {items.map((item, i) => (
   300→        <li key={i}>{item}</li>
   301→      ))}
   302→    </ul>
   303→  );
   304→}
   305→
   306→/** Posting section: sanitized rich HTML when present, else text bullets. */
   307→function PostingSection({ title, html, text }: { title: string; html: string | null; text: string | null }) {
   308→  if (html && html.trim()) {
   309→    return (
   310→      <section className="space-y-3">
   311→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   312→        <div className="rich-text text-base leading-relaxed text-stone" dangerouslySetInnerHTML={{ __html: html }} />
   313→      </section>
   314→    );
   315→  }
   316→  if (text && text.trim()) {
   317→    return (
   318→      <section className="space-y-3">
   319→        <h3 className="text-lg font-semibold tracking-[-0.01em] text-ink">{title}</h3>
   320→        <TextBullets text={text} />
   321→      </section>
   322→    );
   323→  }
   324→  return null;
   325→}

   337→  onSave: () => void;

   345→

   456→export default function PublicLanding() {
   457→  const [jobs, setJobs] = useState<JobWire[] | null>(null);
   458→  const [error, setError] = useState<string | null>(null);
   459→  const [selectedId, setSelectedId] = useState<number | null>(null);
   460→  const [expandedId, setExpandedId] = useState<number | null>(null);
   461→  const [isDesktop, setIsDesktop] = useState(false);
   462→  const lastJson = useRef("");
   463→  const paneRef = useRef<HTMLDivElement>(null);
   464→  const { saved, toggle: toggleSave } = useSavedJobs();
   465→  // Shared apply gate — anonymous visitors get the sign-in prompt, complete
   466→  // applicants get the confirm flow (same as the board), incomplete profiles
   467→  // are sent to finish their profile first (the fast-track was retired).
   468→  const { requestApply, dialogs } = useApplyFlow();
   469→
   470→  const load = useCallback(async (silent: boolean) => {
   471→    try {
   472→      const data = await apiFetch<JobWire[]>("/api/jobs?limit=200");
   473→      const json = JSON.stringify(data);
   474→      if (!silent || json !== lastJson.current) {
   475→        lastJson.current = json;
   476→        setJobs(data);
   477→      }
   478→      setError(null);
   479→    } catch (e) {
   480→      if (!silent) setError(e instanceof Error ? e.message : "Failed to load positions");
   481→    }
   482→  }, []);
   483→
   484→  // Mount + 30s silent poll + refetch on focus (spec §13 cadence).
   485→  useEffect(() => {
   486→    void load(false);
   487→    const timer = setInterval(() => {
   488→      if (!document.hidden) void load(true);
   489→    }, 30_000);
   490→    const onFocus = () => void load(true);
   491→    window.addEventListener("focus", onFocus);
   492→    return () => {
   493→      clearInterval(timer);
   494→      window.removeEventListener("focus", onFocus);
   495→    };
   496→  }, [load]);
   497→
   498→  // Breakpoint sync — desktop keeps both panes; mobile swaps list ↔ detail.
   499→  useEffect(() => {
   500→    const mq = window.matchMedia("(min-width: 1024px)");
   501→    const sync = () => setIsDesktop(mq.matches);
   502→    sync();
   503→    mq.addEventListener("change", sync);
   504→    return () => mq.removeEventListener("change", sync);
   505→  }, []);
   506→
   507→  const openJobs = useMemo(
   508→    () => (jobs ?? []).filter((j) => !deadlineState(j.deadlineDate).overdue),
   509→    [jobs]
   510→  );
   511→
   512→  const selectedJob = useMemo(
   513→    () => openJobs.find((j) => j.id === selectedId) ?? null,
   514→    [openJobs, selectedId]
   515→  );
   516→
   517→  // Desktop opens with the first posting pre-selected (reference-board
   518→  // behavior); a stale selection after a poll snaps back to the top card.
   519→  useEffect(() => {
   520→    if (isDesktop && selectedId === null && openJobs.length > 0) {
   521→      setSelectedId(openJobs[0].id);
   522→    }
   523→  }, [isDesktop, openJobs, selectedId]);
   524→
   525→  useEffect(() => {
   526→    if (selectedId !== null && !openJobs.some((j) => j.id === selectedId)) {
   527→      setSelectedId(openJobs[0]?.id ?? null);
   528→    }
   529→  }, [openJobs, selectedId]);
   530→
   531→  // Reference-board behavior — a freshly picked posting opens from the top,
   532→  // even when the previous posting was scrolled deep into its body.
   533→  useEffect(() => {
   534→    if (paneRef.current) paneRef.current.scrollTop = 0;
   535→  }, [selectedId]);
   536→
   537→  const select = useCallback(
   538→    (id: number) => {
   539→      setSelectedId(id);
   540→      if (!window.matchMedia("(min-width: 1024px)").matches) {
   541→        window.scrollTo({ top: 0, behavior: "smooth" });
   542→      }
   543→    },
   544→    []
   545→  );
   546→
   547→  const saveAndToast = useCallback(
   548→    (job: JobWire) => {
   549→      const wasSaved = saved.includes(job.id);
   550→      toggleSave(job.id);
   551→      toast.success(
   552→        wasSaved
   553→          ? "Removed from your saved jobs"
   554→          : `Saved ${humanize(job.position?.positionTitle || job.title)}`
   555→      );
   556→    },
   557→    [saved, toggleSave]
   558→  );
   559→
   560→  const showList = isDesktop || selectedId === null;
   561→  const showDetail = isDesktop || selectedId !== null;
   562→
   563→  return (
   564→    // Viewport-adaptive board (no app pinning): a normal min-h-screen flex
   565→    // column whose footer always sits at the real bottom — pinned to the
   566→    // viewport on short pages (mt-auto), pushed down naturally on tall ones.
   567→    // Only the posting pane is sticky (viewport-capped); the register is a
   568→    // plain block that scrolls with the page — see the grid below.
   569→    <div className="flex min-h-screen flex-col">
   570→      <SiteHeader />
   571→
   572→      <main className="mx-auto w-full max-w-[1600px] flex-1 px-4 pb-10 pt-6 sm:px-6 lg:px-8">
   573→        {/* Hero — Bevel "Hero Sky" field (rounded 32) anchoring the explorer:
   574→            display headline, then the official brand mark + agency name as
   575→            the quiet institutional proof strip, then the single charcoal
   576→            capsule CTA (real marks only). */}
   577→        <section className="hero-sky rounded-[32px] p-6 sm:p-8 lg:px-10 lg:py-10 lg:[@media(max-height:860px)]:py-8">
   578→          {/* QA 2026-02: header renamed to "Job Opportunities" to match the
   579→              Positions tab; the live "N open" pill was removed — it read as
   580→              an unclickable button. */}
   581→          <h1 className="text-display-hero">Job Opportunities</h1>
   582→          {/* Institutional proof — the same emblems the
   583→              footer carries, moved into the hero as the trust anchor. */}
   584→          <div className="mt-5 flex items-center gap-4">
   585→            <BrandLogo logoSize={36} tone="light" />
   586→            <p className="min-w-0 text-sm font-semibold leading-snug text-ink">
   587→              Metals Industry Research and Development Center
   588→            </p>
   589→          </div>
   590→          <div className="mt-7">
   591→            <button
   592→              onClick={() => navigate("vacant-positions")}
   593→              className="dlg-cta text-nav inline-flex min-h-[44px] items-center gap-2 px-4 font-medium"
   594→            >
   595→              Browse the full board
   596→              <ArrowRight className="h-4 w-4" aria-hidden="true" />
   597→            </button>
   598→          </div>
   599→        </section>
   600→
   601→        {error !== null ? (
   602→          <div className="mt-6 rounded-2xl bg-[var(--bad-bg)] p-5">
   603→            <p className="text-sm text-ink">{error}</p>
   604→            <button
   605→              onClick={() => void load(false)}
   606→              className="mt-2 text-sm text-metric-blue underline underline-offset-2"
   607→            >
   608→              Try again
   609→            </button>
   610→          </div>
   611→        ) : jobs === null ? (
   612→          <div className="mt-6 space-y-3.5" aria-hidden="true">
   613→            {[0, 1, 2].map((i) => (
   614→              <div key={i} className="skel h-28" />
   615→            ))}
   616→          </div>
   617→        ) : openJobs.length === 0 ? (
   618→          <div className="dlg-card mt-6">
   619→            <EmptyState
   620→              icon={Inbox}
   621→              title="No open positions right now"
   622→              description="Check back soon or explore the full board for upcoming announcements."
   623→            />
   624→            <div className="pb-8 text-center">
   625→              <button
   626→                onClick={() => navigate("vacant-positions")}
   627→                className="inline-flex min-h-[44px] items-center gap-2 text-sm text-metric-blue underline underline-offset-2 hover:text-ink"
   628→              >
   629→                Go to the full positions board
   630→                <ArrowRight className="h-4 w-4" aria-hidden="true" />
   631→              </button>
   632→            </div>
   633→          </div>
   634→        ) : (
   635→          // Viewport-adaptive two-pane board. The posting pane sizes to its
   636→          // own content, capped at the LIVE viewport height (100dvh − 88px
   637→          // sticky offset − 24px breathing room — no fixed pixel constant):
   638→          //   • tall screens — the complete description renders in full with
   639→          //     no internal scrolling; the footer only ever sits BELOW it;
   640→          //   • short screens — the pane caps at the available space and
   641→          //     scrolls internally while pinned under the header, and the
   642→          //     grid-row clamp keeps it above the footer at every position.
   643→          // Sticky needs TRAVEL ROOM: a sticky element only stays put while
   644→          // its containing row is TALLER than itself, and the row height is
   645→          // max(register, posting). When the register is shorter than the
   646→          // posting (few vacancies — e.g. the sandbox's two cards), the row
   647→          // would equal the pane's height and the pane could never pin — it
   648→          // would just scroll away with the page. So the row reserves at
   649→          // least 100dvh + 128px (= the pane's cap + a 240px pin phase on
   650→          // cap-limited screens): the pane rides up with the hero, pins at
   651→          // top-88 for ~240px of scroll, then yields only as the footer
   652→          // rises. With a tall register the row is the register's natural
   653→          // height and this floor is inert.
   654→          // The job list is deliberately NOT sticky — a plain block that
   655→          // scrolls with the page (user clarification).
   656→          <div className="mt-6 grid gap-8 sm:mt-8 lg:min-h-[calc(100dvh+128px)] lg:grid-cols-[minmax(340px,430px)_minmax(0,1fr)]">
   657→            {/* Left · the register — plain block, natural height, scrolls
   658→                with the page (NOT sticky, per user clarification). */}
   659→            {showList && (
   660→              <div
   661→                aria-label="Job opportunities list"
   662→                className="scroll-thin space-y-3.5"
   663→              >
   664→                {openJobs.map((job) => (
   665→                  <JobListCard
   666→                    key={job.id}
   667→                    job={job}
   668→                    selected={selectedId === job.id}
   669→                    expanded={expandedId === job.id}
   670→                    saved={saved.includes(job.id)}
   671→                    onSelect={() => select(job.id)}
   672→                    onToggle={() => setExpandedId(expandedId === job.id ? null : job.id)}
   673→                    onSave={() => saveAndToast(job)}
   674→                  />
   675→                ))}
   676→              </div>
   677→            )}
   678→
   679→            {/* Right · the posting — ONE sticky, viewport-capped scroller
   680→                (paneRef sits on the scroller itself, so the select-reset
   681→                actually zeroes it). Its natural height is the full posting,
   682→                capped at the live viewport: complete content with no inner
   683→                scroll when the screen has room; capped + internally
   684→                scrollable when it doesn't. Sticky keeps it in view while the
   685→                list scrolls with the page; the grid-row clamp keeps it above
   686→                the footer. On mobile it is a plain block (single-pane swap,
   687→                page scroll). */}
   688→            {showDetail && (
   689→              <div
   690→                ref={paneRef}
   691→                className="scroll-thin lg:sticky lg:top-[88px] lg:self-start lg:max-h-[calc(100dvh-112px)] lg:overflow-y-auto lg:pr-2"
   692→              >
   693→                {selectedJob ? (
   694→                  <div className="dlg-card p-6 sm:p-8">
   695→                    <JobDetail
   696→                      job={selectedJob}
   697→                      saved={saved.includes(selectedJob.id)}
   698→                      onApply={() => requestApply(selectedJob)}
   699→                      onSave={() => saveAndToast(selectedJob)}
   700→                      onBack={() => setSelectedId(null)}
   701→                    />
   702→                  </div>
   703→                ) : (
   704→                  <div className="dlg-card">
   705→                    <EmptyState
   706→                      icon={Briefcase}
   707→                      title="Select a position"
   708→                      description="Choose a role from the list to read the full posting."
   709→                    />
   710→                  </div>
   711→                )}
   712→              </div>
   713→            )}
   714→          </div>
   715→        )}
   716→      </main>
   717→
   718→      <Footer />
   719→      {/* Shared apply dialog (confirm / MQR / done), mounted once. */}
   720→      {dialogs}
   721→    </div>
   722→  );
   723→}
